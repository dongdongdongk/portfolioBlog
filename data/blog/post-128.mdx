```jsx
// Search Area Dept Popup
this.div_search_btn_dept_onclick = function(obj:nexacro.Button, e:nexacro.ClickEventInfo)
{
    // 부서 검색 팝업을 열기 위한 이벤트 핸들러입니다.
	// nLeft와 nTop은 팝업 창을 화면에서 원하는 위치에 표시하기 위한 좌표값으로 사용됩니다. 팝업 창을 특정 위치에 띄우고 싶을 때 이러한 계산을 통해 팝업 창의 위치를 조절할 수 있습니다.
	var nLeft = system.clientToScreenX(this, 10);
	var nTop = system.clientToScreenY(this, 10);

    // 팝업 창 설정
    var objChild = new ChildFrame("popDept", 0, 0, 300, 400); // 팝업 창을 생성하고 변수 objChild에 할당합니다.
    objChild.set_formurl("Popup::Pop_DeptSearch.xfdl"); // 팝업 창의 내용을 정의한 XFDL 파일의 경로 또는 URL을 설정합니다.
    objChild.set_openalign("center middle"); // 팝업 창을 화면에서 가운데에 위치시킵니다.
    objChild.set_dragmovetype("all"); // 팝업 창을 어디서든 드래그하여 이동할 수 있도록 설정합니다.

    // 팝업 창을 모달로 표시하고 결과를 콜백 함수로 받습니다.
    objChild.showModal(this.getOwnerFrame(),  // 첫 번째 인자: 팝업 창의 부모 프레임(Frame)을 지정합니다.
                     {}, // 두 번째 인자: 팝업 창에 대한 옵션을 설정하는 빈 객체입니다.
                     this,  // 세 번째 인자: 팝업 창을 호출한 Form 객체를 지정합니다.
                     "fn_callback_pop");  // 네 번째 인자: 팝업 창이 닫힐 때 호출될 콜백 함수의 이름을 지정합니다.
};
```


```jsx
// Dept Popup Callback
// 부서 검색 팝업의 결과를 처리하는 콜백 함수입니다.
this.fn_callback_pop = function (sPopupId, sReturn) {
  // sReturn이 정의되지 않았을 경우 빈 문자열로 초기화합니다.
  if (sReturn == undefined) {
    sReturn = ''
  }
  // sPopupId가 "popDept"와 일치하는 경우에만 실행합니다.
  if (sPopupId == 'popDept') {
    // sReturn 문자열을 "|" 기준으로 분할하여 배열로 저장합니다.
    if (sReturn.length > 0) {
      var arrRtn = sReturn.split('|')
      // 검색 결과에서 첫 번째 항목을 부서 코드 입력란에 설정합니다.
      this.div_search.form.edt_dept_cd.set_value(arrRtn[0])
      // 검색 결과에서 두 번째 항목을 부서명 입력란에 설정합니다.
      this.div_search.form.edt_dept_nm.set_value(arrRtn[1])
    }
  }
}
```

```jsx
//부서 검색 화면에서 리턴 보내주는 스크립트
// fn_ok 함수 정의
this.fn_ok = function () {
  // 현재 선택한 행의 "DEPT_CD" 컬럼 값을 가져와 sRtn 변수에 할당합니다.
  var sRtn = this.ds_dept.getColumn(this.ds_dept.rowposition, 'DEPT_CD') + '|'

  // 현재 선택한 행의 "DEPT_NAME" 컬럼 값을 sRtn 변수에 추가합니다.
  sRtn += this.ds_dept.getColumn(this.ds_dept.rowposition, 'DEPT_NAME')

  // 부모 팝업 창을 닫고, sRtn 값을 반환하여 부모 창으로 결과를 전달합니다.
  this.close(sRtn)
}
```
