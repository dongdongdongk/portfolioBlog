## Input.jsx

```jsx
import React from 'react'

import './Input.css'

// Input 컴포넌트는 다양한 입력 요소를 렌더링합니다.
const Input = (props) => {
  // 입력 요소의 종류에 따라 input 또는 textarea를 선택하여 렌더링합니다.
  const element =
    props.element === 'input' ? (
      <input id={props.id} type={props.type} placeholder={props.placeholder} />
    ) : (
      <textarea id={props.id} rows={props.rows || 3} />
    )

  // 입력 요소를 감싸는 form-control 클래스를 가진 div 요소를 반환합니다.
  return (
    <div className={`form-control`}>
      {/* 입력 요소와 연결된 레이블을 표시합니다. */}
      <label htmlFor={props.id}>{props.label}</label>
      {/* 선택한 입력 요소를 렌더링합니다. */}
      {element}
    </div>
  )
}

export default Input
```

## **`reducer` 함수:**

```jsx
reducer 함수는 두 가지 주된 특징을 가지고 있어야 한다:

현재 상태와 액션을 받아 새로운 상태를 반환하는 순수 함수여야 한다. 이를 좀 더 상세하게 설명해보겠다.

현재 상태와 액션을 받아 새로운 상태를 반환하는 함수:

reducer 함수는 현재 어플리케이션 상태와 어떤 액션이 발생했는지에 대한 정보를 받아서 새로운 상태를 반환하는 역할을 한다.

이는 함수의 시그니처가 다음과 같다:

const reducer = (currentState, action) => {
  switch (action.type) {
    // 각각의 액션에 대한 상태 업데이트 로직 작성
    case 'SOME_ACTION':
      // ... 상태 업데이트 로직 ...
      return newState;

    // 다른 액션에 대한 상태 업데이트 로직 작성

    default:
      // 기본적으로는 현재 상태를 그대로 반환
      return currentState;
  }
};

currentState는 현재의 어플리케이션 상태이며, action은 어떤 변화를 일으킬지에 대한 정보가 담긴 액션 객체이다.

함수는 이 두 가지를 기반으로 새로운 상태를 계산하고 반환한다.

순수 함수의 특징을 가져야 함:

순수 함수란 두 가지 주된 특징을 갖추는 함수이다.

	1. 동일한 인자로 호출될 때 항상 동일한 결과를 반환한다:

	동일한 currentState와 action이 주어졌을 때, reducer 함수는 항상 같은 newState를 반환해야 한다.

	예측 가능하고 안정적인 동작을 위해 이 특징이 중요하다.

	2. 부작용이 없다:

	순수 함수는 외부 환경에 영향을 미치지 않으며, 다른 부작용(side effect)을 일으키지 않는다.

	외부 변수의 값을 변경하거나 API 호출 등의 작업을 수행하지 않아야 한다.
```


## **`dispatch` 함수:**


```jsx
액션 객체는 type 속성을 가지고 있어야 한다. 이 type 속성은 액션이 어떤 종류인지 나타내는 문자열이다.

리듀서 함수는 이 type 속성을 기반으로 어떤 상태 업데이트를 수행할지 결정한다.

type은 대개 대문자의 문자열로 작성되며, 어떤 액션이 발생했는지 명확하게 표현해야 한다.
```


```jsx
import React, { useReducer } from 'react'

import './Input.css'

// Input 컴포넌트는 다양한 입력 요소를 렌더링합니다.
const inputReducer = (state, action) => {
  // useReducer로 상태를 관리하는 함수입니다.
  switch (action.type) {
    case 'CHANGE':
      // CHANGE 액션일 경우, 새로운 상태를 반환합니다.
      return {
        ...state, // ...state: 현재 상태를 복사합니다.
        value: action.val, // 입력 값 업데이트
        isValid: true,
      }
    default:
      return state // 기본적으로 현재 상태를 그대로 반환합니다.
  }
}

const Input = (props) => {
  // useReducer를 사용하여 inputState 상태와 dispatch 함수를 얻습니다. 초기 상태는 { value: '', isValid: false }로 설정됩니다.
  const [inputState, dispatch] = useReducer(inputReducer, { value: '', isValid: false })

  const changeHandler = (event) => {
    // CHANGE 액션을 dispatch 하여 상태를 업데이트합니다.
    dispatch({ type: 'CHANGE', val: event.target.value })
  }

  // 입력 요소의 종류에 따라 input 또는 textarea를 선택하여 렌더링합니다.
  const element =
    props.element === 'input' ? (
      <input
        id={props.id}
        type={props.type}
        placeholder={props.placeholder}
        onChange={changeHandler}
        value={inputState.value}
      />
    ) : (
      <textarea
        id={props.id}
        rows={props.rows || 3}
        onChange={changeHandler}
        value={inputState.value}
      />
    )

  // 입력 요소를 감싸는 form-control 클래스를 가진 div 요소를 반환합니다.
  return (
    <div className={`form-control ${!inputState.isValid && 'form-control--invalid'}`}>
      {/* 입력 요소와 연결된 레이블을 표시합니다. */}
      <label htmlFor={props.id}>{props.label}</label>
      {/* 선택한 입력 요소를 렌더링합니다. */}
      {element}
      {!inputState.isValid && <p>{props.errorText}</p>}
    </div>
  )
}

export default Input
```

```jsx
1. **초기 상태 설정:**
   - `useReducer`를 이용하여 `inputReducer` 함수와 초기 상태 `{ value: '', isValid: false }`를 설정한다.

2. **액션 타입 및 리듀서 정의:**
   - `inputReducer` 함수는 `CHANGE` 액션을 처리하고, 입력 값 업데이트 및 유효성 검사를 수행한다.

3. **이벤트 핸들러 (`changeHandler`) 정의:**
   - 입력 값이 변경될 때 호출되는 이벤트 핸들러이다.
   - `dispatch` 함수를 통해 `CHANGE` 액션을 디스패치하고 입력 값에 대한 업데이트를 수행한다.

4. **입력 요소 선택 (`element` 변수):**
   - `props.element` 값에 따라 `input` 또는 `textarea`를 선택하여 렌더링한다.
   - 선택한 입력 요소는 이전에 정의한 이벤트 핸들러와 현재 `inputState`의 값을 연결한다.

5. **클래스 동적 추가 및 반환:**
   - `form-control` 클래스를 갖는 `div` 요소를 반환한다.
   - 유효성이 false인 경우 `form-control--invalid` 클래스를 추가하여 동적으로 스타일을 변경한다.

6. **레이블 및 오류 메시지 표시:**
   - 입력 요소와 연결된 레이블을 표시한다.
   - 유효성이 false인 경우 오류 메시지(`props.errorText`)를 표시한다.

7. **최종 반환:**
   - `form-control`을 갖는 div 요소와 그 안에 레이블, 선택한 입력 요소, 오류 메시지를 포함한 컴포넌트를 반환한다.
```

```jsx
// 유효성 검사 타입 상수 정의
const VALIDATOR_TYPE_REQUIRE = 'REQUIRE' // 필수 입력을 나타내는 유효성 검사 타입 상수

// 각 유효성 검사 타입에 대한 액션 생성자 함수 정의
export const VALIDATOR_REQUIRE = () => ({ type: VALIDATOR_TYPE_REQUIRE }) // 필수 입력 검사를 수행하는 액션 생성자 함수

export const VALIDATOR_FILE = () => ({ type: VALIDATOR_TYPE_FILE }) // 파일 입력을 나타내는 액션 생성자 함수

export const VALIDATOR_MINLENGTH = (val) => ({
  // 최소 길이 검사를 수행하는 액션 생성자 함수
  type: VALIDATOR_TYPE_MINLENGTH,
  val: val,
})

export const VALIDATOR_MAXLENGTH = (val) => ({
  // 최대 길이 검사를 수행하는 액션 생성자 함수
  type: VALIDATOR_TYPE_MAXLENGTH,
  val: val,
})

export const VALIDATOR_MIN = (val) => ({ type: VALIDATOR_TYPE_MIN, val: val }) // 최소 값 검사를 수행하는 액션 생성자 함수

export const VALIDATOR_MAX = (val) => ({ type: VALIDATOR_TYPE_MAX, val: val }) // 최대 값 검사를 수행하는 액션 생성자 함수

export const VALIDATOR_EMAIL = () => ({ type: VALIDATOR_TYPE_EMAIL }) // 이메일 형식 검사를 수행하는 액션 생성자 함수

// 값과 유효성 검사 목록을 받아 유효성 검사 수행하는 함수
export const validate = (value, validators) => {
  let isValid = true

  for (const validator of validators) {
    // 각 유효성 검사 타입에 따라 검사 수행
    if (validator.type === VALIDATOR_TYPE_REQUIRE) {
      isValid = isValid && value.trim().length > 0 // 필수 입력 검사
    }
    if (validator.type === VALIDATOR_TYPE_MINLENGTH) {
      isValid = isValid && value.trim().length >= validator.val // 최소 길이 검사
    }
    if (validator.type === VALIDATOR_TYPE_MAXLENGTH) {
      isValid = isValid && value.trim().length <= validator.val // 최대 길이 검사
    }
    if (validator.type === VALIDATOR_TYPE_MIN) {
      isValid = isValid && +value >= validator.val // 최소 값 검사
    }
    if (validator.type === VALIDATOR_TYPE_MAX) {
      isValid = isValid && +value <= validator.val // 최대 값 검사
    }
    if (validator.type === VALIDATOR_TYPE_EMAIL) {
      isValid = isValid && /^\S+@\S+\.\S+$/.test(value) // 이메일 형식 검사
    }
  }

  return isValid // 최종적으로 유효성 여부 반환
}
```

## **입력 컴포넌트 구현**

```jsx
Input.js 파일에서는 useReducer를 사용하여 입력 상태를 관리한다.

inputReducer 함수는 'CHANGE' 액션과 'TOUCH' 액션을 처리하여 입력 값의 변경 및 터치 여부를 감지한다.

( 이유는 처음부터 유효성 검사 실패가 뜨기 때문에 그걸 막아주기 위해서다 )

또한, 입력값이 변경될 때마다 유효성 검사를 수행하고 그 결과를 상태에 저장한다.
```

```jsx
import React, { useReducer } from 'react'

import { validate } from '../../util/validators'
import './Input.css'

// Input 컴포넌트는 다양한 입력 요소를 렌더링합니다.
const inputReducer = (state, action) => {
  // useReducer로 상태를 관리하는 함수입니다.
  switch (action.type) {
    case 'CHANGE':
      // CHANGE 액션일 경우, 새로운 상태를 반환합니다.
      return {
        ...state,
        value: action.val, // 입력 값 업데이트
        isValid: validate(action.val, action.validators), // 입력 값 유효성 검사
      }
    case 'TOUCH': {
      return {
        ...state,
        isTouched: true, // 터치 이벤트 발생 시, isTouched를 참으로 업데이트
      }
    }
    default:
      return state // 기본적으로 현재 상태를 그대로 반환합니다.
  }
}

const Input = (props) => {
  // useReducer를 사용하여 inputState 상태와 dispatch 함수를 얻습니다.
  const [inputState, dispatch] = useReducer(inputReducer, {
    value: '',
    isValid: false,
    isTouched: false,
  })

  const touchHandler = () => {
    dispatch({
      type: 'TOUCH', // 터치 이벤트 핸들러
    })
  }

  const changeHandler = (event) => {
    // CHANGE 액션을 dispatch 하여 상태를 업데이트합니다.
    dispatch({ type: 'CHANGE', val: event.target.value, validators: props.validators })
  }

  // 입력 요소의 종류에 따라 input 또는 textarea를 선택하여 렌더링합니다.
  const element =
    props.element === 'input' ? (
      <input
        id={props.id}
        type={props.type}
        placeholder={props.placeholder}
        onChange={changeHandler}
        onBlur={touchHandler} // 터치 이벤트 발생 시, 터치 핸들러 호출, 포커스를 벗어날 때
        value={inputState.value}
      />
    ) : (
      <textarea
        id={props.id}
        rows={props.rows || 3}
        onChange={changeHandler}
        onBlur={touchHandler} // 터치 이벤트 발생 시, 터치 핸들러 호출
        value={inputState.value}
      />
    )

  // 입력 요소를 감싸는 form-control 클래스를 가진 div 요소를 반환합니다.
  return (
    <div
      className={`form-control ${!inputState.isValid && inputState.isTouched && 'form-control--invalid'}`}
    >
      {/* 입력 요소와 연결된 레이블을 표시합니다. */}
      <label htmlFor={props.id}>{props.label}</label>
      {/* 선택한 입력 요소를 렌더링합니다. */}
      {element}
      {!inputState.isValid && inputState.isTouched && <p>{props.errorText}</p>}
      {/* 유효성 검사에 실패하고 터치 이벤트가 발생한 경우, 오류 메시지를 표시합니다. */}
    </div>
  )
}

export default Input
```

```jsx
inputReducer 함수:

	useReducer를 사용하여 상태를 관리하는 함수이다.

	'CHANGE' 액션이 발생하면 새로운 상태를 반환하며, 입력 값 업데이트와 함께 유효성 검사를 수행한다.

	'TOUCH' 액션이 발생하면 isTouched를 참으로 설정한다.


Input 컴포넌트:

	useReducer를 사용하여 입력 상태와 디스패치 함수를 얻는다.

	touchHandler 함수는 'TOUCH' 액션을 디스패치하여 터치 이벤트를 처리한다.

	changeHandler 함수는 'CHANGE' 액션을 디스패치하여 입력 값의 변경을 처리하고 유효성 검사를 수행한다.
```


## **Input 컴포넌트의 useEffect**

```jsx
NewPlace 컴포넌트에서 Input 컴포넌트를 사용하면서 useEffect를 활용하는 주된 목적은

입력값과 유효성에 대한 변화를 감지하고, 해당 변화가 일어날 때마다 특정 동작을 수행할 수 있도록 하는 것이다.

Input 컴포넌트의 useEffect는 컴포넌트가 렌더링된 후,

그리고 해당 컴포넌트에 의존성이 있는 데이터가 변경될 때마다 실행된다.

여기서는 id, value, isValid라는 세 가지 상태에 의존성을 두고 있다.

useEffect(() => {
  onInput(id, value, isValid);
}, [id, value, isValid]);
```

```jsx
onInput 함수 호출:

onInput 함수는 NewPlace 컴포넌트에서 전달된 함수로,

해당 함수에 id, value, isValid 값을 전달하여 호출한다.

이렇게 함으로써 입력 값과 유효성에 대한 변경사항을 외부 컴포넌트에 전달한다.

의존성 배열:

useEffect의 두 번째 인자로 제공되는 배열에는 이 효과가 의존하는 값들이 들어간다.
```


## **useCallback**

```jsx
우선, 함수형 컴포넌트에서는 컴포넌트가 리렌더링될 때마다 함수들이 새로 생성된다.

이 말은 간단한 함수라도, 렌더링 시에 계속해서 새로운 함수가 만들어진다는 것이다.

이게 문제가 될 때가 있는데, 특히 자식 컴포넌트에게 콜백 함수를 전달하거나, 의존성이 있는 함수를 다룰 때 성능상의 이슈가 발생할 수 있다.

여기서 useCallback이 등장한다.

이 훅을 사용하면 함수를 "기억"해두고, 함수가 필요한 곳에서 이전에 생성한 함수를 사용할 수 있게 된다.

예를 들어, 버튼 클릭에 따라 어떤 동작을 하는 함수가 있다고 상상해보자.

이 함수가 렌더링될 때마다 새로 생성되면, 그 함수를 props로 전달하는 자식 컴포넌트는 매번 새로운 함수를 받게 된다.

이런 상황에서 useCallback을 사용하면,

함수가 변경될 때만 새로운 함수를 생성하게끔 설정할 수 있다.

이렇게 하면 자식 컴포넌트는 새로운 함수를 받아야 하는 빈도를 줄일 수 있어서 성능 향상에 도움이 된다.
```

```jsx
import Button from '../../shared/components/FormElements/Button'

// 리듀서 함수를 정의합니다.
const formReducer = (state, action) => {
  switch (action.type) {
    // 입력값 변경 액션 처리
    case 'INPUT_CHANGE':
      let formIsValid = true
      // 모든 입력값에 대해 유효성 검사
      for (const inputId in state.inputs) {
        if (inputId === action.inputId) {
          // 현재 처리 중인 입력값이라면 새로운 유효성으로 갱신
          formIsValid = formIsValid && action.isValid
        } else {
          // 다른 입력값은 이전 유효성을 유지
          formIsValid = formIsValid && state.inputs[inputId].isValid
        }
      }
      // 새로운 상태 반환
      return {
        ...state,
        inputs: {
          ...state.inputs,
          [action.inputId]: { value: action.value, isValid: action.isValid },
        },
        isValid: formIsValid,
      }

    // 기본 케이스 (다른 액션이 들어온 경우)
    default:
      return state
  }
```

## Input 컴포넌트 수정

![Untitled](https://res.cloudinary.com/du1acdhxq/image/upload/v1788381319/notion-blog/_/React_NodeJS_Express_MongoDB_-_MERN_/React_js_2/Untitled_4.png)

```jsx
import React, { useEffect, useReducer } from 'react'

import { validate } from '../../util/validators'
import './Input.css'

// Input 컴포넌트는 다양한 입력 요소를 렌더링합니다.
const inputReducer = (state, action) => {
  // useReducer로 상태를 관리하는 함수입니다.
  switch (action.type) {
    case 'CHANGE':
      // CHANGE 액션일 경우, 새로운 상태를 반환합니다.
      return {
        ...state,
        value: action.val, // 입력 값 업데이트
        isValid: validate(action.val, action.validators),
      }
    case 'TOUCH': {
      return {
        ...state,
        isTouched: true,
      }
    }
    default:
      return state // 기본적으로 현재 상태를 그대로 반환합니다.
  }
}
const Input = (props) => {
  const touchHandler = () => {
    dispatch({
      type: 'TOUCH',
    })
  }
  // useReducer를 사용하여 inputState 상태와 dispatch 함수를 얻습니다.
  const [inputState, dispatch] = useReducer(inputReducer, {
    value: props.value || '',
    isValid: props || false,
    isTouched: props || false,
  })

  const { id, onInput } = props
  const { value, isValid } = inputState

  useEffect(() => {
    onInput(id, value, isValid)
  }, [id, value, isValid, onInput])

  const changeHandler = (event) => {
    // CHANGE 액션을 dispatch 하여 상태를 업데이트합니다.
    dispatch({ type: 'CHANGE', val: event.target.value, validators: props.validators })
  }

  // 입력 요소의 종류에 따라 input 또는 textarea를 선택하여 렌더링합니다.
  const element =
    props.element === 'input' ? (
      <input
        id={props.id}
        type={props.type}
        placeholder={props.placeholder}
        onChange={changeHandler}
        onBlur={touchHandler}
        value={inputState.value}
      />
    ) : (
      <textarea
        id={props.id}
        rows={props.rows || 3}
        onChange={changeHandler}
        onBlur={touchHandler}
        value={inputState.value}
      />
    )

  // 입력 요소를 감싸는 form-control 클래스를 가진 div 요소를 반환합니다.
  return (
    <div
      className={`form-control ${!inputState.isValid && inputState.isTouched && 'form-control--invalid'}`}
    >
      {/* 입력 요소와 연결된 레이블을 표시합니다. */}
      <label htmlFor={props.id}>{props.label}</label>
      {/* 선택한 입력 요소를 렌더링합니다. */}
      {element}
      {!inputState.isValid && inputState.isTouched && <p>{props.errorText}</p>}
    </div>
  )
}

export default Input
```

# 커스텀 훅이란?

```jsx
커스텀 훅은 리액트에서 사용되는 함수로, 기능을 하나로 묶어서 재사용 가능하게 만들어주는 것.

예를 들어, 여러 컴포넌트에서 사용하는 특정한 상태나 로직이 있다면,

이를 하나의 커스텀 훅으로 만들어서 여러 곳에서 손쉽게 사용할 수 있게 된다. 이렇게 하면 코드가 간결해지고, 유지보수가 편리해진다.

커스텀 훅을 만들 때, 보통 그 훅이 어떤 상태를 다루는지 또는 어떤 로직을 수행하는지 정의한다.

그리고 나중에 이 훅을 사용하는 컴포넌트에서는 필요한 부분만을 꺼내서 사용하게 된다.
```

## **커스텀 훅의 기본 구조**

```jsx
커스텀 훅은 보통 "use"로 시작하는 함수명을 가지며,

리액트 훅을 사용하여 상태를 다루거나 부가적인 로직을 수행한다.

예를 들어, 간단한 로컬 상태를 다루는 커스텀 훅을 만들어보겠다.
```

```jsx
import { useState } from 'react'

// 커스텀 훅 정의
const useCounter = (initialValue = 0) => {
  const [count, setCount] = useState(initialValue)

  const increment = () => {
    setCount(count + 1)
  }

  const decrement = () => {
    setCount(count - 1)
  }

  return { count, increment, decrement }
}

// 이제 이 커스텀 훅을 사용할 컴포넌트
const CounterComponent = () => {
  // useCounter 훅 사용
  const { count, increment, decrement } = useCounter(0)

  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={increment}>Increment</button>
      <button onClick={decrement}>Decrement</button>
    </div>
  )
}
```

```jsx
useState를 사용하여 커스텀 훅 내에서 로컬 상태인 count와 상태를 업데이트하는 setCount를 정의한다.

increment와 decrement 함수를 만들어 각각 카운트를 증가시키고 감소시킨다.

마지막으로, { count, increment, decrement }를 반환하여 이 값들을 사용하는 컴포넌트에서 이용할 수 있도록 한다.
```

## **사용 방법**

```jsx
useCounter 훅을 호출하여 초기값을 전달한다.

반환된 객체에서 원하는 상태와 함수들을 가져와 사용한다.

이제 CounterComponent는 useCounter 훅을 통해 간편하게 카운터 상태와 관련된 함수들을 사용할 수 있다.

커스텀 훅을 사용하면 여러 컴포넌트 간에 로직을 공유할 수 있고, 코드를 재사용하면서도 각 컴포넌트는 필요한 로직에만 집중할 수 있다.

이는 유지보수가 쉽고 코드의 가독성을 높여준다.
```

### **커스텀 훅으로 로컬 스토리지 상태 관리하기**

```jsx
import { useState } from 'react'

// 커스텀 훅 정의
const useLocalStorage = (key, initialValue) => {
  // 로컬 스토리지에서 특정 키의 값을 가져옵니다.
  const storedValue = localStorage.getItem(key)

  // 저장된 값이 있으면 파싱하여 initial 값으로 설정, 없으면 초기값 사용
  const initial = storedValue ? JSON.parse(storedValue) : initialValue

  // useState 훅을 사용하여 현재 상태와 상태 업데이트 함수를 생성합니다.
  const [value, setValue] = useState(initial)

  // 로컬 스토리지에 값을 설정하는 함수를 만듭니다.
  const setStoredValue = (newValue) => {
    // 현재 상태를 업데이트하고
    setValue(newValue)
    // 새로운 값을 JSON 문자열로 변환하여 로컬 스토리지에 저장합니다.
    localStorage.setItem(key, JSON.stringify(newValue))
  }

  // 상태 값과 로컬 스토리지에 값을 설정하는 함수를 반환합니다.
  return [value, setStoredValue]
}

// 이제 이 커스텀 훅을 사용할 컴포넌트
const LocalStorageComponent = () => {
  // useLocalStorage 훅 사용
  // 'counter'라는 키를 가진 로컬 스토리지 상태를 생성하고, 초기값을 0으로 설정합니다.
  const [count, setCount] = useLocalStorage('counter', 0)

  return (
    <div>
      {/* 현재 상태 값을 화면에 표시합니다. */}
      <p>Count: {count}</p>
      {/* 버튼 클릭 시 setCount 함수를 통해 상태를 증가시킵니다. */}
      <button onClick={() => setCount(count + 1)}>Increment</button>
      {/* 버튼 클릭 시 setCount 함수를 통해 상태를 감소시킵니다. */}
      <button onClick={() => setCount(count - 1)}>Decrement</button>
    </div>
  )
}
```

```jsx
초기 값은 로컬 스토리지에서 가져오거나 제공된 초기 값으로 설정된다.

반환된 setStoredValue 함수를 사용하여 상태를 업데이트하면 동시에 로컬 스토리지에도 값이 저장된다.
```

### 유효성 검사

```jsx
import { useState } from 'react'

// 커스텀 훅 정의
const useInputValidation = (initialValue, validate) => {
  // 입력 값과 유효성 상태를 다루는 상태와 함수를 생성합니다.
  const [value, setValue] = useState(initialValue)
  const [isValid, setIsValid] = useState(true)
  const [errorMessage, setErrorMessage] = useState('')

  // 입력 값이 변경될 때마다 유효성 검사를 수행합니다.
  const handleChange = (event) => {
    const inputValue = event.target.value
    setValue(inputValue)

    // 사용자 정의 유효성 검사 함수를 호출하여 유효성을 판단합니다.
    const validation = validate(inputValue)
    setIsValid(validation.isValid)
    setErrorMessage(validation.errorMessage)
  }

  // 입력 값과 유효성 상태, 유효성 오류 메시지를 반환합니다.
  return { value, isValid, errorMessage, handleChange }
}

// 이제 이 커스텀 훅을 사용할 컴포넌트
const InputValidationComponent = () => {
  // useInputValidation 훅 사용
  // 초기값은 빈 문자열, 유효성 검사 함수는 최소 길이 3을 요구하는 예제
  const { value, isValid, errorMessage, handleChange } = useInputValidation('', (input) => {
    // 여기서 주의할건 (input) => 여기서 부터 모두 하나의 매개변수 로 들어가는 하나의 함수다 ( 이함수가 매개변수다 )  그래서 3번째 단락을 보면 inputValue 가 input 매개변수에 들어가게 된다
    // 유효성 검사 로직: 입력 값이 최소 길이 3 이상이어야 함
    const isValid = input.length >= 3
    return {
      isValid,
      errorMessage: isValid ? '' : '최소 3자 이상 입력하세요.',
    }
  })

  return (
    <div>
      {/* 입력 폼 */}
      <input
        type="text"
        value={value}
        onChange={handleChange}
        placeholder="최소 3자 이상 입력하세요."
      />
```

## Input.jsx

```jsx
// React 및 관련 라이브러리 가져오기
import React, { useReducer, useEffect } from 'react'

// 유틸리티 함수 가져오기
import { validate } from '../../util/validators'

// 스타일 파일 및 컴포넌트 스타일 가져오기
import './Input.css'

// 입력 요소 상태 관리를 위한 리듀서 정의
const inputReducer = (state, action) => {
  switch (action.type) {
    // 입력 값이 변경될 때
    case 'CHANGE':
      return {
        ...state,
        value: action.val,
        isValid: validate(action.val, action.validators),
      }
    // 입력이 터치되었을 때
    case 'TOUCH': {
      return {
        ...state,
        isTouched: true,
      }
    }
    // 기본 케이스
    default:
      return state
  }
}

// 입력 컴포넌트 정의
const Input = (props) => {
  // useReducer를 사용하여 입력 상태 관리
  const [inputState, dispatch] = useReducer(inputReducer, {
    value: props.initialValue || '',
    isTouched: false,
    isValid: props.initialValid || false,
  })

  // props로부터 필요한 값 추출
  const { id, onInput } = props
  const { value, isValid } = inputState

  // useEffect를 사용하여 입력 값이 변경될 때마다 부모 컴포넌트에 알림
  useEffect(() => {
    onInput(id, value, isValid)
  }, [id, value, isValid, onInput])

  // 입력 값 변경 핸들러
  const changeHandler = (event) => {
    dispatch({
      type: 'CHANGE',
      val: event.target.value,
      validators: props.validators,
    })
  }

  // 입력이 터치될 때의 핸들러
  const touchHandler = () => {
    dispatch({
      type: 'TOUCH',
    })
  }

  // 입력 요소 선택 (input 또는 textarea)
  const element =
    props.element === 'input' ? (
      <input
        id={props.id}
        type={props.type}
        placeholder={props.placeholder}
        onChange={changeHandler}
        onBlur={touchHandler}
        value={inputState.value}
      />
    ) : (
      <textarea
        id={props.id}
        rows={props.rows || 3}
        onChange={changeHandler}
        onBlur={touchHandler}
        value={inputState.value}
      />
    )

  // 입력 요소와 관련된 스타일 및 에러 메시지 표시
  return (
    <div
      className={`form-control ${
        !inputState.isValid && inputState.isTouched && 'form-control--invalid'
      }`}
    >
      <label htmlFor={props.id}>{props.label}</label>
      {element}
      {!inputState.isValid && inputState.isTouched && <p>{props.errorText}</p>}
    </div>
  )
}

// 외부에서 사용할 수 있도록 내보내기
export default Input
```

## form-hook.js

```jsx
// React 훅 및 콜백 훅 가져오기
import { useCallback, useReducer } from 'react'

// 폼 상태를 관리하는 리듀서 정의
const formReducer = (state, action) => {
  switch (action.type) {
    // 입력 값 변경 액션
    case 'INPUT_CHANGE':
      // 폼의 전체 유효성을 확인하는 변수 초기화
      let formIsValid = true

      // 모든 입력 요소에 대해 유효성 검사 및 폼 유효성 갱신
      for (const inputId in state.inputs) {
        if (inputId === action.inputId) {
          formIsValid = formIsValid && action.isValid
        } else {
          formIsValid = formIsValid && state.inputs[inputId].isValid
        }
      }

      // 상태 업데이트
      return {
        ...state,
        inputs: {
          ...state.inputs,
          [action.inputId]: { value: action.value, isValid: action.isValid },
        },
        isValid: formIsValid,
      }

    // 기본 케이스
    default:
      return state
  }
}

// useForm 커스텀 훅 정의
export const useForm = (initialInputs, initialFormValidity) => {
  // useReducer를 사용하여 폼 상태 관리
  const [formState, dispatch] = useReducer(formReducer, {
    inputs: initialInputs,
    isValid: initialFormValidity,
  })

  // 입력 핸들러 정의 및 콜백으로 래핑
  const inputHandler = useCallback((id, value, isValid) => {
    // 입력 값이 변경될 때마다 디스패치하여 리듀서에게 상태 업데이트 요청
    dispatch({
      type: 'INPUT_CHANGE',
      value: value,
      isValid: isValid,
      inputId: id,
    })
  }, [])

  // 폼 상태 및 입력 핸들러 반환
  return [formState, inputHandler]
}
```

## **선택 사항 (커스텀) React 훅에 대한 추가 정보**

```jsx
React 훅은 함수형 컴포넌트에서 매우 강력한 기능입니다. "React - 완벽 가이드" 강의 또는 공식 문서에서 이에 대해 자세히 알아볼 수 있습니다: https://reactjs.org/docs/hooks-intro.html

특히 좋은 기능 중 하나는 자신만의 React 훅을 구축할 수 있다는 것입니다 (https://reactjs.org/docs/hooks-custom.html).

이를 통해 스테이트풀 컴포넌트 로직을 한곳에 담아 공유할 수 있습니다.

다른 내장된 훅(예: useState())을 사용하는 훅을 구축할 수 있으며 훅을 사용하는 모든 컴포넌트는 커스텀 훅에서도 사용할 수 있는 내장된 훅을 사용합니다.

이를 통해 이전 강의에서 구축하기 시작한 useForm()훅과 같은 훅을 구축할 수 있습니다. 결과적으로 스테이트풀 양식 로직(이 경우 useReducer() 를 사용함)을 컴포넌트 간에 공유할 수 있다는 것입니다. 이렇게 하면 코드 중복을 방지하고 코드를 쉽게 변경할 수 있으며 더 읽기 쉬운 코드를 작성할 수 있습니다.

'커스텀 훅'이라는 전문 용어를 사용하면 결국 커스텀 훅이 반적인 JavaScript 함수라는 사실을 간과하기 쉽습니다. 절대 잊지 마세요!

컴포넌트 함수에서 useForm() 을 사용하면 컴포넌트의 모든 재평가(즉, 모든 리렌더링 주기)에 대해 호출됩니다. 따라서 커스텀 훅의 모든 로직은 컴포넌트 함수가 실행될 때마다 실행됩니다.

물론 useState() 또는 useReducer() 와 같은 내장된 훅에는 리렌더링 주기 동안 상태 변경이 유지되도록 하는 메커니즘이 있습니다.
```

## **사용자 지정 훅 생성**

### **훅의 역할**

```jsx
사용자 지정 훅인 useForm은 React의 useReducer와 useCallback을 활용하여 양식(form) 상태를 관리하는 함수이다.

폼 의 입력값과 유효성 여부를 추적하며, 입력값이 변경될 때마다 상태를 업데이트한다.
```

### **form-hook.js 파일 구성**

```jsx
form-hook.js 파일에 useReducer를 이용한 상태 관리와 관련된 로직이 들어있다.

formReducer 함수는 폼 입력값의 변경에 따라 상태를 업데이트한다.

useForm 함수는 초기 입력값과 유효성 여부를 받아 useReducer를 호출하고, 상태 및 입력값 변경 핸들러를 반환한다.
```

```jsx
// form-hook.js

// react에서 제공하는 특수 함수들을 import합니다
import { useCallback, useReducer } from 'react'

// 폼의 상태를 업데이트하는 리듀서 함수를 정의합니다
const formReducer = (state, action) => {
  switch (action.type) {
    case 'INPUT_CHANGE':
      // 양식이 유효한지를 판단하는 변수를 초기화합니다
      let formIsValid = true

      // 현재 입력값을 포함한 모든 입력 요소에 대해 반복합니다
      for (const inputId in state.inputs) {
        // 현재 처리 중인 입력 요소와 동일한 경우
        if (inputId === action.inputId) {
          // 현재 입력 값의 유효성을 반영하여 formIsValid 값을 업데이트합니다
          formIsValid = formIsValid && action.isValid
        } else {
          // 다른 입력 요소들은 이전의 유효성을 유지합니다
          formIsValid = formIsValid && state.inputs[inputId].isValid
        }
      }

      // 현재 입력 요소의 값을 업데이트하고, 전체 폼의 유효성을 반영합니다
      return {
        ...state,
        inputs: {
          ...state.inputs,
          [action.inputId]: { value: action.value, isValid: action.isValid },
        },
        isValid: formIsValid,
      }

    default:
      return state
  }
}

// useForm이라는 사용자 지정 훅을 생성합니다
export const useForm = (initialInputs, initialFormValidity) => {
  // useReducer 훅을 사용하여 폼의 상태를 관리합니다
  const [formState, dispatch] = useReducer(formReducer, {
    inputs: initialInputs, // 초기 입력값
    isValid: initialFormValidity, // 초기 폼의 유효성
  })

  // 입력값 변경 핸들러 함수를 생성하고 useCallback으로 감싸서 메모이제이션합니다
  const inputHandler = useCallback((id, value, isValid) => {
    // 리듀서에 전달할 액션을 디스패치하여 상태를 업데이트합니다
    dispatch({
      type: 'INPUT_CHANGE',
      value: value,
      isValid: isValid,
      inputId: id,
    })
  }, []) // 의존성 배열이 비어있으므로 컴포넌트 렌더링 중에만 생성됩니다

  // 사용자 지정 훅이 폼의 상태와 입력값 변경 핸들러를 반환합니다
  return [formState, inputHandler]
}
```

```jsx
      <div className="center">
        <h2>Could not find place!</h2>
      </div>
    )
  }

  return (
    <form className="place-form" onSubmit={placeUpdateSubmitHandler}>
      {/* Input 컴포넌트를 사용하여 제목 입력란을 렌더링 */}
      <Input
        id="title"
        element="input"
        type="text"
        label="Title"
        validators={[VALIDATOR_REQUIRE()]}
        errorText="Please enter a valid title."
        onInput={inputHandler}
        initialValue={formState.inputs.title.value} // 초기값 설정
        initialValid={formState.inputs.title.isValid} // 초기 유효성 설정
      />
      {/* Input 컴포넌트를 사용하여 설명 입력란을 렌더링 */}
      <Input
        id="description"
        element="textarea"
        label="Description"
        validators={[VALIDATOR_MINLENGTH(5)]}
        errorText="Please enter a valid description (min. 5 characters)."
        onInput={inputHandler}
        initialValue={formState.inputs.description.value} // 초기값 설정
        initialValid={formState.inputs.description.isValid} // 초기 유효성 설정
      />
      {/* 업데이트 버튼. 폼이 유효하지 않으면 비활성화됨 */}
      <Button type="submit" disabled={!formState.isValid}>
        UPDATE PLACE
      </Button>
    </form>
  )
}

export default UpdatePlace
```
