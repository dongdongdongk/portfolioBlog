```jsx
import React, { useEffect } from 'react'
import Login from '../components/Login/Login'
import { useSelector } from 'react-redux'
import { useNavigate } from 'react-router-dom'

const LoginPage = () => {
  const navigate = useNavigate()
  const { isAuthenticated } = useSelector((state) => state.user)

  useEffect(() => {
    if (isAuthenticated === true) {
      navigate('/')
    }
  })

  return (
    <div>
      <Login />
    </div>
  )
}

export default LoginPage
```

```jsx
이 부분이 주변을 어둡게 만들어 준다

bg-[#0000004b]으로, 이는 검은색(#000000)의 4B(75%) 투명도의 배경색을 나타낸다.

이 배경은 화면 전체를 덮고, 투명도를 통해 배경이 약간 어둡게 표현된다.

이것은 모달 형태로 전체 화면을 덮어서 사용자의 주의를 모달 창에 집중하도록 하는 효과를 낼 때 자주 사용된다.

두개의 return 을 가지고 있는데

첫 번째 return 문은 화면 전체를 어둡게 만드는 배경 모달을 정의하고 있다.

두 번째 return 문은 실제 장바구니 컨텐츠를 담은 모달 창을 정의하고 있다.

( 아래 return 문을 위의 return 문에 CartSingle 컴포넌트로 넣어준다 )

( 아래 CartSingle 은 하나의 아이템 하나하나고 위의 return 은 모달창 전체이다 )
```

```jsx
import React, { useState } from 'react'
import { Link } from 'react-router-dom'
import styles from '../../styles/styles'
import { categoriesData, productData } from '../../static/data'
import { AiOutlineHeart, AiOutlineSearch, AiOutlineShoppingCart } from 'react-icons/ai'
import { CgProfile } from 'react-icons/cg'
import { IoIosArrowDown, IoIosArrowForward } from 'react-icons/io'
import { BiMenuAltLeft } from 'react-icons/bi'
import DropDown from './DropDown'
import Navbar from './Navbar'
import Cart from '../Cart/Cart'
import { useSelector } from 'react-redux'

// Header 컴포넌트 정의
const Header = ({ activeHeading }) => {
  // Redux의 useSelector를 통해 사용자 정보 및 인증 상태를 가져옴
  const { isAuthenticated, user } = useSelector((state) => state.user)

  // 검색어와 검색결과 상태 관리
  const [searchTerm, setSearchTerm] = useState('')
  const [searchData, setSearchData] = useState(null)

  // 상단 헤더의 스타일을 관리하는 상태
  const [active, setActive] = useState(false)

  // 카테고리 드롭다운 관리 상태
  const [dropDown, setDropDown] = useState(false)

  // 장바구니 및 위시리스트 모달 열기/닫기 관리 상태
  const [openCart, setOpenCart] = useState(false)
  const [openWishlist, setOpenWishlist] = useState(false)

  // 검색어 입력 시 핸들러
  const handleSearchChange = (e) => {
    const term = e.target.value
    setSearchTerm(term)

    // 제품 데이터를 검색어에 따라 필터링하여 검색결과 상태 업데이트
    const filteredProducts = productData.filter((product) =>
      product.name.toLowerCase().includes(term.toLowerCase())
    )
    setSearchData(filteredProducts)
  }

  // 스크롤 이벤트 핸들러, 스크롤 위치에 따라 상단 헤더의 스타일 변경
  window.addEventListener('scroll', () => {
    if (window.scrollY > 70) {
      setActive(true)
    } else {
      setActive(false)
    }
  })

  // JSX 반환
  return (
    <>
      {/* 모바일 및 작은 화면에서 표시되는 상단 헤더 영역 */}
      <div className={`${styles.section}`}>
        <div className="800px:h-[50px] 800px:my-[20px] 800px:flex hidden items-center justify-between">
          {/* 로고 */}
          <div>
            <Link to="/">
              <img src={`http://localhost:5000/images/QuickDealMarketFix.jpg`} alt="" />
            </Link>
          </div>
          {/* 검색창 */}
          <div className="relative w-[50%]">
            <input
              type="text"
              placeholder="Search Product..."
              value={searchTerm}
              onChange={handleSearchChange}
              className="h-[40px] w-full rounded-md border-[2px] border-[#e3bb8b] px-2"
            />
            <AiOutlineSearch size={30} className="absolute top-1.5 right-2 cursor-pointer" />
            {/* 검색 결과가 있을 경우 결과를 표시하는 드롭다운 */}
            {searchData && searchData.length !== 0 ? (
              <div className="shadow-sm-2 absolute z-[9] min-h-[30vh] bg-slate-50 p-4">
                {searchData &&
                  searchData.map((i, index) => {
                    const d = i.name
                    const Product_name = d.replace(/\s+/g, '-')
                    return (
                      <Link to={`/product/${Product_name}`} key={index}>
                        <div className="items-start-py-3 flex w-full">
                          <img
                            src={i.image_Url[0].url}
                            alt=""
                            className="mr-[10px] h-[40px] w-[40px]"
                          />
                          <h1>{i.name}</h1>
                        </div>
                      </Link>
                    )
                  })}
              </div>
            ) : null}
          </div>
          {/* 판매자 등록 링크 */}
          <div className={`${styles.button}`}>
            <Link to="/seller">
              <h1 className="flex items-center text-[#fff]">
                판매자 등록 <IoIosArrowForward className="ml-1" />
              </h1>
            </Link>
          </div>
        </div>
      </div>

      {/* 데스크탑에서 표시되는 상단 헤더 영역 */}
      <div
        className={`${
          active === true ? 'fixed top-0 left-0 z-10 shadow-sm' : null
        } 800px:flex hidden h-[70px] w-full items-center justify-between bg-[#e3bb8b] transition`}
      >
        <div className={`${styles.section} relative ${styles.noramlFlex} justify-between`}>
          {/* 카테고리 드롭다운 */}
          <div onClick={() => setDropDown(!dropDown)}>
            <div className="1000px:block relative mt-[10px] hidden h-[60px] w-[270px]">
              <BiMenuAltLeft size={30} className="absolute top-3 left-2" />
              <button
                className={`flex h-[100%] w-full items-center justify-between rounded-t-md bg-white pl-10 font-sans text-lg font-[500] select-none`}
              >
                All Categories
              </button>
              <IoIosArrowDown
                size={20}
                className="absolute top-4 right-2 cursor-pointer"
                onClick={() => setDropDown(!dropDown)}
              />
              {/* 카테고리 드롭다운 컴포넌트 */}
              {dropDown ? (
                <DropDown categoriesData={categoriesData} setDropDown={setDropDown} />
              ) : null}
            </div>
          </div>
          {/* 네비게이션 메뉴 */}
          <div className={`${styles.noramlFlex}`}>
            <Navbar active={activeHeading} />
          </div>
          {/* 하트(위시리스트), 장바구니, 프로필 아이콘 */}
          <div className="flex">
            {/* 위시리스트 아이콘 */}
            <div className={`${styles.noramlFlex}`}>
              <div className="relative mr-[15px] cursor-pointer">
                <AiOutlineHeart size={30} color="rgb(255 255 255 / 83%)" />
                <span className="top right absolute top-0 right-0 m-0 h-4 w-4 rounded-full bg-[#3bc177] p-0 text-center font-mono text-[12px] leading-tight text-white">
                  0
                </span>
              </div>
            </div>
            {/* 장바구니 아이콘 */}
            <div className={`${styles.noramlFlex}`}>
              <div className="relative mr-[15px] cursor-pointer" onClick={() => setOpenCart(true)}>
                <AiOutlineShoppingCart size={30} color="rgb(255 255 255 / 83%)" />
                <span className="top right absolute top-0 right-0 m-0 h-4 w-4 rounded-full bg-[#3bc177] p-0 text-center font-mono text-[12px] leading-tight text-white">
                  1
                </span>
              </div>
            </div>
            {/* 프로필(인증된 경우) 또는 로그인 링크(인증되지 않은 경우) */}
            <div className={`${styles.noramlFlex}`}>
              <div className="relative mr-[15px] cursor-pointer">
                {isAuthenticated ? (
                  <Link to="/profile">
                    <img
                      src={`http://localhost:5000/${user.avatar}`}
                      className="h-[35px] w-[35px] rounded-full"
                      alt=""
                    />
                  </Link>
                ) : (
                  <Link to="/login">
                    <CgProfile size={30} color="rgb(255 255 255 / 83%)" />
                  </Link>
                )}
              </div>
            </div>
            {/* 장바구니 목록 팝업 */}
            {openCart ? <Cart setOpenCart={setOpenCart} /> : null}
          </div>
        </div>
      </div>
    </>
  )
}

// Header 컴포넌트 내보내기
export default Header
```

```jsx
import React, { useEffect, useState } from 'react'
import { productData } from '../../static/data'

import ProductCard from '../Route/ProductCard'

const SuggestedProduct = ({ data }) => {
  // 관련된 제품 데이터를 상태로 관리
  const [products, setProducts] = useState(null)

  useEffect(() => {
    // 부모 컴포넌트에서 전달된 data의 카테고리에 해당하는 제품 필터링
    const filteredProducts = productData && productData.filter((i) => i.category === data.category)

    // 필터링된 제품을 상태에 설정
    setProducts(filteredProducts)
  }, [data.category]) // data.category가 변경될 때마다 useEffect 다시 실행

  return (
    <div>
      {data ? (
        <div className={`p-4 ${styles.section}`}>
          {/* 관련 제품 섹션 제목 */}
          <h2 className={`${styles.heading} mb-5 border-b text-[25px] font-[500]`}>
            Related Product

            {
              // products 상태가 있고 각 제품에 대해 ProductCard 컴포넌트를 렌더링
              products &&
                products.map((product, index) => <ProductCard data={product} key={index} />)
```

```jsx
columns 배열에는 주문 정보를 표시할 테이블의 각 열에 대한 정의가 있다.

field: 데이터와 연결된 필드명을 지정한다.

headerName: 테이블 헤더에 표시될 열의 이름을 지정한다.
```

## ProtectedRoute.js

```jsx
import { Navigate } from 'react-router-dom'

// 인증된 사용자만 접근을 허용하는 ProtectedRoute 컴포넌트
const ProtectedRoute = ({ isAuthenticated, children }) => {
  // 만약 사용자가 인증되어 있지 않다면 로그인 페이지로 이동
  if (!isAuthenticated) {
    return <Navigate to="/login" replace />
  }
  // 사용자가 인증되어 있으면 해당 컴포넌트(children)를 렌더링
  return children
}

// ProtectedRoute 컴포넌트를 외부에서 사용할 수 있도록 내보내기
export default ProtectedRoute
```

```jsx
children: ProtectedRoute 컴포넌트 안에 중첩된 컴포넌트들이다.
```

```jsx
import React from 'react'
import styles from '../../styles/styles'
```

```jsx
  console.log(active)

  return (
    <div className="flex w-full justify-center">
      {/* 단계 표시 부분 */}
      <div className="800px:w-[50%] flex w-[90%] flex-wrap items-center">
```

```jsx
CheckoutSteps 컴포넌트는 주문 진행 단계를 표시하는 UI를 생성한다.

active prop은 현재 활성화된 단계를 나타낸다.

각 단계는 버튼과 표시 바로 나누어져 있다.

각 버튼은 해당 단계를 나타내며, active prop에 따라 색상이 변경된다.

각 표시 바는 현재 진행 중인 단계까지의 진행을 시각적으로 나타낸다.

( 결제하기로 다음 단계로 넘어가면 그 페이지에서는 active 가 2 또는 3으로 변해서 진행 단계가 변하게 된다 )
```
