from itemadapter import ItemAdapter

import logging  # 로그를 작성하기 위해 logging 모듈을 가져옵니다.

class MongodbPipeline:
    collection_name = 'transcripts'  # MongoDB에서 사용할 컬렉션 이름을 설정합니다.

    def open_spider(self, spider):
        """
        스파이더가 열릴 때 실행되는 메서드.
        MongoDB 클라이언트와 데이터베이스에 연결합니다.
        """
        # MongoDB 클라이언트를 생성하고 클러스터에 연결합니다.
        self.client = pymongo.MongoClient("mongodb+srv://dhk9309:kim14582367@cluster0.6ebciyv.mongodb.net/?retryWrites=true&w=majority&appName=Cluster0")

        # 데이터베이스를 선택합니다. 데이터베이스가 존재하지 않으면 새로 생성됩니다.
        self.db = self.client['My_Database']

    def close_spider(self, spider):
        """
        스파이더가 닫힐 때 실행되는 메서드.
        여기서는 스파이더가 종료되었음을 로그에 기록합니다.
        """
        logging.warning('Spider Closed - Pipeline')  # 스파이더가 닫혔음을 경고 레벨로 로그에 기록합니다.
        self.client.close()

    def process_item(self, item, spider):
        """
        스크랩된 각 아이템을 처리하는 메서드.
        아이템을 MongoDB 컬렉션에 삽입합니다.
        """
        # 아이템을 딕셔너리로 변환하고, 설정된 컬렉션에 삽입합니다.
        self.db[self.collection_name].insert_one(dict(item))

        # 처리된 아이템을 반환합니다.
        return item

# SQLite3 연결해서 사용하기 ( SQLite3 )

```python
from itemadapter import ItemAdapter

import logging
import pymongo
import sqlite3

class SQLLitePipeline:
    collection_name = "transcripts"

    def open_spider(self, spider):
        # SQLite 데이터베이스 연결 및 테이블 생성
        self.connection = sqlite3.connect("transcripts.db")  # SQLite 데이터베이스 연결
        self.c = self.connection.cursor()  # 커서 생성

        # transcripts 테이블 생성 쿼리 실행
        try:
            self.c.execute(
                """
                CREATE TABLE transcripts (
                    title TEXT,
                    plot TEXT,
                    transcript TEXT,
                    url TEXT
                )
                """
            )
            self.connection.commit()  # 변경사항 커밋
        except sqlite3.OperationalError:
            pass  # 테이블이 이미 존재하는 경우 예외 처리

    def close_spider(self, spider):
        # 스파이더 종료 시 SQLite 연결 닫기
        self.connection.close()

    def process_item(self, item, spider):
        # SQLite에 아이템 삽입
        self.c.execute(
            '''
            INSERT INTO transcripts (title, plot, transcript, url)
            VALUES (?, ?, ?, ?)
            ''', (
                item.get('title'),
                item.get('plot'),
                item.get('transcript'),
                item.get('url')
            )
        )
        self.connection.commit()  # 변경사항 커밋
        return item
```
