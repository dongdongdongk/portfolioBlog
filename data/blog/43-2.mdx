class Car {
//필드
	Tire frontLeftTire = new Tire();   // Tire 클래스를 이용하여 Tire 타입으로 4개의 필드 생성
	Tire frontRightTire = new Tire();
	Tire backLeftTire = new Tire();
	Tire backRightTire = new Tire();

//메소드
void rum() {
	frontLeftTire.roll();               // Tire 클래스에 있는 roll 메소드 호출
	frontRightTire.roll();
	backLeftTire.roll();
	backRightTire.roll();
}

실행 class

Car myCar = new Car(); //객체생성
myCar.frontRightTire = new HankookTire(); // 필드에 대입되는 객체를 다른 객체 사용
myCar.backLeftTire = new KumhoTire();     // 결과적으로 roll 메소드 는 Tire 가 아닌 재정의된
myCar.run();                              // HankookTire 가 실행됨

* 이때 HankookTire 같은 클래스가 Tire 에게 상속 받아야함

```java
class Driver {
	void drive(Vehicle vehicle) {  // 클래스 타입으로 매개변수 선언 (참조형)
		vehicle.run(); //vehicle 의 run 메소드 호출
	}
}

//사용할때

Driver driver = new Driver(); // driver 객체생성
Vehicle vehicle = new Vehicle(); // vehicle 객체생성
driver.drive(vehicle); // driver 객체의 drive 메소드를 호출할때 매개변수 값으로 vehicle
                       // vehicle 의 주소값을 메서드 drive 에 복사 대입

```

```java
//Vehicle 클래스의 자식 클래스로 Bus 클래스가 있다면

Driver driver = new Driver();

Bus bus = new Bus(); // Vehicl vehicle = new Vehicle 객채생성 대신에

driver.drive(bus); //bus 객체의 주소값을 복사 대입해준다.

//자동 타입 변환 발생

Vehicle vehicle = bus;

//drive 라는 메소드가 부모클래스 Vehicle 을 가지고 있기 때문에 bus 가 매개값으로 대입이 되면
//위와 같은 효과가 발생

//결과적으론 아래와 같은
class Driver {
	void drive(Vehicle vehicle) { //bus 의 주소를 가지고 있는 vehicle
		bus.run(); //vehicle.run 인데 bus의 주소를 가지고 있는 vehicle = bus.run();
	}
}

// 물론 이때 bus 에서 run 이라는 메소드를 재정의 하고 있어야 한다.

//(Vehicle vehicle) 같은 매개변수는 어렵게 생각하지 말고 (int x) 처럼 타입-변수 개념으로
// 쉽게 생각하자
```

- 매개 변수의 다형성은 매개값으로 어떤 자식 객체가 제공되느냐에 따라 메소드의 실행 결과가 다양해질 수 있다는 것
- 자식 객체가 부모의 메소드를 재정의 했다면 메소드 내부에서 재정의된 메소드를 호출함으로써 메소드의 실행결과는 다양해짐
- driver.drive(bus); 의 bus 대신에 다양한 Vehicle 을 상속하고 있는 자식 클래스를 활용하고 ex)taxi 재정의 하여 다형성을 부여할 수 있다.
