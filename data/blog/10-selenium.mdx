# **크롬 창 자동으로 닫히지 않고 유지시켜주는 옵션 추가**

```python
from selenium import webdriver

```


## 화면 크기 지정하는 옵션 추가, 유저 에이전트(user agent) 사용법

```python
from selenium import webdriver
from bs4 import BeautifulSoup
from selenium.webdriver.chrome.options import Options
import time

user_agent = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36"

options = Options()

options.add_experimental_option("detach", True) # 바로 종료되지 않게 옵션 추가

options.add_argument("--start-maximized") # 화면을 최대 크기로 시작
# options.add_argument("--start-fullscreen") # 화면을 전체화면으로 시작
# options.add_argument("window-size=500,500") # 화면을 500 500 으로

options.add_argument(f"user-agent={user_agent}") # user_agent 추가

driver = webdriver.Chrome(options=options)

url = "https://naver.com"

driver.get(url)

```

```python

# options.add_argument("window-size=500,500") # 화면을 500 500 으로

이 부분이 화면의 크기를 조절하는 부분이다
```

# 화면없이 사용하는 헤드리스 모드 옵션 추가

```python
# 필요한 라이브러리를 가져옵니다
from selenium import webdriver  # 웹 브라우저를 자동화하기 위한 라이브러리
from bs4 import BeautifulSoup  # HTML을 파싱하기 위한 라이브러리
from selenium.webdriver.chrome.options import Options  # 크롬 웹드라이버 옵션을 설정하기 위한 라이브러리
import time  # 시간 관련 기능을 사용하기 위한 라이브러리

# 사용자 에이전트(User-Agent)를 설정합니다
user_agent = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36"

# 크롬 웹드라이버의 옵션을 설정합니다
options = Options()

# 브라우저를 닫더라도 웹드라이버가 종료되지 않도록 설정합니다
options.add_experimental_option("detach", True)

# 브라우저를 실행할 때 최대화된 상태로 시작하도록 옵션을 추가합니다
options.add_argument("--start-maximized")

# 사용자 에이전트를 설정합니다
options.add_argument(f"user-agent={user_agent}")

# 헤드리스 모드로 실행하도록 설정합니다 (화면 출력 없이 실행)
options.add_argument("--headless")

# GPU를 사용하지 않도록 설정합니다 (Linux 환경에서 필요한 옵션)
options.add_argument("--disable-gpu")

# 크롬 웹드라이버를 초기화합니다
driver = webdriver.Chrome(options=options)

# 접속할 URL을 설정합니다
url = "https://naver.com"

# 지정한 URL로 브라우저를 이동시킵니다
driver.get(url)

# 페이지 소스를 출력합니다 (최대 1000글자까지만 출력)
print(driver.page_source[:1000])

# 웹드라이버를 종료합니다
driver.quit()

```

```python
브라우저 없이 셀레니움을 사용하는 방법이다

이 방법은 모든 작업이 완료된 후에 사용하는것이 일반적이다

# 헤드리스 모드로 실행하도록 설정합니다 (화면 출력 없이 실행)
options.add_argument("--headless")

# GPU를 사용하지 않도록 설정합니다 (Linux 환경에서 필요한 옵션)
options.add_argument("--disable-gpu")

사용법은 위의 두 코드를 추가 해주는것인데 두번째 GPU 관련 옵션은 과거에는 사용해야 했지만 지금은

괜찮지만 써줘도 상관없다.
```

# 브라우저에 자동화된 테스트 소프트웨어에 의해 제어되고 있습니다 없에기 ( subprocess 사용 )

```python
이 방법은 일반적인 방법하고 다르게 subprocess 모듈을 사용하여 새로운 Chrome 을 실행하고

Chrome의 원격 디버깅 기능을 활성화하고 원격 모드로 셀레니움을 사용하는 방법이다
```

```python
from selenium import webdriver
from selenium.webdriver.chrome.options import Options
import subprocess

user_agent = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36"

subprocess.Popen(r'C:\Program Files\Google\Chrome\Application\chrome.exe --remote-debugging-port=9222 --user-data-dir="C:\chromeCookie"')

options = Options()

options.add_experimental_option("debuggerAddress", "127.0.0.1:9222")

options.add_argument("--start-maximized") # 화면을 최대 크기로 시작

options.add_argument(f"user-agent={user_agent}") # user_agent 추가

driver = webdriver.Chrome(options=options)

url = "https://naver.com"

driver.get(url)
```

```python
subprocess.Popen(r'C:\\Program Files\\Google\\Chrome\\Application\\chrome.exe --remote-debugging-port=9222 --user-data-dir="C:\\chromeCookie"')

여기에는 실행할 프로그램의 경로와 이름이 지정되어 있다. 이 경우 Chrome 브라우저를 실행한다.

--remote-debugging-port=9222

이 옵션은 Chrome의 원격 디버깅 기능을 활성화하고, 9222 포트를 사용하도록 지시한다.

이를 통해 Selenium은 Chrome 인스턴스를 제어할 수 있다.

--user-data-dir="C:\\chromeCookie"

이 옵션은 Chrome이 사용자 데이터(쿠키, 캐시 등)를 저장할 디렉토리를 지정한다.

이렇게 하면 이전 세션의 데이터를 유지할 수 있다.

따라서 이 줄의 코드는 Chrome 브라우저의 새로운 인스턴스를 실행하고,

원격 디버깅 포트를 열며, 사용자 데이터를 저장할 디렉토리를 지정하는 역할을 한다.

options.add_experimental_option("debuggerAddress", "127.0.0.1:9222")

Selenium이 실행된 Chrome 인스턴스의 원격 디버깅 주소를 지정한다.

이렇게 하면 Selenium이 해당 Chrome 인스턴스를 제어할 수 있게 된다.

이 방식을 사용하면 Chrome 프로파일을 유지할 수 있고,

원격 디버깅 기능을 활용할 수 있어 더 많은 제어 기능을 제공한다.

하지만 코드가 조금 더 복잡해지는 단점이 있다.

+ 헤드리스 모드를 사용하지 못한다

* r 문자열 접두사는 백슬래시(\)를 이스케이프 문자가 아닌 일반 문자로 인식하게 한다
```

## subprocess 모듈

```python
subprocess 모듈은 Python에서 새로운 프로세스를 생성하고 실행할 수 있게 해준다.

운영 체제의 명령어를 실행하거나 외부 프로그램을 호출할 때 사용된다. ( 현재 실행되는 프로그램과는 별개로 실행되는 다른 프로그램 )

이 모듈은 다음과 같은 주요 기능을 제공한다.
```

```python
**subprocess.run()**

명령어를 실행하고 완료될 때까지 기다립니다.

명령어의 출력, 반환 코드, 실행 시간 등의 정보를 CompletedProcess 객체로 반환합니다.

**subprocess.Popen()**

새로운 프로세스를 생성하고 제어할 수 있게 해줍니다.

프로세스의 입출력 스트림을 제어할 수 있습니다.

프로세스가 완료되기 전에 다른 작업을 수행할 수 있습니다.

**subprocess.call()**

명령어를 실행하고 완료될 때까지 기다립니다.

반환 코드만 반환합니다.

**subprocess.check_call()**

명령어를 실행하고 완료될 때까지 기다립니다.

반환 코드가 0이 아니면 예외를 발생시킵니다.
```
