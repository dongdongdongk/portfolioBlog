## MailManager 생성

![Untitled](https://res.cloudinary.com/du1acdhxq/image/upload/v1788381059/notion-blog/_/_/Spring_/_/Untitled.png)

```java
@Component
@Slf4j
public class MailManager {

	@Value("${spring.mail.username}")
	private String sender;

	@Autowired
	private JavaMailSender javaMailSender;

	public void send(String to,String sub,String con) throws MessagingException {
		//html 추가해서 보낼수 있음 (정교함)
		MimeMessage mimeMessage = javaMailSender.createMimeMessage();
		mimeMessage.setFrom(sender);
		mimeMessage.addRecipient(RecipientType.TO, new InternetAddress(to));
		mimeMessage.setSubject(sub);
		mimeMessage.setText(con);
		javaMailSender.send(mimeMessage);


//		HTML태그를 무시하고 그대로 Text로 전송됨
```

## email 유효성 검사를 위한 js 작성

```jsx
$(document).ready(function () {
  function emailCheck(email) {
    // 이메일 유효성 검사를 위한 정규식
    const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
    return emailPattern.test(email)
  }

  // email 찾기
  $('.find_btn').click(function () {
    const email = $('.email').val()
    if (!emailCheck(email)) {
      alert('올바른 이메일 주소를 입력해주세요.')
      return
    }
    $.ajax({
      url: 'findId',
      type: 'POST',
      data: {
        email: email,
      },
    })
      .done(function () {
        alert('메일발송 완료')
      })
      .fail(function () {
        alert('에러가 발생했습니다')
      })
  })

  // id 찾기
  $('.find_btnPw').click(function () {
    const id = $('.id').val()

    $.ajax({
      url: 'findPw',
      type: 'POST',
      data: {
        id: id,
      },
    })
      .done(function () {
        alert('메일발송 완료')
      })
      .fail(function () {
        alert('에러가 발생했습니다')
      })
  })
})
```

## Controller 생성

```java
@PostMapping("findId")
    public ResponseEntity<Object> sendEmail(String email) throws Exception {
        String to = email;
        UserVO userVO=userService.findId(email);
        String sub ="아이디 확인";
        String con = "아이디는 " + userVO.getId() + "입니다.";
        log.error("id는"+  userVO.getId());
        log.error(email);
        if( userVO.getId() != null) {
            mailManager.send(to,sub,con);
        }

        return new ResponseEntity<Object>(HttpStatus.OK);  //서버가 작업을 완료한 후 모든 것이 잘 되었는지 여부를 브라우저에 알리는기능
    }

    @PostMapping("findPw")
    @ResponseBody
    public String findPw(String id) throws Exception {
        UserVO userVO = userService.findEmail(id);
        UUID uuidPw = UUID.randomUUID();
        String tempPw = uuidPw.toString().substring(0, 7); // UUID 문자열 중 7자리만 사용하여 인증번호 생성
        String to = userVO.getEmail();
        String sub ="임시 비밀번호 전송";
        String con = "임시 비밀번호 : "+tempPw;
        userVO.setPassword(passwordEncoder.encode(tempPw));
        userVO.setId(id);
        log.error("이메일 : " + to);
        mailManager.send(to,sub,con);
        int result = userService.saveTempPw(userVO);
        return tempPw;
    }
```
