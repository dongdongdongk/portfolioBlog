const mongoose = require('mongoose')
const cities = require('./cities')
const { places, descriptors } = require('./seedHelpers')
const Campground = require('../models/campground')

mongoose.connect('mongodb://localhost:27017/yelp-camp', {
  useNewUrlParser: true,
  useUnifiedTopology: true,
})

const db = mongoose.connection

db.on('error', console.error.bind(console, 'connection error:'))
db.once('open', () => {
  console.log('Database connected')
})

const sample = (array) => array[Math.floor(Math.random() * array.length)]

const seedDB = async () => {
  await Campground.deleteMany({}) // 모두 삭제 후 진행
  for (let i = 0; i < 50; i++) {
    // 50 개의 랜덤 데이터를 시드에서 가져와서 생성
    const random1000 = Math.floor(Math.random() * 1000)
    const camp = new Campground({
      location: `${cities[random1000].city}, ${cities[random1000].state}`,
      title: `${sample(descriptors)} ${sample(places)}`,
    })
    await camp.save()
  }
}

seedDB().then(() => {
  // 완료 후 DB 닫기
  mongoose.connection.close()
})

import { useEffect, useState } from 'react'

const CampGroundDetail = () => {

      </button>

  )

```jsx
여기서

const [camp, setCamp] = useState({});

이부분을 null 이나 다른 [] 같은것으로 하면

**깜빡임 문제**

현재 useEffect 내에서 비동기 함수로 데이터를 가져오고 있다.

비동기 함수가 실행되는 동안 setCamp(null)이 호출되면 컴포넌트가 null로 렌더링된다.

이후 데이터가 로드되면 setCamp(response.data);으로 상태를 업데이트하게 된다.

이렇게 되면 컴포넌트가 처음에 null로 렌더링되고, 나중에 데이터가 로드되면 다시 렌더링되는데,

이로 인해 UI에서 깜빡이는 현상이 발생할 수 있다.

**데이터 유형 불일치**

useState(null)로 상태를 초기화하면, 컴포넌트가 처음 렌더링될 때

camp은 null이므로 {camp.title} 및 {camp.location}에서 null.title 및 null.location을 읽으려고 시도하면 오류가 발생할 것이다.

**결과**

데이터를 로드하기 전에 빈 객체로 초기화하면 오류를 방지하면서 깜빡임 없이 로딩 상태를 다룰 수 있다.

만약 로딩 상태를 명시적으로 다루고 싶다면, useState(null)을 사용할 수도 있다.

이 경우에는 UI에서 로딩 상태를 처리해주어야 한다.
```

import { useState } from 'react'

const CampGroundAdd = () => {

        <input type="text" name="title" id="title" value={formData.title} onChange={handleChange} />
        <label htmlFor="location">location</label>
