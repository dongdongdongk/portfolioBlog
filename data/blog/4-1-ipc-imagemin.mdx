## main.js

```jsx
const { app, BrowserWindow, Menu, globalShortcut } = require('electron')

// Set env

process.env.NODE_ENV = 'development'

const isDev = process.env.NODE_ENV !== 'production' ? true : false
const isMac = process.platform === 'darwin' ? true : false

let aboutWindow
let mainWindow

function createMainWindow() {
  mainWindow = new BrowserWindow({
    title: 'ImageShrink',
    width: 500,
    height: 600,
    icon: './assets/Icon_256x256.png',
    resizable: isDev,
    backgroundColor: 'white',
  })
  // mainWindow.loadURL('https://www.base64decode.org/')
  // mainWindow.loadURL(`file://${__dirname}/app/index.html`);
  mainWindow.loadFile('./app/index.html')
}

function createAboutWindow() {
  aboutWindow = new BrowserWindow({
    title: 'About ImageShrink',
    width: 300,
    height: 300,
    icon: './assets/Icon_256x256.png',
    resizable: false,
    backgroundColor: 'white',
  })
  aboutWindow.loadFile('./app/about.html')
}

app.on('ready', () => {
  createMainWindow()

  const mainMenu = Menu.buildFromTemplate(menu)
  Menu.setApplicationMenu(mainMenu)

  globalShortcut.register('CmdOrCtrl+R', () => mainWindow.reload())
  globalShortcut.register(isMac ? 'Command+Alt+I' : 'Ctrl+Shift+C', () =>
    mainWindow.toggleDevTools()
  )

  mainWindow.on('closed', () => (mainWindow = null))
})

const menu = [
  ...(isMac
    ? [
        {
          label: app.name,
          submenu: [
            {
              label: 'About',
              click: createAboutWindow,
            },
          ],
        },
      ]
    : []),
  {
    role: 'fileMenu',
  },
  ...(!isMac
    ? [
        {
          label: 'Help',
          submenu: [
            {
              label: 'About',
              click: createAboutWindow,
            },
          ],
        },
      ]
    : []),
  ...(isDev
    ? [
        {
          label: 'Developer',
          submenu: [
            { role: 'reload' },
            { type: 'separator' },
            { role: 'forcereload' },
            { type: 'separator' },
            { role: 'toggledevtools' },
          ],
        },
      ]
    : []),
]

app.on('activate', () => {
  if (BrowserWindow.getAllWindows().length === 0) {
    createMainWindow()
  }
})

app.on('window-all-closed', () => {
  if (!isMac) {
    app.quit()
  }
})

app.allowRendererProcessReuse = true
```

## 1️⃣ Electron의 주요 개념

```jsx
Electron은 HTML, CSS, JavaScript로 데스크톱 애플리케이션을 개발할 수 있도록 도와주는 프레임워크예요.
```

### 🖥️ Electron의 두 가지 프로세스

```jsx
1. 메인 프로세스 (Main Process)

		Node.js 환경에서 실행됨

		앱의 창을 생성하고 관리함

		파일 시스템, OS와 상호작용 가능

		main.js에서 실행됨

2. 렌더러 프로세스 (Renderer Process)

		Chrome 웹 브라우저처럼 동작

		HTML, CSS, JavaScript 실행

		preload.js를 통해 메인 프로세스와 통신
```

### Electron 앱의 동작 흐름

```jsx
메인 프로세스 (main.js) → 창(BrowserWindow) 생성 → index.html 로드

렌더러 프로세스 (index.html + renderer.js) → UI 조작
```

## 3️⃣ `main.js` (메인 프로세스)

```jsx
📌 역할:

		앱을 실행하고 창을 띄움 (BrowserWindow)

		다운로드 폴더와 프로젝트 폴더 경로를 계산

		렌더러 프로세스가 요청하면, 해당 경로를 전달
```

### `main.js` 코드

```jsx
const { app, BrowserWindow, ipcMain } = require('electron')
const path = require('path') // 경로 조작을 위한 Node.js 내장 모듈
const os = require('os') // OS 관련 정보 가져오기 위한 Node.js 내장 모듈

let mainWindow

function createMainWindow() {
  mainWindow = new BrowserWindow({
    title: 'ImageShrink',
    width: 800,
    height: 600,
    backgroundColor: 'white',
    webPreferences: {
      nodeIntegration: false, // 보안 강화를 위해 Node.js 기능 비활성화
      contextIsolation: true, // 메인 프로세스와 렌더러 프로세스를 격리
      preload: path.join(__dirname, 'preload.js'), // preload.js 로드
    },
  })

  mainWindow.loadFile('./app/index.html') // HTML 파일 로드

  mainWindow.on('closed', () => {
    mainWindow = null
  })
}

// 📌 IPC (프로세스 간 통신) - 렌더러 프로세스에서 요청하면 실행됨
ipcMain.handle('getOutputPath', async () => {
  return {
    downloads: path.join(os.homedir(), 'Downloads'), // 사용자의 다운로드 폴더 경로
    project: __dirname, // 현재 프로젝트 폴더 경로
  }
})

// Electron 앱 실행
app.whenReady().then(createMainWindow)

app.on('window-all-closed', () => {
  if (process.platform !== 'darwin') {
    app.quit()
  }
})
```

## 4️⃣ `preload.js` (메인 ↔ 렌더러 프로세스 연결)

```jsx
📌 preload.js 역할:

		메인 프로세스(main.js)와 렌더러 프로세스(index.html) 간 통신을 중계

		contextBridge.exposeInMainWorld를 이용해 보안 문제 방지
```

### `preload.js` 코드

```jsx
// Electron에서 `contextBridge`와 `ipcRenderer` 모듈 가져오기
// - `contextBridge` : 보안 강화를 위해 메인 프로세스 기능을 렌더러 프로세스에서 안전하게 사용할 수 있도록 함
// - `ipcRenderer` : 렌더러 프로세스에서 메인 프로세스로 데이터를 주고받을 수 있도록 하는 API
const { contextBridge, ipcRenderer } = require('electron')

// `contextBridge.exposeInMainWorld`를 사용하여 `electron` 객체를 `window` 전역 객체에 등록
// - 이렇게 하면 렌더러 프로세스(HTML/JS)에서 `window.electron.getOutputPath()` 형태로 사용할 수 있음
contextBridge.exposeInMainWorld('electron', {
  // `getOutputPath`라는 함수를 등록 (렌더러에서 호출 가능)
  getOutputPath: () => ipcRenderer.invoke('getOutputPath'),
  // `ipcRenderer.invoke('getOutputPath')` :
  //   - 메인 프로세스(main.js)에 'getOutputPath' 요청을 보냄
  //   - 요청을 받은 메인 프로세스는 해당 경로 데이터를 계산 후 응답을 반환
  //   - 응답 데이터는 `Promise` 형태로 반환됨 (비동기 처리됨)
})
```

```jsx
window.electron.getOutputPath()를 호출하면

ipcRenderer.invoke('getOutputPath')가 실행됨

main.js에서 ipcMain.handle('getOutputPath', ...)가 응답을 반환

경로 데이터를 렌더러 프로세스로 전달
```

## 6️⃣ `renderer.js` (렌더러 프로세스)

```jsx
📌 renderer.js 역할:

메인 프로세스로부터 다운로드 폴더와 프로젝트 폴더 경로를 받아서 HTML에 표시
```

```jsx
// Electron의 메인 프로세스에서 `getOutputPath` 함수를 호출하고, 그 결과를 받아옴
window.electron
  .getOutputPath()
  .then((paths) => {
    // 비동기 응답이 오면 실행되는 콜백 함수
    // 📂 다운로드 폴더 경로를 콘솔에 출력
    console.log('📂 다운로드 폴더:', paths.downloads)

    // 📁 프로젝트 폴더 경로를 콘솔에 출력
    console.log('📁 프로젝트 폴더:', paths.project)

    // HTML 문서에서 ID가 'downloads-path'인 요소를 찾아 다운로드 폴더 경로를 표시
    document.getElementById('downloads-path').innerText = paths.downloads

    // HTML 문서에서 ID가 'project-path'인 요소를 찾아 프로젝트 폴더 경로를 표시
    document.getElementById('project-path').innerText = paths.project
  })
  .catch((err) => console.error('경로 가져오기 실패:', err)) // 만약 오류가 발생하면 콘솔에 에러 메시지 출력
```

```jsx
window.electron.getOutputPath() 호출

preload.js에서 ipcRenderer.invoke('getOutputPath') 실행

main.js에서 ipcMain.handle('getOutputPath', ...)이 다운로드/프로젝트 폴더 경로 반환

HTML id="downloads-path", id="project-path" 부분을 업데이트
```

### 호출 되자마자 실행되는 getOutputPath()

```jsx
이 코드는 함수 정의가 아니라 함수 호출이야.

JavaScript는 위에서 아래로 코드가 실행되기 때문에,

renderer.js가 로드되는 순간 자동으로 실행됨.

즉, 어떤 이벤트를 기다리는 것이 아니라, renderer.js가 실행되면서 즉시 실행되는 코드임.
```

# { `invoke`의 의미 ( 여러 ipcRenderer 통신 방법 ) }

```jsx
ipcRenderer.invoke(channel, ...args)는 Promise 기반의 IPC(Inter-Process Communication) 요청을 의미합니다.

즉, 렌더러 프로세스가 메인 프로세스에 요청을 보내고,

메인 프로세스에서 처리한 후 응답을 반환하는 방식입니다.

아래 코드에서 invoke('dialog:selectFile')를 실행하면:

dialog:selectFile이라는 채널을 통해 메인 프로세스에 요청을 보냄.

메인 프로세스에서 해당 요청을 처리한 후 Promise 형태로 응답을 반환함.

렌더러 프로세스는 await 또는 .then()을 사용하여 응답을 받을 수 있음.
```

## 여러 ipcRenderer 통신 방법

![image.png](https://res.cloudinary.com/du1acdhxq/image/upload/v1788381247/notion-blog/_/Electron/Electron_From_Scratch_JavaScript_/_4_1-_IPC_ImageMin_/image_2.png)

## 사용 예시 시나리오

```jsx
1️⃣ invoke (비동기 요청-응답)

사용자가 "파일 열기" 버튼을 누르면, 파일 선택 대화상자를 띄우고 선택한 파일 경로를 렌더러로 반환

2️⃣ send (단방향 메시지 전송)

사용자가 "로그 저장" 버튼을 누르면, 메인 프로세스에 로그 저장 요청을 보냄 (응답 불필요)

3️⃣ sendSync (동기 요청-응답)

앱 시작 시 config.json에서 "언어 설정" 값을 즉시 가져와야 함 (UI 표시 전에 필요)

4️⃣ on (이벤트 수신)

메인 프로세스가 1초마다 CPU 사용량을 계산하여 렌더러에 전달 (UI 실시간 업데이트)

5️⃣ once (한 번만 수신)

앱 최초 실행 시 "환경 설정값"을 메인 프로세스에서 로드한 후 렌더러로 전송 (추가 호출 없음)

6️⃣ reply (메인 → 렌더러 응답)
```

## 1. 기본 개념

```jsx
Electron은 크게 두 개의 프로세스로 구성됩니다:

1. 메인 프로세스 (main.js)

		운영체제와 직접 상호작용

		파일 시스템 접근 가능

		시스템 다이얼로그 실행 가능

2. 렌더러 프로세스 (renderer.js)

		웹 페이지를 표시

		사용자 인터페이스 담당

		보안상 시스템 직접 접근 불가
```

## 2. 통신 과정

```jsx
1. 사용자가 '파일 선택' 버튼 클릭

2. renderer.js의 클릭 이벤트 핸들러 실행

3. window.electron.selectFile() 호출

4. preload.js를 통해 메인 프로세스에 'dialog:selectFile' 이벤트 전송

5. main.js에서 시스템 파일 선택 다이얼로그 표시

6. 사용자가 파일 선택

7. 선택된 파일 경로가 렌더러로 전달

8. 파일 경로가 화면에 표시
```

```jsx
// 이미지 압축을 수행하는 비동기 함수
async function shrinkImage({ imgPath, quality, dest }) {
  try {
    // 필요 라이브러리 동적 가져오기 (ES 모듈 방식) import 를 사용하지 못해서 이렇게 가져왔다
    const { default: imagemin } = await import('imagemin') // 이미지 최적화 라이브러리
    const { default: imageminMozjpeg } = await import('imagemin-mozjpeg') // JPEG 압축 플러그인
    const { default: imageminPngquant } = await import('imagemin-pngquant') // PNG 압축 플러그인
    const slash = (await import('slash')).default // Windows 경로를 UNIX 스타일 경로로 변환

    // PNG 압축 품질을 0~1 범위로 변환 (예: 80 -> 0.8)
    const pngQuality = quality / 100

    // 이미지 압축 실행
    const files = await imagemin([slash(imgPath)], {
      // 선택한 이미지 파일 경로를 변환하여 압축 실행
      destination: dest, // 압축된 파일 저장 경로
      plugins: [
        imageminMozjpeg({ quality }), // JPEG 파일 품질 설정 (0~100)
        imageminPngquant({
          quality: [pngQuality, pngQuality], // PNG 파일 품질 설정 (0~1 범위)
        }),
      ],
    })

    // 압축된 파일 정보 출력 (배열 형태)
    console.log(files)

    // 압축된 이미지가 저장된 폴더를 파일 탐색기로 엶
    shell.openPath(dest)
  } catch (error) {
    // 오류 발생 시 콘솔에 출력
    console.log(error)
  }
}
```

```jsx
이미지 압축 라이브러리 동적 로드

		import() 사용 → Electron의 require() 대신 최신 방식 적용

		imagemin, imagemin-mozjpeg, imagemin-pngquant 를 가져옴 ( import 를 사용하지 못해서 이렇게 가져왔다 )

이미지 경로 변환

		slash(imgPath) : 윈도우와 리눅스/맥의 경로 차이를 해결

압축 실행

		imageminMozjpeg({ quality }) : JPEG 품질 설정

		imageminPngquant({ quality: [pngQuality, pngQuality] }) : PNG 품질 설정

		압축된 파일을 지정된 폴더 (dest)에 저장

				* 왜 [pngQuality, pngQuality] 같은 값을 쓰나요?

					quality: [0.8, 0.8]처럼 최소와 최대를 동일하게 설정하면,

					항상 고정된 품질로 압축됨.

					만약 quality: [0.6, 0.8]처럼 최소와 최대를 다르게 설정하면,

					플러그인이 최적의 품질을 자동 선택함.

					코드에서는 사용자가 설정한 품질을 유지하려고 [pngQuality, pngQuality]로 고정한 것.

압축 완료 후 저장 폴더 열기

		shell.openPath(dest) : 사용자에게 압축된 파일 확인 가능
```
