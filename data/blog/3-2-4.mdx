## 1. 홈 화면에서 영화 데이터 불러오기

이전에 만든 `useFetch` + `fetchMovies`를 조합하면 단 몇 줄로 끝납니다.

```tsx
// app/(tabs)/index.tsx

import { ActivityIndicator, FlatList, Image, ScrollView, Text, View } from 'react-native'
import { images, icons } from '@/constants'
import { useRouter } from 'expo-router'
import useFetch from '@/services/useFetch'
import { fetchMovies } from '@/services/api'
import SearchBar from '@/components/SearchBar'
import MovieCard from '@/components/MovieCard'

export default function Home() {
  const router = useRouter()

  // useFetch에 fetchMovies를 넘기면
  // 로딩/에러/데이터 관리를 알아서 해줌
  const {
    data: movies, // data를 movies로 이름 변경
    loading: moviesLoading,
    error: moviesError,
  } = useFetch(() => fetchMovies(''))
  // 빈 문자열("") → 인기 영화 목록 가져오기
  // "Iron Man" 처럼 검색어를 넣으면 검색 결과 가져오기

  return (
    <View className="bg-primary flex-1">
      {/* 배경 이미지 */}
      <Image source={images.bg} className="absolute z-0 w-full" />

      <ScrollView
        className="flex-1 px-5"
        showsVerticalScrollIndicator={false}
        contentContainerStyle={{ minHeight: '100%', paddingBottom: 10 }}
      >
        {/* 앱 로고 */}
        <Image source={icons.logo} className="mx-auto mt-20 mb-5 h-10 w-12" />

        {/* 로딩 중 → 스피너 표시 */}
        {moviesLoading ? (
          <ActivityIndicator size="large" color="#0000ff" className="mt-10 self-center" />
        ) : // 에러 발생 → 에러 메시지 표시
        moviesError ? (
          <Text className="mt-10 text-center text-red-500">에러: {moviesError?.message}</Text>
        ) : (
          // 정상 → 검색바 + 영화 목록 표시
          <View className="mt-5 flex-1">
            {/* 검색창 — 누르면 검색 화면으로 이동 */}
            <SearchBar placeholder="영화를 검색해보세요" onPress={() => router.push('/search')} />

            {/* 영화 목록 섹션 */}
            <>
              <Text className="mt-5 mb-3 text-lg font-bold text-white">최신 영화</Text>

              {/* FlatList로 영화 목록 렌더링 */}
              <FlatList
                data={movies} // 영화 배열 데이터
                renderItem={({ item }) => (
                  <MovieCard {...item} /> // 각 영화를 카드로 표시
                )}
                keyExtractor={(item) => item.id.toString()} // 각 항목의 고유 키
                numColumns={3} // 3열로 표시
                columnWrapperStyle={{
                  justifyContent: 'flex-start',
                  gap: 20,
                  paddingRight: 5,
                  marginBottom: 10,
                }}
                className="mt-2 pb-32"
                scrollEnabled={false} // ScrollView 안에 있으므로 자체 스크롤 끄기
              />
            </>
          </View>
        )}
      </ScrollView>
    </View>
  )
}
```

```
components/
└── MovieCard.tsx   ← 새로 생성
```

### 전체 코드 + 주석

```tsx
// components/MovieCard.tsx

import { Link } from 'expo-router'
import { Image, Text, TouchableOpacity, View } from 'react-native'
import { icons } from '@/constants/icons'

// Movie 타입에서 필요한 props만 받아옴
// (id, poster_path, title, vote_average, release_date)
const MovieCard = ({ id, poster_path, title, vote_average, release_date }: Movie) => {
  return (
    // Link: 카드 클릭 시 영화 상세 페이지로 이동
    // href={`/movie/${id}`} → 동적 라우트 /movie/[id]로 이동
    // asChild: Link 자체가 아닌 안에 있는 자식(TouchableOpacity)이 클릭 영역이 됨
    <Link href={`/movie/${id}`} asChild>
      {/* TouchableOpacity: 눌렀을 때 투명도가 낮아지는 시각적 피드백 */}
      {/* w-[30%]: 화면 너비의 30% (3열 레이아웃에 맞춤) */}
      <TouchableOpacity className="w-[30%]">
        {/* 영화 포스터 이미지 */}
        <Image
          source={{
            uri: poster_path
              ? // 포스터가 있으면 TMDB 이미지 서버에서 불러옴
                `https://image.tmdb.org/t/p/w500${poster_path}`
              : // 포스터가 없으면 회색 placeholder 이미지 표시
                'https://placehold.co/600x400/1a1a1a/FFFFFF.png',
          }}
          className="h-52 w-full rounded-lg"
          resizeMode="cover" // 이미지가 영역을 꽉 채우도록 크롭
        />

        {/* 영화 제목 — numberOfLines={1}로 한 줄만 표시 (넘치면 ... 처리) */}
        <Text className="mt-2 text-sm font-bold text-white" numberOfLines={1}>
          {title}
        </Text>

        {/* 별점 영역 */}
        <View className="flex-row items-center justify-start gap-x-1">
          <Image source={icons.star} className="size-4" />
          <Text className="text-xs font-bold text-white uppercase">
            {/* vote_average는 10점 만점 → 2로 나눠서 5점 만점으로 변환 */}
            {Math.round(vote_average / 2)}
          </Text>
        </View>

        {/* 개봉 연도 */}
        <View className="flex-row items-center justify-between">
          <Text className="text-light-300 mt-1 text-xs font-medium">
            {/* "2024-07-15" → "2024" (연도만 추출) */}
            {release_date?.split('-')[0]}
          </Text>
          <Text className="text-light-300 text-xs font-medium uppercase">Movie</Text>
        </View>
      </TouchableOpacity>
    </Link>
  )
}

export default MovieCard
```

## 4. 핵심 개념 정리

### FlatList 주요 props

| prop                 | 역할                      | 예시                           |
| -------------------- | ------------------------- | ------------------------------ |
| `data`               | 렌더링할 배열             | `data={movies}`                |
| `renderItem`         | 각 항목을 어떻게 표시할지 | `({ item }) => <MovieCard />`  |
| `keyExtractor`       | 각 항목의 고유 키         | `(item) => item.id.toString()` |
| `numColumns`         | 몇 열로 표시할지          | `numColumns={3}`               |
| `columnWrapperStyle` | 각 행의 스타일            | `{ gap: 20 }`                  |
| `scrollEnabled`      | 자체 스크롤 여부          | `scrollEnabled={false}`        |

> 💡 `numColumns`를 바꾸면 앱을 **반드시 재시작**해야 적용됩니다.
>
> 저장만 해서는 반영되지 않습니다.

### props 전달 방법 — 스프레드 연산자

```tsx
// 방법 1: 하나씩 전달 (번거로움)
<MovieCard
  id={item.id}
  title={item.title}
  poster_path={item.poster_path}
  vote_average={item.vote_average}
  release_date={item.release_date}
/>

// 방법 2: 스프레드로 한 번에 전달 (간결함) ✅
<MovieCard {...item} />
// item 객체의 모든 속성이 props로 전달됨
```

### 날짜에서 연도만 추출

```tsx
const release_date = '2024-07-15'

release_date.split('-')
// → ["2024", "07", "15"]

release_date.split('-')[0]
// → "2024"  ← 연도만 가져옴
```

### 별점 변환 (10점 → 5점)

```tsx
// TMDB는 10점 만점 기준
const vote_average = 7.8

Math.round(vote_average / 2)
// 7.8 / 2 = 3.9 → Math.round → 4
// → 별 4개로 표시
```

# { 🔍 검색 화면(Search Screen) 완전 가이드 }

> 실시간 검색 + 디바운싱 + 빈 상태 처리까지

---

## 📋 이번 강의에서 만드는 것

```
검색 화면 (search.tsx)
├── 배경 이미지
└── FlatList
    ├── ListHeaderComponent (스크롤해도 고정되는 헤더)
    │   ├── 로고
    │   ├── SearchBar (입력창)
    │   ├── 로딩 스피너 (검색 중일 때)
    │   ├── 에러 메시지 (에러 발생 시)
    │   └── "검색 결과: [검색어]" 텍스트
    ├── renderItem → MovieCard × N개
    └── ListEmptyComponent (검색 결과 없을 때)
        └── "검색어 없음" or "결과 없음" 메시지
```

---

## 💡 홈 화면과 검색 화면의 차이

|              | 홈 화면                   | 검색 화면                 |
| ------------ | ------------------------- | ------------------------- |
| 레이아웃     | ScrollView + FlatList     | FlatList만 사용           |
| 검색바 위치  | ScrollView 안에 그냥 배치 | FlatList의 헤더로 배치    |
| 데이터 fetch | 자동 (`autoFetch: true`)  | 수동 (`autoFetch: false`) |
| 검색어       | 없음 (인기 영화)          | useState로 관리           |

---

## 1. 기본 구조 — FlatList 하나로 전체 화면 구성

홈 화면은 `ScrollView` 안에 `FlatList`를 넣었지만,
검색 화면은 검색바가 **항상 위에 고정**되어야 하므로 `FlatList` 하나로 해결합니다.

```tsx
// 핵심 구조
<FlatList
  data={movies}
  renderItem={...}
  ListHeaderComponent={...}  // ← 목록 맨 위에 고정되는 영역 (로고 + 검색바)
  ListEmptyComponent={...}   // ← 결과가 없을 때 보여줄 내용
/>
```

## 2. 전체 코드 + 상세 주석

```tsx
// app/(tabs)/search.tsx

import { useState, useEffect } from 'react'
import { View, Text, ActivityIndicator, FlatList, Image } from 'react-native'
import { images } from '@/constants/images'
import { icons } from '@/constants/icons'
import useFetch from '@/services/useFetch'
import { fetchMovies } from '@/services/api'
import SearchBar from '@/components/SearchBar'
import MovieCard from '@/components/MovieCard'

const Search = () => {
  // 사용자가 입력한 검색어를 저장하는 상태
  const [searchQuery, setSearchQuery] = useState('')

  const {
    data: movies = [], // 검색 결과 영화 목록 (기본값: 빈 배열)
    loading, // 검색 중 여부
    error, // 에러 정보
    refetch: loadMovies, // 수동으로 데이터를 가져오는 함수
    reset, // 상태를 초기화하는 함수
  } = useFetch(
    () => fetchMovies({ query: searchQuery }), // 현재 검색어로 API 호출
    false // ← autoFetch: false → 자동으로 fetch하지 않음
    //   사용자가 검색어를 입력할 때만 fetch
  )

  // 검색어가 바뀔 때마다 실행되는 함수
  const handleSearch = (text: string) => {
    setSearchQuery(text)
    // setSearchQuery가 실행되면 아래 useEffect가 자동으로 호출됨
  }

  // ── 디바운싱(Debouncing) ────────────────────────────
  // 문제: 타이핑할 때마다 API를 호출하면?
  //   "I", "Ir", "Iro", "Iron", "Iron " ... → 수십 번 API 호출
  //
  // 해결: 마지막 입력 후 500ms 동안 추가 입력이 없으면 그때 API 호출
  //   "Iron Man" 타이핑 완료 → 0.5초 기다림 → API 1번만 호출
  useEffect(() => {
    // setTimeout: 지정한 시간(500ms) 후에 함수를 실행
    const timeoutId = setTimeout(async () => {
      const func = async () => {
        if (searchQuery.trim()) {
          // 검색어가 있으면 영화 검색 실행
          await loadMovies()
        } else {
          // 검색어가 비어있으면 결과 초기화
          reset()
        }
      }

      func()
    }, 500) // 500ms = 0.5초

    // 클린업 함수: searchQuery가 바뀌면 이전 타이머를 취소
    // → 타이핑 중에는 타이머가 계속 리셋됨
    // → 타이핑을 멈춰야 500ms 후에 API가 호출됨
    return () => clearTimeout(timeoutId)
  }, [searchQuery]) // searchQuery가 바뀔 때마다 이 effect 실행
  // ────────────────────────────────────────────────────

  return (
    <View className="bg-primary flex-1">
      {/* 배경 이미지 */}
      <Image source={images.bg} className="absolute z-0 w-full flex-1" resizeMode="cover" />

      <FlatList
        className="px-5"
        data={movies as Movie[]}
        keyExtractor={(item) => item.id.toString()}
        renderItem={({ item }) => <MovieCard {...item} />}
        numColumns={3}
        columnWrapperStyle={{
          justifyContent: 'flex-start',
          gap: 16,
          marginVertical: 16,
        }}
        contentContainerStyle={{ paddingBottom: 100 }}
        // ── ListHeaderComponent ────────────────────────
        // 목록 맨 위에 고정되는 영역
        // 스크롤해도 항상 위에 있는 게 아니라, 목록과 함께 스크롤됨
        // 하지만 목록보다 항상 위에 렌더링됨
        ListHeaderComponent={
          <>
            {/* 앱 로고 */}
            <View className="mt-20 w-full flex-row items-center justify-center">
              <Image source={icons.logo} className="h-10 w-12" />
            </View>

            {/* 검색창 */}
            <View className="my-5">
              <SearchBar
                placeholder="영화를 검색하세요"
                value={searchQuery} // 현재 입력된 검색어
                onChangeText={handleSearch} // 텍스트 변경 시 handleSearch 호출
              />
            </View>

            {/* 로딩 중일 때 스피너 표시 */}
            {loading && <ActivityIndicator size="large" color="#0000ff" className="my-3" />}

            {/* 에러 발생 시 에러 메시지 표시 */}
            {error && <Text className="my-3 px-5 text-red-500">에러: {error.message}</Text>}

            {/* 검색 결과가 있을 때 "검색 결과: [검색어]" 표시 */}
            {!loading && !error && searchQuery.trim() && movies?.length > 0 && (
              <Text className="text-xl font-bold text-white">
                검색 결과: <Text className="text-accent">{searchQuery}</Text>
              </Text>
            )}
          </>
        }
        // ── ListEmptyComponent ─────────────────────────
        // data 배열이 비어있을 때 보여줄 컴포넌트
        // 로딩 중이거나 에러가 있으면 null (다른 UI가 이미 표시되므로)
        ListEmptyComponent={
          !loading && !error ? (
            <View className="mt-10 px-5">
              <Text className="text-center text-gray-500">
                {searchQuery.trim()
                  ? '검색 결과가 없습니다' // 검색했는데 결과 없음
                  : '영화 제목을 검색해보세요'}{' '}
                // 아직 검색 안 함
              </Text>
            </View>
          ) : null
        }
      />
    </View>
  )
}
```

## 3. 핵심 개념 — 디바운싱(Debouncing)

타이핑할 때마다 API를 호출하면 서버에 과부하가 걸립니다.
**디바운싱**은 마지막 입력 후 일정 시간이 지나야 API를 호출하는 기법입니다.

```
❌ 디바운싱 없을 때:
  "I"    → API 호출
  "Ir"   → API 호출
  "Iro"  → API 호출
  "Iron" → API 호출
  총 4번 호출

✅ 디바운싱 있을 때 (500ms):
  "I"    → 타이머 시작 (0ms)
  "Ir"   → 타이머 리셋 (100ms)
  "Iro"  → 타이머 리셋 (200ms)
  "Iron" → 타이머 리셋 (300ms)
  ── 0.5초 동안 추가 입력 없음 ──
           → API 호출 1번!
```

```tsx
useEffect(() => {
  // 500ms 후에 실행될 타이머 등록
  const timeoutId = setTimeout(async () => {
    if (searchQuery.trim()) {
      await loadMovies()
    } else {
      reset()
    }
  }, 500)

  // 검색어가 바뀌면 이전 타이머를 취소 → 타이머 리셋
  return () => clearTimeout(timeoutId)
}, [searchQuery])
```

### 함수 1: `updateSearchCount` — 검색 기록 저장/업데이트

```tsx
export const updateSearchCount = async (query: string, movie: Movie) => {
  // query: 사용자가 검색한 단어 (예: "어벤져스")
  // movie: 검색 결과의 첫 번째 영화

  try {
    // ── STEP 1: DB에서 이 검색어의 기록을 찾기 ──────────
    const result = await database.listDocuments(
      DATABASE_ID, // 어느 데이터베이스에서?
      COLLECTION_ID, // 어느 컬렉션에서?
      [
        // Query.equal: 특정 필드가 특정 값과 같은 문서를 검색
        // SQL로 비유: WHERE searchTerm = '어벤져스'
        Query.equal('searchTerm', query),
      ]
    )

    // ── STEP 2: 기록이 있으면 count 증가, 없으면 새로 생성 ──
    if (result.documents.length > 0) {
      // 이미 검색된 적 있음 → count를 1 올림
      const existingMovie = result.documents[0] // 첫 번째 문서 가져오기

      await database.updateDocument(
        DATABASE_ID,
        COLLECTION_ID,
        existingMovie.$id, // $id: Appwrite가 자동으로 부여한 문서 고유 ID
        {
          count: existingMovie.count + 1, // 기존 count에 1 추가
        }
      )
      // 예: count가 3이었으면 → 4로 업데이트
    } else {
      // 처음 검색됨 → 새 문서 생성
      await database.createDocument(
        DATABASE_ID,
        COLLECTION_ID,
        ID.unique(), // 새 문서의 ID를 자동으로 생성
        {
          searchTerm: query, // 검색어
          movie_id: movie.id, // 영화 ID
          title: movie.title, // 영화 제목
          count: 1, // 처음이므로 count = 1
          poster_url: `https://image.tmdb.org/t/p/w500${movie.poster_path}`,
          // 포스터 이미지 전체 URL 조합해서 저장
        }
      )
    }
  } catch (error) {
    console.error('검색 카운트 업데이트 실패:', error)
    throw error // 에러를 다시 던져서 호출한 곳에서도 알 수 있게 함
  }
}
```

```tsx
export const getTrendingMovies = async (): Promise<TrendingMovie[] | undefined> => {
  try {
    const result = await database.listDocuments(DATABASE_ID, COLLECTION_ID, [
      Query.limit(5), // 상위 5개만 가져오기

    return undefined // 에러 발생 시 undefined 반환 (앱이 터지지 않도록)
  }
}
```

## 1. 두 개의 useEffect 분리 — 버그 수정

검색 화면에서 `loadMovies`와 `updateSearchCount`를 같은 useEffect에 넣으면
두 함수가 경쟁해서 DB가 제대로 업데이트되지 않는 문제가 발생합니다.

```tsx
// ❌ 잘못된 방법 — 같은 useEffect에서 둘 다 처리
useEffect(() => {
  const timeoutId = setTimeout(async () => {
    if (searchQuery.trim()) {
      await loadMovies()
      await updateSearchCount(searchQuery, movies[0]) // movies가 아직 비어있음!
    }
  }, 500)
  return () => clearTimeout(timeoutId)
}, [searchQuery])
```

```tsx
// ✅ 올바른 방법 — useEffect를 역할에 따라 분리

// 1번 useEffect: 검색어가 바뀌면 영화를 불러옴
useEffect(() => {
  const timeoutId = setTimeout(async () => {
    if (searchQuery.trim()) {
      await loadMovies() // 여기서 movies 상태가 업데이트됨
    } else {
      reset()
    }
  }, 500)
  return () => clearTimeout(timeoutId)
}, [searchQuery]) // 검색어가 바뀔 때 실행

// 2번 useEffect: movies가 업데이트되면 검색 카운트를 저장
useEffect(() => {
  if (movies && movies.length > 0 && movies[0]) {
    updateSearchCount(searchQuery, movies[0]) // movies가 채워진 후 실행됨
  }
}, [movies]) // movies가 바뀔 때 실행
```

> 💡 **의존성 배열의 차이**
>
> - `[searchQuery]` → 검색어가 바뀔 때마다 실행
> - `[movies]` → 영화 데이터가 바뀔 때마다 실행
>
> 두 번째 useEffect는 첫 번째가 완료되어 movies가 채워진 **후에** 실행되므로 경쟁 조건이 해결됩니다.

```tsx
        <Image source={icons.logo} className="mx-auto mt-20 mb-5 h-10 w-12" />

        {/* 둘 중 하나라도 로딩 중이면 스피너 표시 */}
        {moviesLoading || trendingLoading ? (
          <ActivityIndicator size="large" color="#0000ff" className="mt-10 self-center" />
        ) : // 둘 중 하나라도 에러면 에러 메시지 표시
        moviesError || trendingError ? (
          <Text>에러: {moviesError?.message || trendingError?.message}</Text>
        ) : (
          // 정상 상태
          <View className="mt-5 flex-1">
            {/* 검색바 */}
            <SearchBar onPress={() => router.push('/search')} placeholder="영화를 검색하세요" />

            {/* 트렌딩 영화 섹션 — 데이터가 있을 때만 표시 */}
            {trendingMovies && (
              <View className="mt-10">
                <Text className="mb-3 text-lg font-bold text-white">🔥 트렌딩 영화</Text>

                {/* 가로 스크롤 FlatList */}
                <FlatList
                  horizontal // ← 가로 방향으로 스크롤
                  showsHorizontalScrollIndicator={false}
                  className="mt-3 mb-4"
                  data={trendingMovies}
                  contentContainerStyle={{ gap: 26 }}
                  renderItem={({ item, index }) => (
                    <TrendingCard movie={item} index={index} />
                    // index: 0, 1, 2... → 카드에 1, 2, 3 숫자 표시용
                  )}
                  keyExtractor={(item) => item.movie_id.toString()}
                  ItemSeparatorComponent={() => <View className="w-4" />}
                  // ItemSeparatorComponent: 각 카드 사이에 넣을 구분 컴포넌트
                />
              </View>
            )}

            {/* 최신 영화 섹션 */}
            <>
              <Text className="mt-5 mb-3 text-lg font-bold text-white">최신 영화</Text>

              <FlatList
                data={movies}
                renderItem={({ item }) => <MovieCard {...item} />}
                keyExtractor={(item) => item.id.toString()}
                numColumns={3}
                columnWrapperStyle={{
                  justifyContent: 'flex-start',
                  gap: 20,
                  paddingRight: 5,
                  marginBottom: 10,
                }}
                className="mt-2 pb-32"
                scrollEnabled={false} // ScrollView 안이므로 자체 스크롤 끄기
              />
            </>
          </View>
        )}
      </ScrollView>
    </View>
  )
}
```

## 5. 가로 FlatList vs 세로 FlatList

|          | 트렌딩 (가로)                            | 최신 영화 (세로)                       |
| -------- | ---------------------------------------- | -------------------------------------- |
| 방향     | `horizontal` prop 추가                   | 기본값 (세로)                          |
| 스크롤   | 좌우로 스크롤                            | 상하로 스크롤                          |
| 열 개수  | 1행 (가로로 나열)                        | `numColumns={3}`                       |
| 구분선   | `ItemSeparatorComponent`                 | `columnWrapperStyle`의 `gap`           |
| 스크롤바 | `showsHorizontalScrollIndicator={false}` | `showsVerticalScrollIndicator={false}` |

> ⚠️ **같은 방향의 FlatList를 중첩하면 에러 발생**
>
> - ScrollView(세로) + FlatList(세로) → ❌ 에러
> - ScrollView(세로) + FlatList(가로) → ✅ 정상

## 6. useFetch에 함수 넘기는 두 가지 방법

```tsx
// 방법 1: 인자가 있는 함수 → 화살표 함수로 감싸기
useFetch(() => fetchMovies({ query: '' }))

// 방법 2: 인자가 없는 함수 → 직접 넘기기 (더 간결)
useFetch(getTrendingMovies)
```

> 💡 `getTrendingMovies`는 호출할 때 인자가 필요 없으므로 직접 넘길 수 있습니다.
> `fetchMovies`는 query가 필요하므로 화살표 함수로 감싸야 합니다.

---

## 🔑 핵심 요약

1. **useEffect 분리** = 경쟁 조건 버그 수정 — 역할이 다른 로직은 분리
2. **`horizontal` prop** = FlatList를 가로 스크롤로 전환하는 한 단어
3. **`ItemSeparatorComponent`** = 각 아이템 사이에 자동으로 렌더링되는 구분 컴포넌트
4. **`MaskedView`** = 텍스트/도형 모양대로 이미지를 잘라서 보여주는 효과
5. **`index + 1`** = 배열 인덱스는 0부터 시작하므로 +1해서 1, 2, 3으로 표시
6. **두 FlatList 방향 달리하기** = 세로 ScrollView 안에서 가로 FlatList는 허용됨
7. **`trendingMovies &&`** = 데이터가 있을 때만 섹션 전체를 렌더링

```tsx
export const fetchMovieDetails = async (movieId: string): Promise<MovieDetails> => {
```

```tsx
const Details = () => {
```

```tsx
// genres: [{ name: "액션" }, { name: "어드벤처" }, { name: "드라마" }]

movie?.genres?.map((g) => g.name).join(' • ')
// → "액션 • 어드벤처 • 드라마"
```

### 큰 숫자를 백만 단위로 표시

```tsx
// budget: 310000000 (3억 1천만)

;`$${movie?.budget / 1_000_000} million`
// → "$310 million"

`$${Math.round(movie?.revenue / 1_000_000)} million`
// → "$458 million"  (반올림 포함)
```

### 날짜에서 연도만 추출

```tsx
// release_date: "2024-11-15"

movie?.release_date?.split('-')[0]
// → "2024"
```

### null/undefined 대비 안전한 접근

```tsx
// ?. (옵셔널 체이닝): 값이 없으면 에러 대신 undefined 반환
movie?.overview // movie가 null이면 undefined

// ?? (nullish coalescing): null/undefined일 때 기본값 사용
movie?.vote_average ?? 0 // vote_average가 없으면 0

// ||: falsy한 값(null, undefined, 0, "") 모두 대체
value || 'N/A' // value가 없으면 "N/A" 표시
```

## 6. ScrollView vs FlatList — 성능 비교

이 강의에서 가장 중요한 최적화 개념입니다.

```
ScrollView
├── 장점: 구현 간단
└── 단점: 모든 자식 컴포넌트를 한 번에 렌더링
          → 아이템이 많으면 느려지고 메모리 낭비

FlatList
├── 장점: 화면에 보이는 것만 렌더링 (lazy rendering)
│         화면 밖으로 나간 아이템은 메모리에서 제거
└── 단점: 구현이 조금 더 복잡

[결론]
아이템 적음 (20개 이하) → ScrollView, FlatList 둘 다 OK
아이템 많음 (수백~수천개) → 반드시 FlatList 사용!
```

### 주의: 같은 방향 중첩 금지

```
❌ 문제 있는 구조
ScrollView (세로)
  └── FlatList (세로, scrollEnabled=false)
       — FlatList의 장점(lazy rendering)을 전혀 못 씀

✅ 올바른 구조 (작은 앱)
ScrollView (세로)
  └── FlatList (가로) ← 방향이 달라서 OK

✅ 더 나은 구조 (큰 앱)
FlatList (세로)
  └── ListHeaderComponent → 로고, 검색바, 트렌딩 카드
  └── renderItem → MovieCard
      — 전체를 FlatList 하나로 처리
```

## 🔑 핵심 요약

1. **`StatusBar hidden`** = 상태바 숨기기, `Stack` 형제로 놓으려면 Fragment 필요
2. **`MovieInfo` 컴포넌트** = 반복되는 레이블+값 UI를 재사용 컴포넌트로 분리
3. **`useLocalSearchParams`** = URL의 동적 파라미터 읽기 (`[id]` → `id`)
4. **`router.back`** = 이전 화면으로 이동 (`router.push`의 반대)
5. **`?.` 옵셔널 체이닝** = null/undefined 안전하게 접근
6. **`?? 0`** = null/undefined일 때 기본값 0으로 대체
7. **`rotate-180`** = 화살표 이미지를 CSS로 뒤집어서 뒤로가기 아이콘으로 활용
8. **FlatList > ScrollView** = 아이템이 많을수록 FlatList가 성능상 유리
