# 셀레니움을 사용하여 검색어 입력하고 검색 버튼 눌러 검색 결과로 넘어가기

```python
from selenium import webdriver
from selenium.webdriver.chrome.options import Options
from selenium.webdriver.common.by import By
import time

# Chrome WebDriver 옵션 설정
options = Options()

# 브라우저를 최대화하여 시작
options.add_argument("--start-maximized")
# WebDriver가 실행된 후에도 브라우저를 닫지 않고 유지
options.add_experimental_option("detach", True)

# Chrome WebDriver 객체 생성
driver = webdriver.Chrome(options=options)

# 접속할 URL 설정
url = "https://naver.com"

# 지정된 URL로 이동
driver.get(url)

# 2초 대기
time.sleep(2)

# 검색어 입력 필드에 "뉴진스" 입력
driver.find_element(By.ID, "query").send_keys("뉴진스")

# 2초 대기
time.sleep(2)

# 검색 버튼 클릭
driver.find_element(By.CSS_SELECTOR, ".btn_search").click()

```

```python

from selenium.webdriver.common.by import By

사용은 driver.find_element(By.ID, "query").send_keys("뉴진스") 이런식으로 사용하는데
```

# VIEW탭 클릭하여 페이지 넘어가기(XPATH 사용하여 원하는 텍스트 찾아내는 방법)

```python
XPath는 XML 문서의 특정 요소를 찾기 위한 경로 언어이다.

웹 스크래핑에서는 주로 HTML 문서에서 요소를 선택하기 위해 사용된다.

XPath는 다양한 문법을 제공하여 요소를 정확하게 식별할 수 있다.

( xpath 는 공부가 조금 필요한 언어이고 텍스트를 선택하는 등 다양한 방법으로 사용 가능하다 )

* 강의자는 css 선택자를 더 추천했다
```

```python
from selenium import webdriver
from selenium.webdriver.chrome.options import Options
from selenium.webdriver.common.by import By
import time

options = Options()

options.add_argument("--start-maximized")
options.add_experimental_option("detach", True)

driver = webdriver.Chrome(options=options)

url = "https://naver.com"

driver.get(url)
time.sleep(2)

driver.find_element(By.ID, "query").send_keys("뉴진스")
time.sleep(2)

driver.find_element(By.CSS_SELECTOR, ".btn_search").click()
time.sleep(3)

# driver.find_element(By.CSS_SELECTOR, ".spnew2.ico_nav_news").click()
# time.sleep(3)

driver.find_element(By.XPATH, '//a[contains(text(), "뉴스")]').click()

time.sleep(2)
```

```python
**요소 이름으로 선택하기**

문법: //요소이름
예제: //a는 페이지의 모든 링크를 선택합니다

**클래스 속성으로 선택하기**

문법: //요소이름[@class='클래스명']
예제: //div[@class='container']는 class가 'container'인 모든 <div> 요소를 선택합니다.

**ID 속성으로 선택하기**

문법: //*[@id='아이디값']
예제: //*[@id='header']는 id가 'header'인 모든 요소를 선택합니다.

**속성으로 선택하기**

문법: //요소이름[@속성명='속성값']
예제: //input[@type='text']는 type이 'text'인 모든 <input> 요소를 선택합니다.

**텍스트로 선택하기**

문법: //요소이름[text()='텍스트값']
예제: //a[text()='Next']는 텍스트가 'Next'인 모든 링크를 선택합니다.

**조건으로 선택하기**

문법: //요소이름[조건]
예제: //a[@href]는 href 속성을 가진 모든 링크를 선택합니다.

**하위 요소로 선택하기**

문법: //부모요소이름/자식요소이름
예제: //ul/li는 모든 <ul> 요소의 하위에 있는 모든 <li> 요소를 선택합니다.
```

## 예제

```python
driver.find_element(By.XPATH, '//*[text()="뉴스"]').click()

뉴스 라은 텍스트를 가진 요소를 클릭하게 한다

driver.find_element(By.XPATH, '//*[contains(text(), "뉴스")]').click()

contains() 함수는 XPath에서 문자열을 포함하는 요소를 선택하는 데 사용. 두 가지 매개변수를 가지며,

첫 번째 매개변수는 비교할 속성이나 텍스트를 포함하는 XPath 식이고, 두 번째 매개변수는 비교할 문자

driver.find_element(By.XPATH, '//*[@role="tab"]').click()

일부 속성 값이 일치하는 요소를 찾아 클릭 ( '@' 기호는 속성을 지칭 )

```

# 여러번 스크롤하여 더 많은 정보 크롤링하기

```python
from selenium import webdriver
from selenium.webdriver.chrome.options import Options
from selenium.webdriver.common.by import By
from selenium.webdriver.common.keys import Keys
from bs4 import BeautifulSoup
import time

options = Options()

options.add_argument("--start-maximized")
options.add_experimental_option("detach", True)

driver = webdriver.Chrome(options=options)

url = "https://naver.com"

driver.get(url)
time.sleep(2)

driver.find_element(By.ID, "query").send_keys("뉴진스")
time.sleep(2)

driver.find_element(By.CSS_SELECTOR, ".btn_search").click()
time.sleep(3)

driver.find_element(By.CSS_SELECTOR, ".spnew2.ico_nav_news").click()
time.sleep(3)

# driver.find_element(By.XPATH, '//a[contains(text(), "뉴스")]').click()

time.sleep(2)

driver.find_element(By.NAME, "query").clear()
time.sleep(2)

driver.find_element(By.NAME, "query").send_keys("에스파")
time.sleep(2)

driver.find_element(By.NAME, "query").send_keys(Keys.ENTER)
time.sleep(2)

for i in range(10):
    driver.find_element(By.TAG_NAME, "body").send_keys(Keys.END)
    time.sleep(1)

html = driver.page_source

soup = BeautifulSoup(html, "html.parser")

items = soup.select(".news_area")

for rank, item in enumerate(items,1):
    name = item.select_one(".info.press")
    title = item.select_one(".news_tit")

    print(f"<<{rank}>>")
    print(name.text)
    print(title.text)
    print("")

# driver.save_screenshot(r'F:\2024WebProject\Crawling\img\screenshot.png')

```
