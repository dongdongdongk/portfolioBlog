# 웹소켓이란?

```jsx
웹소켓은 웹 애플리케이션에서 실시간으로 데이터를 주고받을 수 있는 기술로, HTTP 프로토콜을 확장하여 양방향 통신을 지원한다.

HTTP 통신은 클라이언트가 서버에 요청을 보내면, 서버가 응답을 보내고 나면 연결이 끊어지는 일반적인 방식을 사용한다.

이는 클라이언트에서 서버로의 단방향 통신으로 제한되어 있다. 그러나 웹소켓은 이와 다르게 클라이언트와 서버 간에 연결을 유지하고 있으며, 한 번 연결을 설정하면 지속적으로 데이터를 주고받을 수 있다. 이를 통해 실시간으로 데이터를 전송하고 받을 수 있다.

웹소켓은 HTTP와 비슷한 프로토콜을 사용하지만, 초기 연결 시에는 HTTP를 통해 웹소켓 연결을 설정한다. 그러나 이후에는 HTTP와는 다른 프로토콜을 사용하여 데이터를 주고받는다. 이를 통해 HTTP의 제약을 극복하고 실시간으로 데이터를 전송할 수 있는 것이다.

예를 들어, 온라인 채팅 애플리케이션을 생각해보자 HTTP를 사용하여 채팅 메시지를 보내고 받을 경우, 메시지가 전송되고 응답이 올 때까지 기다려야 한다.

그러나 웹소켓을 사용하면 채팅 메시지를 보내면 서버가 즉시 그 메시지를 받아 다른 사용자들에게 전달할 수 있다. 이러한 특성은 실시간 채팅, 주식 시장 정보 업데이트, 온라인 게임 등의 애플리케이션에서 매우 유용하다.

따라서 웹소켓은 HTTP보다 훨씬 빠르고 효율적으로 실시간 데이터 통신을 가능하게 한다.
```

import axios from 'axios' // axios를 가져옴
import React, { useRef, useState } from 'react' // React 및 관련 훅들을 가져옴
import { useEffect } from 'react' // useEffect 훅을 가져옴
import { useDispatch, useSelector } from 'react-redux' // useDispatch 및 useSelector를 가져옴
import { useNavigate } from 'react-router-dom' // useNavigate 훅을 가져옴
import { AiOutlineArrowRight, AiOutlineSend } from 'react-icons/ai' // react-icons에서 필요한 아이콘을 가져옴

import { format } from 'timeago.js' // timeago.js의 format 함수를 가져옴

// 소켓 서버의 엔드포인트 및 소켓 ID 설정
const ENDPOINT = 'http://localhost:4000/'
const socketId = socketIO(ENDPOINT, { transports: ['websocket'] })

// 대시보드 메시지 컴포넌트
const DashboardMessages = () => {
  // Redux store에서 판매자 정보 및 대화 목록을 가져옴
  const { seller } = useSelector((state) => state.seller)
  const [conversations, setConversations] = useState([]) // 대화 목록 상태
  const [arrivalMessage, setArrivalMessage] = useState(null) // 도착한 메시지 상태
  const [currentChat, setCurrentChat] = useState() // 현재 대화 상태
  const [messages, setMessages] = useState([]) // 메시지 목록 상태
  const [userData, setUserData] = useState(null) // 사용자 데이터 상태
  const [newMessage, setNewMessage] = useState('') // 새 메시지 상태
  const [onlineUsers, setOnlineUsers] = useState([]) // 온라인 사용자 목록 상태
  const [activeStatus, setActiveStatus] = useState(false) // 활성 상태 상태
  const [open, setOpen] = useState(false) // 열린 상태 상태
  const scrollRef = useRef(null) // 스크롤 참조

  // 소켓으로부터 메시지 받기
  useEffect(() => {
    socketId.on('getMessage', (data) => {
      setArrivalMessage({
        sender: data.senderId,
        text: data.text,
        createdAt: Date.now(),
      })
    })
  }, [])

  // 도착한 메시지를 메시지 목록에 추가
  useEffect(() => {
    arrivalMessage &&
      currentChat?.members.includes(arrivalMessage.sender) &&
      setMessages((prev) => [...prev, arrivalMessage])
  }, [arrivalMessage, currentChat])

  // 대화 목록 가져오기
  useEffect(() => {
    const getConversation = async () => {
      try {
        const response = await axios.get(
          `http://localhost:5000/api/v2/conversation/get-all-conversation-seller/${seller?._id}`,
          {
            withCredentials: true,
          }
        )

        setConversations(response.data.conversations)
      } catch (error) {
        // 오류 처리
      }
    }
    getConversation()
  }, [seller, messages]) // 판매자 및 메시지 변경 시 호출

  // 소켓에 판매자 추가 및 온라인 사용자 가져오기
  useEffect(() => {
    if (seller) {
      const userId = seller?._id
      socketId.emit('addUser', userId)
      socketId.on('getUsers', (data) => {
        setOnlineUsers(data)
      })
    }
  }, [seller]) // 판매자 변경 시 호출

  // 온라인 상태 확인
  const onlineCheck = (chat) => {
    const chatMembers = chat.members.find((member) => member !== seller?._id)
    const online = onlineUsers.find((user) => user.userId === chatMembers)
    return online ? true : false
  }

  // 메시지 가져오기
  useEffect(() => {
    const getMessage = async () => {
      try {
        const response = await axios.get(
          `http://localhost:5000/api/v2/message/get-all-messages/${currentChat?._id}`
        )
        setMessages(response.data.messages)
      } catch (error) {
        // 오류 처리
      }
    }
    getMessage()
  }, [currentChat]) // 현재 대화 변경 시 호출

  // 새 메시지 보내기
  const sendMessageHandler = async (e) => {
    e.preventDefault()

    const message = {
      sender: seller._id,
      text: newMessage,
      conversationId: currentChat._id,
    }
    const receiverId = currentChat.members.find((member) => member.id !== seller._id)

    socketId.emit('sendMessage', {
      senderId: seller._id,
      receiverId,
      text: newMessage,
    })

    try {
      if (newMessage !== '') {
        await axios
          .post(`http://localhost:5000/api/v2/message/create-new-message`, message)
          .then((res) => {
            setMessages([...messages, res.data.message])
            updateLastMessage()
          })
          .catch((error) => {
            // 오류 처리
          })
      }
    } catch (error) {
      // 오류 처리
    }
  }

  // 마지막 메시지 업데이트
  const updateLastMessage = async () => {
    socketId.emit('updateLastMessage', {
      lastMessage: newMessage,
      lastMessageId: seller._id,
    })

    await axios
      .put(`http://localhost:5000/api/v2/conversation/update-last-message/${currentChat._id}`, {
        lastMessage: newMessage,
        lastMessageId: seller._id,
      })
      .then((res) => {
        console.log(res.data.conversation)
        setNewMessage('')
      })
      .catch((error) => {
        // 오류 처리
      })
  }

  // 메시지 스크롤
  useEffect(() => {
    // useEffect 훅을 사용하여 메시지가 업데이트될 때마다 스크롤을 자동으로 아래로 이동시킨다
    scrollRef.current?.scrollIntoView({ beahaviour: 'smooth' })
  }, [messages])

  // UI 렌더링
  return (
    <div className="m-5 h-[85vh] w-[90%] overflow-y-scroll rounded bg-white">
      {!open && (
        <>
          <h1 className="font-Poppins py-3 text-center text-[30px]">All Messages</h1>
          {/* 모든 대화 목록 */}
          {conversations &&
            conversations.map((item, index) => (
              <MessageList
                data={item}
                key={index}
                index={index}
                setOpen={setOpen}
                setCurrentChat={setCurrentChat}
                me={seller._id}
                setUserData={setUserData}
                userData={userData}
                online={onlineCheck(item)}
                setActiveStatus={setActiveStatus}
              />
            ))}
        </>
      )}

      {open && (
        <SellerInbox
          setOpen={setOpen}
          newMessage={newMessage}
          setNewMessage={setNewMessage}
          sendMessageHandler={sendMessageHandler}
          messages={messages}
          sellerId={seller._id}
          userData={userData}
          activeStatus={activeStatus}
          scrollRef={scrollRef}
          setMessages={setMessages}
        />
      )}
    </div>
  )
}

Socket 구성 및 이벤트 처리:

socketIO를 사용하여 소켓을 구성하고, ENDPOINT에 연결한다. WebSocket을 통해 통신한다.

useEffect 훅을 사용하여 컴포넌트가 마운트될 때 한 번만 실행되는 메시지 수신 이벤트를 등록한다.

메시지 수신 처리:

getMessage 이벤트를 수신하면, 받은 데이터를 사용하여 도착한 메시지를 설정한다.

도착한 메시지는 송신자 ID, 메시지 텍스트 및 생성 시간으로 구성된다.

메시지 송신 및 송신 상태 업데이트:

sendMessageHandler 함수는 메시지를 송신하는 역할을 합니다. 이 함수는 폼 제출 이벤트에 의해 호출된다.

송신할 메시지 객체를 생성하고, 현재 선택된 대화의 ID를 포함한다.

송신자와 수신자 ID를 확인하고, 이 정보를 사용하여 소켓을 통해 메시지를 전송한다.

메시지가 성공적으로 전송되면, 서버로부터 반환된 응답을 통해 새로운 메시지를 상태에 추가하고, 마지막 메시지를 업데이트한다.

만약 메시지가 비어있다면 전송을 하지 않는다.

이러한 방식으로, 소켓을 사용하여 실시간으로 메시지를 수신하고 송신할 수 있으며, 메시지 송수신 상태를 적절하게 업데이트할 수 있다.

// 필요한 모듈 및 파일을 가져옵니다.
const Messages = require('../model/messages')
const ErrorHandler = require('../util/ErrorHandler')
const catchAsyncErrors = require('../middleware/catchAsyncErrors')
const express = require('express')
const { upload } = require('../multer')

// Express 라우터를 생성합니다.
const router = express.Router()

// 새로운 메시지를 생성하는 엔드포인트를 정의합니다.
router.post(
  '/create-new-message',
  // 파일 업로드를 위한 미들웨어를 사용합니다.
  upload.array('images'),
  // 비동기 함수로 랩핑된 미들웨어를 사용합니다.
  catchAsyncErrors(async (req, res, next) => {
    try {
      // 요청 데이터를 가져옵니다.
      const messageData = req.body

      // 이미지가 업로드되었는지 확인합니다.
      if (req.files) {
        const files = req.files
        // 업로드된 이미지의 URL을 배열에 저장합니다.
        const imageUrls = files.map((file) => `${file.fileName}`)
        messageData.images = imageUrls
      }

      // 메시지 데이터에서 필요한 정보를 추출합니다.
      messageData.conversationId = req.body.conversationId
      messageData.sender = req.body.sender
      messageData.text = req.body.text

      // Messages 모델을 사용하여 새로운 메시지 객체를 생성합니다.
      const message = new Messages({
        conversationId: messageData.conversationId,
        text: messageData.text,
        sender: messageData.sender,
        images: messageData.images ? messageData.images : undefined,
      })

      // 새로운 메시지를 저장합니다.
      await message.save()

      // 성공 응답을 반환합니다.
      res.status(201).json({
        success: true,
        message,
      })
    } catch (error) {
      // 에러가 발생한 경우 ErrorHandler를 사용하여 에러를 처리합니다.
      return next(new ErrorHandler(error.response.message), 500)
    }
  })
)

const server = http.createServer(app) // express 애플리케이션을 기반으로 서버를 생성합니다.

  // 사용자를 추가하는 함수를 정의합니다.
  !users.some((user) => user.userId === userId) && // 중복된 userId가 없는 경우에만 추가합니다.
    users.push({ userId, socketId }) // 사용자 정보를 배열에 추가합니다.
}

  // 사용자를 제거하는 함수를 정의합니다.
  users = users.filter((user) => user.socketId !== socketId) // socketId에 해당하는 사용자를 제거합니다.
}

const getUser = (receiverId) => {
  // 특정 사용자 정보를 가져오는 함수를 정의합니다.
  return users.find((user) => user.userId === receiverId) // userId에 해당하는 사용자 정보를 반환합니다.

const createMessage = ({ senderId, receiverId, text, images }) => ({
  senderId,
  receiverId,
  text,
  images,
  seen: false, // 기본값으로 "seen" 속성을 false로 설정합니다. seen"은 메시지의 읽음 상태를 나타내는 속성입니다. 일반적으로 메시지를 읽은 경우에는 "seen" 속성이 true로 설정됩니다. 이렇게 함으로써 메시지를 읽은 시점을 추적하고 사용자에게 새로운 메시지가 도착했는지 여부를 알려줄 수 있습니다.
})

  // 클라이언트가 연결되었음을 콘솔에 출력합니다.

          senderId,
          receiverId,
          messageId,
        })
      }

    io.emit('getUsers', users)

같은 방에 접속하게 하는 원리

주어진 코드에서는 Socket.IO를 사용하여 클라이언트가 같은 방에 접속하게 된다. 여기서 "같은 방에 접속하다"는 의미는 서버 측에서 클라이언트들을 동일한 그룹으로 그룹화하는 것을 의미한다. 이렇게 하면 해당 방에 속한 클라이언트끼리 메시지를 주고받을 수 있다.

주요 접속 과정은 다음과 같다:

클라이언트가 앱에 접속하면, 클라이언트는 Socket.IO를 사용하여 서버에 연결한다. 이때 클라이언트는 웹 소켓(WebSocket)을 통해 실시간 통신을 수행할 수 있다.

클라이언트가 서버에 연결되면, 서버에서는 클라이언트의 연결을 확인하고 해당 클라이언트를 적절한 방 또는 그룹에 조인시킨다. 방 또는 그룹은 보통 특정 사용자 그룹 또는 특정 대화 상자(채팅방)와 관련된 클라이언트들로 구성된다.

클라이언트가 특정 방에 조인되면, 해당 방에 속한 다른 클라이언트들과 실시간으로 메시지를 교환할 수 있다. 이를 통해 동일한 방에 속한 클라이언트들끼리 채팅이 가능해진다.

클라이언트가 방에서 나가면, 서버는 해당 클라이언트를 방에서 제거하고 연결을 종료한다.

이 과정에서 클라이언트가 서버에 방에 조인하는 부분은 일반적으로 Socket.IO의 socket.join() 메서드 또는 이와 유사한 메서드를 사용하여 이루어진다. 클라이언트는 서버에 특정 방에 조인하는 요청을 보내고, 서버는 이를 확인하고 해당 클라이언트를 방에 조인시킨다.

주어진 코드에서는 서버 측에서 socketId.emit("addUser", userId)와 같은 방식으로 클라이언트가 방에 조인하는 부분이 이루어진다. 이를 통해 클라이언트는 해당 방에 접속하게 되고, 동일한 방에 속한 다른 클라이언트들과 실시간으로 메시지를 교환할 수 있다.
