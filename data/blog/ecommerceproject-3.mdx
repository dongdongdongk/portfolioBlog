```jsx
// Material-UI 및 React 아이콘을 import
import { Button } from '@material-ui/core'
import { DataGrid } from '@material-ui/data-grid'
import React, { useEffect } from 'react'
import { AiOutlineDelete, AiOutlineEye } from 'react-icons/ai'
import { useDispatch, useSelector } from 'react-redux'
import { Link } from 'react-router-dom'
import { getAllProductsShop, deleteProduct } from '../../redux/product/productAction'
import Loader from '../Layout/Loader'

// 모든 제품을 나열하는 함수형 컴포넌트
const AllProducts = () => {
  // Redux 스토어의 product 및 seller 슬라이스에서 필요한 데이터를 가져오기
  const { products, isLoading } = useSelector((state) => state.product)
  const { seller } = useSelector((state) => state.seller)

  // Redux useDispatch 훅을 사용하여 액션 디스패치를 위한 디스패치 함수 가져오기
  const dispatch = useDispatch()

  // 컴포넌트가 마운트되면 모든 제품 정보를 가져오는 비동기 액션 디스패치
  useEffect(() => {
    dispatch(getAllProductsShop(seller._id))
  }, [dispatch])

  // 제품 삭제 핸들러 함수
  const handleDelete = (id) => {
    dispatch(deleteProduct(id)) // deleteProduct 비동기 액션 디스패치
    window.location.reload() // 페이지 새로고침
  }

  // 테이블의 열 설정
  const columns = [
    { field: 'id', headerName: 'Product Id', minWidth: 150, flex: 0.7 },
    {
      field: 'name',
      headerName: 'Name',
      minWidth: 180,
      flex: 1.4,
    },
    {
      field: 'price',
      headerName: 'Price',
      minWidth: 100,
      flex: 0.6,
    },
    {
      field: 'Stock',
      headerName: 'Stock',
      type: 'number',
      minWidth: 80,
      flex: 0.5,
    },
    {
      field: 'sold',
      headerName: 'Sold out',
      type: 'number',
      minWidth: 130,
      flex: 0.6,
    },
    {
      field: 'Preview',
      flex: 0.8,
      minWidth: 100,
      headerName: '',
      type: 'number',
      sortable: false,
      // 미리보기 버튼 렌더링 함수
      renderCell: (params) => {
        const d = params.row.name
        const product_name = d.replace(/\s+/g, '-')
        return (
          <>
            <Link to={`/product/${product_name}`}>
              <Button>
                <AiOutlineEye size={20} />
              </Button>
            </Link>
          </>
        )
      },
    },
    {
      field: 'Delete',
      flex: 0.8,
      minWidth: 120,
      headerName: '',
      type: 'number',
      sortable: false,
      // 삭제 버튼 렌더링 함수
      renderCell: (params) => {
        return (
          <>
            <Button onClick={() => handleDelete(params.id)}>
              <AiOutlineDelete size={20} />
            </Button>
          </>
        )
      },
    },
  ]

  // 데이터 그리드에 표시할 행 데이터 배열 초기화
  const row = []

  // 제품 목록이 존재하면 각 제품을 데이터 그리드 행 형식으로 변환하여 배열에 추가
  products &&
    products.forEach((item) => {
      row.push({
        id: item._id,
        name: item.name,
        price: 'US$ ' + item.discountPrice,
        Stock: item.stock,
        sold: 10,
      })
    })

  // 로딩 중인지 확인하여 Loader 또는 데이터 그리드 표시
  return (
    <>
      {isLoading ? (
        <Loader /> // 로딩 중일 때 Loader 컴포넌트 표시
      ) : (
        <div className="mx-8 mt-10 w-full bg-white pt-1">
          <DataGrid
            rows={row} // 데이터 그리드에 표시할 행 데이터 배열 전달
            columns={columns} // 데이터 그리드에 표시할 열 설정 전달
            pageSize={10} // 페이지당 표시할 행의 수 설정
            disableSelectionOnClick // 클릭 시 행 선택 비활성화
            autoHeight // 자동으로 높이 설정
          />
        </div>
      )}
    </>
  )
}

export default AllProducts
```

```jsx
import axios from 'axios'
import { createAsyncThunk } from '@reduxjs/toolkit'

// 이벤트 생성
export const createEvent = createAsyncThunk('event/createEvent', async (newForm) => {
  try {
    // HTTP 요청 헤더 설정
    const config = { headers: { 'Content-Type': 'multipart/form-data' } }

    // 서버에 새 이벤트 생성 요청
    const { data } = await axios.post(
      `http://localhost:5000/api/v2/event/create-event`,
      newForm,
      config
    )

    // 성공적으로 생성된 이벤트 데이터 반환
    return data.event
  } catch (error) {
    // 에러가 발생하면 에러 메시지를 던짐
    throw error.response.data.message
  }
})

// 모든 이벤트 가져오기
export const getAllEventsShop = createAsyncThunk('event/getAllEventsShop', async (id) => {
  try {
    // 특정 상점의 모든 이벤트를 서버에서 가져옴
    const { data } = await axios.get(`http://localhost:5000/api/v2/event/get-all-events/${id}`)

    // 성공적으로 가져온 이벤트 목록 반환
    return data.events
  } catch (error) {
    // 에러가 발생하면 에러 메시지를 던짐
    throw error.response.data.message
  }
})

// 이벤트 삭제
export const deleteEvent = createAsyncThunk('event/deleteEvent', async (id) => {
  try {
    // 서버에서 특정 이벤트를 삭제하는 요청

    // 성공적으로 삭제된 메시지 반환

  } catch (error) {
    // 에러가 발생하면 에러 메시지를 던짐
```

## eventReducer.js 추가

```jsx
import { createSlice } from '@reduxjs/toolkit'
import { createEvent, getAllEventsShop, deleteEvent } from './eventAction'

// Redux Slice 생성
const eventSlice = createSlice({
  name: 'event', // 슬라이스의 이름
  initialState: {
    isLoading: true, // 비동기 작업 진행 중 여부
    success: false, // 비동기 작업 성공 여부
    error: null, // 에러 메시지
    event: null, // 생성된 이벤트 데이터
    events: [], // 조회된 이벤트 목록
    message: null, // 삭제된 이벤트에 대한 메시지
  },
  extraReducers: (builder) => {
    builder
      // 이벤트 생성
      .addCase(createEvent.pending, (state) => {
        state.isLoading = true // 요청이 진행 중임을 나타냄
      })
      .addCase(createEvent.fulfilled, (state, action) => {
        state.isLoading = false // 요청이 성공적으로 완료됨을 나타냄
        state.event = action.payload // 생성된 이벤트 데이터 저장
        state.success = true // 성공 플래그 설정
      })
      .addCase(createEvent.rejected, (state, action) => {
        state.isLoading = false // 요청이 실패함을 나타냄
        state.error = action.error.message // 에러 메시지 저장
        state.success = false // 성공 플래그 해제
      })

      // 모든 이벤트 조회
      .addCase(getAllEventsShop.pending, (state) => {
        state.isLoading = true // 요청이 진행 중임을 나타냄
      })
      .addCase(getAllEventsShop.fulfilled, (state, action) => {
        state.isLoading = false // 요청이 성공적으로 완료됨을 나타냄
        state.events = action.payload // 조회된 이벤트 목록 저장
      })
      .addCase(getAllEventsShop.rejected, (state, action) => {
        state.isLoading = false // 요청이 실패함을 나타냄
        state.error = action.payload // 에러 메시지 저장
      })

      // 이벤트 삭제
      .addCase(deleteEvent.pending, (state) => {
        state.isLoading = true // 요청이 진행 중임을 나타냄
      })
      .addCase(deleteEvent.fulfilled, (state, action) => {
        state.isLoading = false // 요청이 성공적으로 완료됨을 나타냄
        state.message = action.payload // 삭제된 이벤트에 대한 메시지 저장
      })
      .addCase(deleteEvent.rejected, (state, action) => {
        state.isLoading = false // 요청이 실패함을 나타냄
        state.error = action.payload // 에러 메시지 저장
      })
  },
})

export default eventSlice.reducer
```

## event.js 컨트롤러

```jsx
const express = require('express')
const catchAsyncErrors = require('../middleware/catchAsyncErrors')
const { upload } = require('../multer')
const Shop = require('../model/shop')
const Event = require('../model/event')
const ErrorHandler = require('../util/ErrorHandler')
const { isSeller } = require('../middleware/auth')
const router = express.Router()
const fs = require('fs')

// create event
router.post(
  '/create-event',
  upload.array('images'),
  catchAsyncErrors(async (req, res, next) => {
    try {
      // 요청에서 받은 데이터 확인
      const shopId = req.body.shopId
      const shop = await Shop.findById(shopId)

      // 상점이 존재하지 않으면 에러 반환
      if (!shop) {
        return next(new ErrorHandler('상점Id가 존재하지 않습니다!', 400))
      } else {
        // 업로드된 이미지 파일 정보 확인 및 URL 생성
        const files = req.files
        const imageUrls = files.map((file) => `${file.filename}`)

        // 이벤트 데이터 구성
        const eventData = req.body
        eventData.images = imageUrls
        eventData.shop = shop

        // 이벤트 생성 및 응답
        const event = await Event.create(eventData)
        res.status(201).json({
          success: true,
          event,
        })
      }
    } catch (error) {
      // 에러 발생 시 에러 핸들링
      return next(new ErrorHandler(error, 400))
    }
  })
)

// get all events
router.get('/get-all-events', async (req, res, next) => {
  try {
    // 모든 이벤트 조회
    const events = await Event.find()
    res.status(201).json({
      success: true,
      events,
    })
  } catch (error) {
    // 에러 발생 시 에러 핸들링
    return next(new ErrorHandler(error, 400))
  }
})

// get all events of a shop
router.get(
  '/get-all-events/:id',
  catchAsyncErrors(async (req, res, next) => {
    try {
      // 특정 상점의 모든 이벤트 조회
      const events = await Event.find({ shopId: req.params.id })

      res.status(201).json({
        success: true,
        events,
      })
    } catch (error) {
      // 에러 발생 시 에러 핸들링
      return next(new ErrorHandler(error, 400))
    }
  })
)

// delete event of a shop
router.delete(
  '/delete-shop-event/:id',
  isSeller,
  catchAsyncErrors(async (req, res, next) => {
    try {
      // 삭제할 이벤트의 ID 확인
      const productId = req.params.id

      // 이벤트 데이터 조회
      const eventData = await Event.findById(productId)

      // 이벤트에 속한 이미지 파일 삭제
      eventData.images.forEach((imageUrl) => {
        const filename = imageUrl
        const filePath = `uploads/${filename}`

        fs.unlink(filePath, (err) => {
          if (err) {
            console.log(err)
          }
        })
      })

      // 이벤트 삭제 및 응답
      const event = await Event.findByIdAndDelete(productId)

      if (!event) {
        return next(new ErrorHandler('이 id에서 이벤트를 찾지 못하였습니다.', 500))
      }

      res.status(201).json({
        success: true,
        message: '이벤트 삭제 성공!',
      })
    } catch (error) {
      // 에러 발생 시 에러 핸들링
      return next(new ErrorHandler(error, 400))
    }
  })
)

module.exports = router
```

```jsx
startDate.getTime()은 시작 날짜를 밀리초 단위의 숫자로 반환한다.

3 * 24 * 60 * 60 * 1000은 3일을 밀리초로 변환한 값이다.

endDateRef.current.min = minEndDate.toISOString().slice(0, 10);

endDateRef.current를 통해 endDateRef의 현재 값, 즉 실제 DOM 요소에 접근한다.
```

```jsx
      // 새로운 쿠폰 코드 생성 및 저장
      const couponCode = await CouponCode.create(req.body)

      // 성공 응답 반환
      res.status(201).json({
        success: true,
        couponCode,
      })
    } catch (error) {
      // 예외 발생 시 에러 핸들링
      return next(new ErrorHandler(error, 400))
    }
  })
)

// 상점의 모든 쿠폰 코드 조회
router.get(
  '/get-coupon/:id',
  isSeller,
  catchAsyncErrors(async (req, res, next) => {
    try {
      // 상점의 모든 쿠폰 코드 조회
      const couponCodes = await CouponCode.find({ shopId: req.seller.id })
      res.status(201).json({
        success: true,
        couponCodes,
      })
    } catch (error) {
      // 예외 발생 시 에러 핸들링
      return next(new ErrorHandler(error, 400))
    }
  })
)

// 상점의 쿠폰 코드 삭제
router.delete(
  '/delete-coupon/:id',
  isSeller,
  catchAsyncErrors(async (req, res, next) => {
    try {
      // 쿠폰 코드 삭제
      const couponCode = await CouponCode.findByIdAndDelete(req.params.id)

      // 삭제된 쿠폰 코드가 존재하지 않으면 에러 반환
      if (!couponCode) {
        return next(new ErrorHandler('쿠폰 코드가 존재하지 않습니다!', 400))
      }
      // 성공 응답 반환
      res.status(201).json({
        success: true,
        message: '쿠폰 코드 삭제 성공!',
      })
    } catch (error) {
      // 예외 발생 시 에러 핸들링
      return next(new ErrorHandler(error, 400))
    }
  })
)

module.exports = router
```

```jsx
import { Button } from '@material-ui/core'
import { DataGrid } from '@material-ui/data-grid'
import axios from 'axios'
import React, { useEffect, useState } from 'react'
import { AiOutlineDelete } from 'react-icons/ai'
import { RxCross1 } from 'react-icons/rx'
import { useDispatch, useSelector } from 'react-redux'
import styles from '../../styles/styles'
import Loader from '../Layout/Loader'
import { toast } from 'react-toastify'

// 모든 쿠폰 정보를 나열하고 관리하는 컴포넌트
const AllCoupons = () => {
  // state 정의
  const [open, setOpen] = useState(false)
  const [name, setName] = useState('')
  const [isLoading, setIsLoading] = useState(false)
  const [coupons, setcoupons] = useState([])
  const [minAmount, setMinAmout] = useState(null)
  const [maxAmount, setMaxAmount] = useState(null)
  const [selectedProducts, setSelectedProducts] = useState(null)
  const [value, setValue] = useState(null)
  const { seller } = useSelector((state) => state.seller)
  const { products } = useSelector((state) => state.product)

  const dispatch = useDispatch()

  // 모든 쿠폰 정보를 서버에서 가져오는 useEffect
  useEffect(() => {
    setIsLoading(true)
    axios
      .get(`http://localhost:5000/api/v2/coupon/get-coupon/${seller._id}`, {
        withCredentials: true,
      })
      .then((res) => {
        setIsLoading(false)
        setcoupons(res.data.couponCodes)
      })
      .catch((error) => {
        setIsLoading(false)
      })
  }, [dispatch])

  // 쿠폰 삭제 핸들러
  const handleDelete = async (id) => {
    axios
      .delete(`http://localhost:5000/api/v2/coupon/delete-coupon/${id}`, {
        withCredentials: true,
      })
      .then((res) => {
        toast.success('Coupon code deleted succesfully!')
      })
    window.location.reload()
  }

  // 쿠폰 생성 핸들러
  const handleSubmit = async (e) => {
    e.preventDefault()

    await axios
      .post(
        `http://localhost:5000/api/v2/coupon/create-coupon-code`,
        {
          name,
          minAmount,
          maxAmount,
          selectedProducts,
          value,
          shopId: seller._id,
        },
        { withCredentials: true }
      )
      .then((res) => {
        toast.success('Coupon code created successfully!')
        setOpen(false)
        window.location.reload()
      })
      .catch((error) => {
        toast.error(error.response.data.message)
      })
  }

  // 표의 열 정보(columns) 정의
  const columns = [
    { field: 'id', headerName: 'Id', minWidth: 150, flex: 0.7 },
    {
      field: 'name',
      headerName: 'Coupon Code',
      minWidth: 180,
      flex: 1.4,
    },
    {
      field: 'price',
      headerName: 'Value',
      minWidth: 100,
      flex: 0.6,
    },
    {
      field: 'Delete',
      flex: 0.8,
      minWidth: 120,
      headerName: '',
      type: 'number',
      sortable: false,
      renderCell: (params) => {
        return (
          <>
            <Button onClick={() => handleDelete(params.id)}>
              <AiOutlineDelete size={20} />
            </Button>
          </>
        )
      },
    },
  ]

  // 표의 행 정보(rows) 정의
  const row = []

  coupons &&
    coupons.forEach((item) => {
      row.push({
        id: item._id,
        name: item.name,
        price: item.value + ' %',
        sold: 10,
      })
    })

  // 렌더링
  return (
    <>
      {isLoading ? (
        <Loader />
      ) : (
        <div className="mx-8 mt-10 w-full bg-white pt-1">
          <div className="flex w-full justify-end">
            {/* 쿠폰 코드 생성 버튼 */}
            <div
              className={`${styles.button} mr-3 mb-3 !h-[45px] !w-max !rounded-[5px] px-3`}
              onClick={() => setOpen(true)}
            >
              <span className="text-white">쿠폰 코드 만들기</span>
            </div>
          </div>

          {/* 쿠폰 정보를 나타내는 테이블 */}
          <DataGrid rows={row} columns={columns} pageSize={10} disableSelectionOnClick autoHeight />

          {/* 쿠폰 코드 생성 모달 */}
          {open && (
            <div className="fixed top-0 left-0 z-[20000] flex h-screen w-full items-center justify-center bg-[#00000062]">
              <div className="800px:w-[40%] h-[80vh] w-[90%] rounded-md bg-white p-4 shadow">
                <div className="flex w-full justify-end">
                  {/* 모달 닫기 버튼 */}
                  <RxCross1 size={30} className="cursor-pointer" onClick={() => setOpen(false)} />
                </div>

                {/* 쿠폰 코드 생성 폼 */}
                <h5 className="font-Poppins text-center text-[30px]">쿠폰 코드 만들기</h5>
                <form onSubmit={handleSubmit} aria-required={true}>
                  <br />
                  <div>
                    {/* 쿠폰 코드명 입력 */}
                    <label className="pb-2">
                      쿠폰코드명 <span className="text-red-500">*</span>
                    </label>
                    <input
                      type="text"
                      name="name"
                      required
                      value={name}
                      className="mt-2 block h-[35px] w-full appearance-none rounded-[3px] border border-gray-300 px-3 placeholder-gray-400 focus:border-yellow-500 focus:ring-yellow-500 focus:outline-none sm:text-sm"
                      onChange={(e) => setName(e.target.value)}
                      placeholder="쿠폰 코드 이름 입력"
                    />
                  </div>
                  <br />
                  <div>
                    {/* 할인율 입력 */}
                    <label className="pb-2">
                      할인율 <span className="text-red-500">*</span>
                    </label>
                    <input
                      type="text"
                      name="value"
                      value={value}
                      required
                      className="mt-2 block h-[35px] w-full appearance-none rounded-[3px] border border-gray-300 px-3 placeholder-gray-400 focus:border-yellow-500 focus:ring-yellow-500 focus:outline-none sm:text-sm"
                      onChange={(e) => setValue(e.target.value)}
                      placeholder="쿠폰 코드 할인율 입력"
                    />
                  </div>
                  <br />
                  <div>
                    {/* 최소금액 입력 */}
                    <label className="pb-2">최소금액</label>
                    <input
                      type="number"
                      name="value"
                      value={minAmount}
                      className="mt-2 block h-[35px] w-full appearance-none rounded-[3px] border border-gray-300 px-3 placeholder-gray-400 focus:border-yellow-500 focus:ring-yellow-500 focus:outline-none sm:text-sm"
                      onChange={(e) => setMinAmout(e.target.value)}
                      placeholder="쿠폰 코드 최소 금액 입력"
                    />
                  </div>
                  <br />
                  <div>
                    {/* 최대금액 입력 */}
                    <label className="pb-2">최대금액</label>
                    <input
                      type="number"
                      name="value"
                      value={maxAmount}
                      className="mt-2 block h-[35px] w-full appearance-none rounded-[3px] border border-gray-300 px-3 placeholder-gray-400 focus:border-yellow-500 focus:ring-yellow-500 focus:outline-none sm:text-sm"
                      onChange={(e) => setMaxAmount(e.target.value)}
                      placeholder="쿠폰 코드 최대 금액 입력"
                    />
                  </div>
                  <br />
                  <div>
                    {/* 제품 선택 */}
                    <label className="pb-2">제품선택</label>
                    <select
                      className="mt-2 h-[35px] w-full rounded-[5px] border"
                      value={selectedProducts}
                      onChange={(e) => setSelectedProducts(e.target.value)}
                    >
                      <option value="Choose your selected products">제품을 선택해주세요</option>
                      {products &&
                        products.map((i) => (
                          <option value={i.name} key={i.name}>
                            {i.name}
                          </option>
                        ))}
                    </select>
                  </div>
                  <br />
                  <div>
                    {/* 쿠폰 코드 생성 버튼 */}
                    <input
                      type="submit"
                      value="Create"
                      className="mt-2 block h-[35px] w-full appearance-none rounded-[3px] border border-gray-300 px-3 placeholder-gray-400 focus:border-yellow-500 focus:ring-yellow-500 focus:outline-none sm:text-sm"
                    />
                  </div>
                </form>
              </div>
            </div>
          )}
        </div>
      )}
    </>
  )
}

export default AllCoupons
```
