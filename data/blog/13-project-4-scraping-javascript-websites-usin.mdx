  }
end

## 스파이더 코드 작성하기

```python
import scrapy
from scrapy_splash import SplashRequest

# 스파이더 클래스 정의
class AdamchoiSpider(scrapy.Spider):
    name = "adamchoi"  # 스파이더의 이름을 지정합니다.
    allowed_domains = ["www.adamchoi.co.uk", "localhost"]  # 크롤링이 허용된 도메인 목록입니다.

    # LUA 스크립트 정의
    script = '''
        function main(splash, args)
            splash.private_mode_enabled = false  # Splash의 프라이빗 모드를 비활성화합니다.
            assert(splash:go(args.url))  # 주어진 URL로 이동합니다.
            assert(splash:wait(3))  # 페이지가 로드될 때까지 3초 동안 대기합니다.
            all_maches = assert(splash:select_all("label.btn.btn-sm.btn-primary"))  # 특정 CSS 셀렉터에 해당하는 모든 요소를 선택합니다.
            all_maches[2]:mouse_click()  # 선택된 요소 중 세 번째 요소를 클릭합니다.
            assert(splash:wait(3))  # 클릭 후 페이지가 다시 로드될 때까지 3초 동안 대기합니다.
            splash:set_viewport_full()  # 전체 페이지를 볼 수 있도록 뷰포트를 설정합니다.
            return {
                splash:html(),  # 현재 페이지의 HTML을 반환합니다.
            }
        end
    '''

    # 스파이더가 시작할 때 호출되는 메서드입니다.
    def start_requests(self):
        yield SplashRequest(
            url='https://www.adamchoi.co.uk/overs/detailed',  # 크롤링을 시작할 URL입니다.
            callback=self.parse,  # 요청이 완료된 후 호출될 콜백 함수입니다.
            endpoint='execute',  # Splash에서 사용할 엔드포인트입니다. 이 경우 'execute'는 LUA 스크립트를 실행합니다.
            args={'lua_source': self.script},  # 실행할 LUA 스크립트를 전달합니다.
        )

    # 응답을 처리하는 메서드입니다.
    def parse(self, response):
        print(response.body)  # 응답 본문을 출력합니다.
```

```python
    # 스파이더가 시작할 때 호출되는 메서드입니다.
    def start_requests(self):
        yield SplashRequest(
            url='https://www.adamchoi.co.uk/overs/detailed',  # 크롤링을 시작할 URL입니다.
            callback=self.parse,  # 요청이 완료된 후 호출될 콜백 함수입니다.
            endpoint='execute',  # Splash에서 사용할 엔드포인트입니다. 이 경우 'execute'는 LUA 스크립트를 실행합니다.
            args={'lua_source': self.script},  # 실행할 LUA 스크립트를 전달합니다.
        )


    * endpoint='execute'는 Splash 서버가 사용할 엔드포인트를 지정한다.

	    Splash 서버는 여러 엔드포인트를 제공하며, 그 중 execute는 LUA 스크립트를 실행하기 위한 엔드포인트이다.

	    render.html: 기본적으로 페이지의 HTML을 렌더링하여 반환
			render.png: 페이지의 스크린샷을 반환
			render.json: 페이지의 HTML, 스크린샷, HAR 파일 등의 정보를 JSON 형식으로 반환
			execute: LUA 스크립트를 실행하여 사용자 지정 동작을 수행
			( execute 엔드포인트를 사용하면, 우리가 작성한 LUA 스크립트를 통해 복잡한 상호작용을 수행할 수 있다.
				예를 들어, 버튼 클릭, 폼 입력, 페이지 스크롤 등을 할 수 있다 )


	 * args={'lua_source': self.script}는 SplashRequest에 전달될 인자(argument)들을 지정한다.

		 lua_source 키는 실행할 LUA 스크립트의 내용을 포함하고 있다.


		 이 부분은 SplashRequest에게 "이런 내용을 가지고 요청을 보내라"고 지시하는 역할을 한다.

		 args는 여러 가지 설정을 포함하는 "가방"이고, 그 안에 lua_source라는 "키"가 들어있다.

		 lua_source는 "이 스크립트를 실행해라"는 명령을 가지고 있다.

			args: Splash에게 보내는 설정들이 들어있는 "가방"

			lua_source: 실행할 LUA 스크립트를 포함하는 "명령"
```

### Scrapy와 Splash의 통신:

```python
1, Scrapy는 웹 스크래핑을 수행하는 라이브러리이고, Splash는 자바스크립트를 렌더링하는 웹 브라우저이다.

2. Scrapy는 Splash를 통해 자바스크립트를 포함한 웹 페이지를 요청하고 렌더링된 결과를 받아온다.

3. Splash는 로컬 서버에서 실행되며, 일반적으로 http://localhost:8050 또는 http://127.0.0.1:8050 주소를 사용한다.
```

### 오프사이트 요청 필터링:

```python
Scrapy는 기본적으로 allowed_domains 목록에 포함되지 않은 도메인으로의 요청을 차단한다.

이는 웹 크롤러가 예상치 못한 웹사이트로 이동하지 않도록 하기 위함이다.

만약 allowed_domains에 localhost가 포함되지 않으면, Scrapy는 localhost로의 요청을 "오프사이트 요청"으로 간주하고 차단한다.
```

```python
import scrapy
from scrapy_splash import SplashRequest

class AdamchoiSpider(scrapy.Spider):
    name = "adamchoi"  # 이 스파이더의 이름을 정의합니다.
    allowed_domains = ["www.adamchoi.co.uk", "localhost"]  # 스파이더가 크롤링할 수 있는 도메인을 정의합니다.

    # start_urls = ["https://www.adamchoi.co.uk"]

    # Lua 스크립트를 정의합니다. 이 스크립트는 Splash에서 실행되어 페이지를 렌더링하고 특정 동작을 수행합니다.
    script = '''
        function main(splash, args)
            splash.private_mode_enabled = false  # Splash의 프라이빗 모드를 비활성화합니다.
            assert(splash:go(args.url))  # 지정된 URL로 이동합니다.
            assert(splash:wait(3))  # 페이지가 로드될 때까지 3초 기다립니다.
            all_maches = assert(splash:select_all("label.btn.btn-sm.btn-primary"))  # 모든 일치하는 요소를 선택합니다.
            all_maches[2]:mouse_click()  # 세 번째 요소를 클릭합니다.
            assert(splash:wait(3))  # 클릭 후 3초 기다립니다.
            splash:set_viewport_full()  # 전체 페이지를 캡처하기 위해 뷰포트를 설정합니다.
            return splash:html()  # 페이지의 HTML을 반환합니다.
        end
    '''

    def start_requests(self):
        # SplashRequest를 생성하여 시작 URL과 함께 Lua 스크립트를 실행합니다.
        yield SplashRequest(
            url='https://www.adamchoi.co.uk/overs/detailed',
            callback=self.parse,  # 요청이 완료되면 parse 메서드를 콜백으로 호출합니다.
            endpoint='execute',  # Splash에서 사용할 엔드포인트입니다. 'execute'는 Lua 스크립트를 실행합니다.
            args={'lua_source': self.script},  # Lua 스크립트와 함께 전달할 인자들입니다.
        )

    def parse(self, response):
        # XPath를 사용하여 HTML에서 모든 <tr> 요소를 선택합니다.
        rows = response.xpath('//tr')

        # 각 행(row)을 순회하며 데이터를 추출합니다.
        for row in rows:
            date = row.xpath('./td[1]/text()').get()  # 첫 번째 <td> 요소에서 날짜를 추출합니다.
            home_team = row.xpath('./td[2]/text()').get()  # 두 번째 <td> 요소에서 홈 팀 이름을 추출합니다.
            score = row.xpath('./td[3]/text()').get()  # 세 번째 <td> 요소에서 점수를 추출합니다.
            away_team = row.xpath('./td[4]/text()').get()  # 네 번째 <td> 요소에서 어웨이 팀 이름을 추출합니다.

            # 추출된 데이터를 딕셔너리 형태로 생성하고 yield를 사용하여 반환합니다.
            yield {
                'date': date,
                'home_team': home_team,
                'score': score,
                'away_team': away_team,
            }
```
