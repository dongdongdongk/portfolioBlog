```sql
-- DataBase 선택

use db;

-- 크게 오라클과 다를바 없지만 약간의 차이가 있다.
-- Oracle 영어 한글자 1byte, 한글 1글자 2~3
-- Mysql 영어및 한글 1글자는 크기가 같다

CREATE TABLE TEST ( -- 대 소문자를 구분
	NUM BIGINT AUTO_INCREMENT PRIMARY KEY, -- AUTO_INCREMENT 시퀸서랑 비슷한 기능
	NAME VARCHAR(100), -- VARCHAR2 가 VARCHAR 임
	INFO LONGTEXT,  -- 오라클의 CLOB 같은 큰 글자를 담을 때
	BIRTH DATE
)

SELECT *
FROM db.TEST;  -- use db 를 사용했기 때문에 db. 생략가능

INSERT INTO TEST (NUM, NAME, INFO, BIRTH)
VALUES(NULL, 'IU', 'singer', now())  -- now 는 지금 시간  NULL 을 넣어서 시퀸스 사용

INSERT INTO TEST
VALUES (0, 'WINTER', 'Group', now()) -- 자동증가 옵션을 주고싶으면 0 이나 null 을 주기

INSERT INTO TEST (NAME, INFO, BIRTH )
VALUES('CHOA', 'TEST', now() )      -- 아예 생략도 가능하다  (총 3가지 null,0,제거)
```

## DataType

### 1. Number(숫자)

```jsx
1. BIT(N)  : 1-8(byte)
2. INT     : -21억 ~ +21억
3. BIGINT  : Long과 같다
4. FLOAT   : 소숫점 7자리
5. DOUBLE  : 소숫점 15자리
6. DECIMAL(전체자릿수, 소숫점자리수)
```

### 2. 문자열

```jsx
1. CHAR(1-255)  : 글자수, 고정길이 문자열
2. VARHCAR(?)   : 가변길이 문자열
3. TEXT         : 65535
4. MEDIUMTEXT   : 1677715
4. LONGTEXT     : 4GB
```

### 3. 날짜

```jsx
1. DATE : YYYY-MM-DD
2. TIME : HH:MM:SS
3. DATETIME : YYYY-MM-DD HH:MM:SS
4. TIMESTAMP : DB의 TIME_ZONE 시스템 변수와 관련이 있음,
```

### 4. 기타

```jsx
1. JSON    :  JSON문서 저장, MYSQL 5.7.8이상 부터 지원
2. GEOMETRY : 선, 점 , 도형 같은 공간 데이터 객체를 저장
```

## Limit

```jsx
SELECT 때 일정한 갯수 만큼 조회할 때 사용
Oracle의 ROWNUM 대신 사용

LIMIT 시작인덱스번호 , 조회할 갯수 (몇번부터 , 몇개까지 가져올거냐)

SELECT * FROM QNA
ORDER BY NUM DESC
LIMIT 0 , 10

SELECT * FROM QNA
ORDER BY NUM DESC
LIMIT 10 , 10

SELECT * FROM QNA
ORDER BY NUM DESC
LIMIT 20 , 10
```

  ```java
  **sqlsession 사용**

  package com.example.dao;

  import java.util.List;

  import org.apache.ibatis.session.SqlSession;
  import org.springframework.beans.factory.annotation.Autowired;
  import org.springframework.stereotype.Repository;

  import com.example.dto.NoticeDTO;

  @Repository
  public class NoticeDAOImpl implements NoticeDAO {

      @Autowired
      private SqlSession sqlSession;

      private static final String NAMESPACE = "com.example.mapper.NoticeMapper";

      @Override
      public List<NoticeDTO> getAllNotices() {
          return sqlSession.selectList(NAMESPACE + ".selectAllNotices");
      }

      @Override
      public NoticeDTO getNoticeById(int id) {
          return sqlSession.selectOne(NAMESPACE + ".selectNoticeById", id);
      }

      @Override
      public void addNotice(NoticeDTO notice) {
          sqlSession.insert(NAMESPACE + ".insertNotice", notice);
      }

      @Override
      public void updateNotice(NoticeDTO notice) {
          sqlSession.update(NAMESPACE + ".updateNotice", notice);
      }

      @Override
      public void deleteNotice(int id) {
          sqlSession.delete(NAMESPACE + ".deleteNotice", id);
      }

  }
  ```
