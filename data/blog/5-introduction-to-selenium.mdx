# 테이블 추출하기 ( Extracting Data from a Table )

```python
from selenium import webdriver
from selenium.webdriver.chrome.options import Options
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

# Chrome 옵션 설정 (브라우저를 닫지 않도록 설정)
options = Options()
options.add_experimental_option("detach", True)

# 스크랩할 웹사이트 URL
website = 'https://www.adamchoi.co.uk/overs/detailed'

# Chrome WebDriver 객체 생성
driver = webdriver.Chrome(options=options)

# 웹사이트 열기
driver.get(website)

# 'All matches' 버튼을 XPATH를 통해 찾아 클릭 (모든 경기 데이터를 로드)
all_matches_button = driver.find_element(By.XPATH, '//label[@analytics-event="All matches"]')
all_matches_button.click()

# 'tr' 요소가 나타날 때까지 최대 10초 대기 (경기 데이터 로드 완료를 기다림)
WebDriverWait(driver, 10).until(EC.presence_of_element_located((By.TAG_NAME, 'tr')))

# 모든 'tr' 요소를 찾아 리스트로 저장 (경기 데이터의 각 행)
matches = driver.find_elements(By.TAG_NAME, 'tr')

# 각 경기 데이터 행을 순회
for match in matches:
    # 첫 번째 'td' 요소를 찾아 날짜를 저장 (각 경기의 날짜)
    date = match.find_element(By.XPATH, './td[1]')
    # 두 번째 'td' 요소를 찾아 홈팀을 저장 (각 경기의 홈팀)
    home_team = match.find_element(By.XPATH, './td[2]')
    # 세 번째 'td' 요소를 찾아 점수를 저장 (각 경기의 점수)
    score = match.find_element(By.XPATH, './td[3]')
    # 네 번째 'td' 요소를 찾아 상대팀을 저장 (각 경기의 상대팀)
    away_team = match.find_element(By.XPATH, './td[4]')

    # 각 경기의 데이터를 출력
    print("날짜", date.text)
    print("홈팀", home_team.text)
    print("점수", score.text)
    print("상대팀", away_team.text)
    print(" ")
    print(" ")

# 주의: 코드 실행 후 driver.quit()을 통해 드라이버를 종료할 수 있음
# driver.quit()

```

```python
1. 모든 'tr' 요소를 찾아 리스트로 저장:

matches = driver.find_elements(By.TAG_NAME, 'tr'): 'tr' 태그를 가진 모든 요소를 찾아 리스트에 저장. 각 'tr'은 경기 데이터를 나타냄.

2. 각 경기 데이터를 순회하며 정보 추출 및 출력:

각 'tr' 요소(경기 데이터 행)마다 내부의 'td' 요소를 찾아 날짜, 홈팀, 점수, 상대팀 정보를 추출.

print 문을 통해 각 경기의 정보를 출력.

* xpath 는 인덱스 0 부터가 아니라 1 부터다
```

# 요소 기다리게 하기 ( WebDriverWait, expected_conditions)

## WebDriverWait

```python
WebDriverWait는 주어진 조건이 충족될 때까지 대기하는 기능을 제공한다.

대기 시간은 최대 대기 시간과 간격을 설정할 수 있으며, 설정한 조건이 충족되면 즉시 다음 단계로 넘어간다.
```

```python
from selenium.webdriver.support.ui import WebDriverWait

* 임포트 해줘야한다
```

## expected_conditions

```python
expected_conditions (또는 EC)는 Selenium에서 자주 사용되는 여러 가지 조건을 미리 정의해 놓은 모듈이다.

이 조건들은 웹 페이지의 요소가 특정 상태가 될 때까지 대기하는 데 사용된다.
```

### expected_conditions에는 다양한 조건 소개

```python
presence_of_element_located(locator): 지정된 요소가 DOM에 존재하는지 확인

visibility_of_element_located(locator): 지정된 요소가 DOM에 존재하고, 표시되는지 확인

element_to_be_clickable(locator): 지정된 요소가 표시되고, 클릭할 수 있는 상태인지 확인

text_to_be_present_in_element(locator, text_): 지정된 요소에 특정 텍스트가 포함되어 있는지 확인
```

## 예제

```python
# 'tr' 요소가 나타날 때까지 최대 10초 대기
WebDriverWait(driver, 10).until(EC.presence_of_element_located((By.TAG_NAME, 'tr')))

1. WebDriverWait 객체 생성: WebDriverWait(driver, 10)은 최대 10초 동안 기다리겠다는 의미

2. until 메서드 사용: until 메서드는 특정 조건이 충족될 때까지 대기한다.

	 여기서는 EC.presence_of_element_located((By.TAG_NAME, 'tr')) 조건이 충족될 때까지 대기한다.

3. 조건 충족 시 다음 단계 실행: tr 요소가 DOM에 존재하게 되면, WebDriverWait은 대기를 멈추고

		다음 코드(모든 tr 요소를 찾고 텍스트를 출력하는 코드)를 실행한다.
```

### 예제 1: 요소가 클릭 가능할 때까지 대기하기

```python
from selenium import webdriver
from selenium.webdriver.chrome.options import Options
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

options = Options()
options.add_experimental_option("detach", True)

driver = webdriver.Chrome(options=options)
driver.get('https://example.com')  # 실제 테스트할 웹사이트 URL

# 특정 버튼이 클릭 가능할 때까지 최대 10초 대기
button = WebDriverWait(driver, 10).until(EC.element_to_be_clickable((By.ID, 'myButton')))

# 버튼 클릭
button.click()

driver.quit()
```

### 예제 2: 텍스트가 특정 요소에 나타날 때까지 대기하기

```python
from selenium import webdriver

from selenium.webdriver.common.by import By
```

### 예제 4: 여러 요소가 나타날 때까지 대기하기

```python
from selenium import webdriver

from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC

options = Options()
options.add_experimental_option("detach", True)

driver = webdriver.Chrome(options=options)
driver.get('https://example.com')  # 실제 테스트할 웹사이트 URL

# 여러 요소가 나타날 때까지 최대 10초 대기
elements = WebDriverWait(driver, 10).until(EC.presence_of_all_elements_located((By.CLASS_NAME, 'myClass')))

```

```jsx
# 드롭다운 메뉴를 찾기 위해 먼저 요소를 검색합니다. 여기서는 드롭다운 메뉴의 ID가 'country'인 요소를 찾습니다.
dropdown = Select(driver.find_element(By.ID, 'country'))

# 'Spain'이라는 텍스트가 표시된 옵션을 드롭다운 메뉴에서 선택합니다.
dropdown.select_by_visible_text('Spain')
```

```jsx
1. 드롭다운 메뉴 요소 찾기:

먼저, 드롭다운 메뉴 요소를 찾는다.

find_element 메소드를 사용하여 드롭다운 메뉴의 HTML 요소를 찾아야 한다.

이 예제에서는 드롭다운 메뉴의 id 속성을 사용하여 요소를 찾는다.

2. Select 객체 생성:

드롭다운 메뉴 요소를 찾은 후, 이 요소를 Select 클래스의 인스턴스로 만든다.

이를 통해 드롭다운 메뉴와 상호작용할 수 있다.

3. 옵션 선택:

드롭다운 메뉴에서 특정 옵션을 선택하기 위해 select_by_visible_text 메소드를 사용한다.

이 메소드는 옵션의 표시된 텍스트를 기준으로 옵션을 선택한다.
```
