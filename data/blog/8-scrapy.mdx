```python
import scrapy

class AudibleSpider(scrapy.Spider):
    # 스파이더 이름을 정의합니다.
    name = "audible"
    # 스파이더가 크롤링할 수 있는 도메인을 정의합니다.
    allowed_domains = ["www.audible.com"]
    # 크롤링을 시작할 URL을 정의합니다.
    start_urls = ["https://www.audible.com/search"]

    # 스파이더가 페이지를 크롤링할 때 호출되는 함수입니다.
    def parse(self, response):
        # Audible 웹페이지에서 제품 컨테이너를 찾기 위한 XPath 쿼리입니다.
        product_container = response.xpath('//div[@class="adbl-impression-container "]//li[contains(@class, "productListItem")]')

        # 각 제품 컨테이너에 대해 반복합니다.
        for product in product_container:
            # 제품 컨테이너 내에서 책 제목을 찾기 위한 XPath 쿼리입니다.
            book_title = product.xpath('.//h3[contains(@class, "bc-heading")]//a/text()').get()
            # 제품 컨테이너 내에서 저자 이름을 찾기 위한 XPath 쿼리입니다.
            book_author = product.xpath('.//li[contains(@class, "authorLabel")]//span//a/text()').getall()
            # 제품 컨테이너 내에서 책 길이를 찾기 위한 XPath 쿼리입니다.
            book_length = product.xpath('.//li[contains(@class, "runtimeLabel")]//span/text()').get()

            # 수집한 데이터를 딕셔너리 형태로 반환합니다.
            yield {
                'title' : book_title,  # 책 제목
                'author' : book_author,  # 저자 이름
                'length' : book_length,  # 책 길이
            }
```

# 페이지 네이션 ( Pagination )

```python
import scrapy

class AudibleSpider(scrapy.Spider):
    name = "audible"  # 스파이더의 이름 정의. 실행 시 이 이름으로 스파이더를 호출할 수 있음
    allowed_domains = ["www.audible.com"]  # 크롤링을 허용할 도메인 목록
    start_urls = ["https://www.audible.com/search"]  # 스파이더가 시작할 URL 목록

    def parse(self, response):
        # 제품 목록을 포함하는 컨테이너를 XPath로 선택
        product_container = response.xpath('//div[@class="adbl-impression-container "]//li[contains(@class, "productListItem")]')

        # 각 제품에 대해 필요한 정보를 추출
        for product in product_container:
            book_title = product.xpath('.//h3[contains(@class, "bc-heading")]//a/text()').get()  # 책 제목 추출
            book_author = product.xpath('.//li[contains(@class, "authorLabel")]//span//a/text()').getall()  # 저자 정보 추출
            book_length = product.xpath('.//li[contains(@class, "runtimeLabel")]//span/text()').get()  # 책 길이(시간) 정보 추출

            # 추출한 정보를 딕셔너리 형태로 반환
            yield {
                'title': book_title,
                'author': book_author,
                'length': book_length,
            }

        # 페이지네이션 요소를 선택
        pagination = response.xpath('//ul[contains(@class, "pagingElements")]')
        # 다음 페이지의 URL을 추출
        next_page_url = pagination.xpath('.//span[contains(@class, "nextButton")]//a//@href').get()
        # 다음 페이지 버튼의 비활성화 여부를 확인
        button_disabled = pagination.xpath('.//span[contains(@class , "nextButton")]/a/@aria-disabled').get()

        # 다음 페이지 URL이 있고, 버튼이 비활성화되어 있지 않으면 다음 페이지로 이동
        if next_page_url and button_disabled == None:
            yield response.follow(url=next_page_url, callback=self.parse)
```

```python
아래 부분을 조금더 상세하게 설명하자면

        if next_page_url and button_disabled == None:
            yield response.follow(url=next_page_url, callback=self.parse)


if next_page_url and button_disabled == None: 다음 페이지가 존재하고 버튼이 활성화되어 있으면,

다음 페이지로 이동한다. response.follow를 사용하여 새로운 요청을 보내고 parse 메서드를 다시 호출한다.

url=next_page_url: 다음 페이지로 이동하기 위한 URL

callback=self.parse: 새로운 페이지가 로드된 후 호출할 콜백 함수이다. 이 경우, self.parse가 콜백 함수로 지정되어있다.
```

### 콜백 함수 (callback=self.parse)

```python
callback: 특정 작업이 완료된 후 호출될 함수를 지정한다.

여기서 callback은 새로운 페이지가 로드된 후 실행될 함수를 의미한다.

self.parse: 현재 클래스의 parse 메서드를 콜백 함수로 지정한다.

즉, 다음 페이지가 로드되면 parse 메서드가 다시 호출되어 데이터를 추출하는 작업을 반복하게 된다.
```

## 코드에서 User-Agent 변경

```python
import scrapy

class AudibleSpider(scrapy.Spider):
    name = "audible"  # 스파이더의 이름 정의
    allowed_domains = ["www.audible.com"]  # 크롤링을 허용할 도메인 목록
    start_urls = ["https://www.audible.com/search"]  # 시작할 URL 목록

    def start_requests(self):
        # 첫 요청을 생성하여 User-Agent 헤더를 설정
        yield scrapy.Request(
            url="https://www.audible.com/search",
            callback=self.parse,
            headers={
                "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/125.0.0.0 Safari/537.36"
            },
        )

    def parse(self, response):
        # 제품 컨테이너를 선택하는 XPath
        product_container = response.xpath(
            '//div[@class="adbl-impression-container "]//li[contains(@class, "productListItem")]'
        )

        # 각 제품에 대해 데이터 추출
        for product in product_container:
            book_title = product.xpath(
                './/h3[contains(@class, "bc-heading")]//a/text()'
            ).get()  # 책 제목을 추출
            book_author = product.xpath(
                './/li[contains(@class, "authorLabel")]//span//a/text()'
            ).getall()  # 책 저자를 추출 (여러 저자일 경우 모두 추출)
            book_length = product.xpath(
                './/li[contains(@class, "runtimeLabel")]//span/text()'
            ).get()  # 책 길이를 추출

            # 추출한 데이터를 딕셔너리 형태로 yield
            yield {
                "title": book_title,
                "author": book_author,
                "length": book_length,
                "User-Agent": response.request.headers["User-Agent"],  # 요청의 User-Agent 헤더 값 포함
            }

        # 페이지네이션을 위한 XPath 선택자
        pagination = response.xpath('//ul[contains(@class, "pagingElements")]')
        next_page_url = pagination.xpath(
            './/span[contains(@class, "nextButton")]//a//@href'
        ).get()  # 다음 페이지 URL 추출
        button_disabled = pagination.xpath(
            './/span[contains(@class , "nextButton")]/a/@aria-disabled'
        ).get()  # 다음 페이지 버튼의 비활성화 상태 확인

        # 다음 페이지 URL이 존재하고, 버튼이 비활성화되지 않은 경우 다음 페이지로 이동
        if next_page_url and button_disabled == None:
            yield response.follow(
                url=next_page_url,
                callback=self.parse,
                headers={
                    "User-Agent": response.request.headers["User-Agent"].decode("utf-8")
                },  # User-Agent 헤더를 설정하여 다음 페이지 요청 생성
            )

```
