# { SCSS 변수 및 기본 스타일 }

```jsx
/* ============== Google Fonts ==================== */
// Google Fonts에서 'Rubik' 폰트를 가져옵니다. 웹페이지에서 이 폰트를 사용할 수 있게 해줍니다.
@import url('https://fonts.googleapis.com/css2?family=Rubik:ital,wght@0,300..900;1,300..900&display=swap');

// ==================== Variables SCSS =================
// SCSS의 변수 선언입니다. 여기서는 색상, 폰트, 그림자, 박스의 모서리 둥글기 등의 스타일을 변수로 설정하여
// 여러 곳에서 재사용할 수 있게 만들었습니다.

$first-color: hsl(353, 100%, 65%);      // 첫 번째 주요 색상 (주로 버튼 등에 사용)
$title-color: hsl(244, 24%, 26%);       // 제목에 사용할 색상
$text-color: hsl(244, 16%, 43%);        // 본문 텍스트에 사용할 색상
$body-color: hsl(258, 60%, 98%);        // 배경 색상
$container-color: #fff;                 // 컨테이너의 배경 색상 (흰색)

// ==== Font and Typography =======
// 폰트와 타이포그래피와 관련된 변수입니다. 폰트 크기와 폰트 종류를 선언하여 여러 곳에서 쉽게 재사용할 수 있습니다.
$body-font: 'Rubik', sans-serif;        // 기본 폰트 설정
$h1-font-size: 2.25rem;                 // h1(제목) 태그에 적용할 폰트 크기
$h2-font-size: 1.5rem;                  // h2 태그에 적용할 폰트 크기
$h3-font-size: 1.25rem;                 // h3 태그에 적용할 폰트 크기
$normal-font-size: 1rem;                // 일반 본문 텍스트의 폰트 크기
$small-font-size: 0.875rem;             // 작은 텍스트에 사용할 폰트 크기
$smaller-font-size: 0.813rem;           // 더 작은 텍스트에 사용할 폰트 크기

// ==== Font weight =======
// 폰트의 두께를 설정하는 변수입니다. 숫자가 클수록 폰트가 굵어집니다.
$font-medium: 500;                      // 중간 굵기 폰트
$font-semibold: 600;                    // 약간 굵은 폰트
$font-bold: 700;                        // 굵은 폰트

// ==== Box Shadow =======
// 박스 그림자 효과를 설정하는 변수입니다.
$shadow: 0px 5px 20px 0px rgb(69 67 96 / 10%);   // 박스 그림자의 크기와 색상 투명도 설정

// ==== Box Radius =======
// 박스의 모서리 둥글기를 설정하는 변수입니다.
$border-radius: 20px;                   // 박스의 모서리를 20px 만큼 둥글게 설정

// Responsive typography
// 반응형 폰트 크기를 설정하기 위한 Mixin입니다. 화면 크기가 1024px 이하로 작아질 때,
// 폰트 크기를 조정하여 가독성을 유지합니다.
@mixin responsive-font-sizes {
    @media screen and (max-width: 1024px) {
        $h1-font-size: 1.75rem;         // 작은 화면에서는 h1 크기를 줄임
        $h2-font-size: 1.25rem;         // h2 폰트 크기 줄임
        $h3-font-size: 1rem;            // h3 폰트 크기 줄임
        $normal-font-size: 0.938rem;    // 일반 텍스트 크기 줄임
        $small-font-size: 0.813rem;     // 작은 텍스트 크기 줄임
        $smaller-font-size: 0.75rem;    // 더 작은 텍스트 크기 줄임
    }
}

// ==== Base =======
// 기본 설정입니다. 모든 요소에 대해 여백(margin)과 패딩(padding)을 0으로 설정하고,
// box-sizing을 설정하여 박스의 크기를 예측 가능하게 만듭니다.
* {
    margin: 0;
    padding: 0;
    box-sizing: border-box;
}

html {
    scroll-behavior: smooth;   // 페이지를 부드럽게 스크롤되게 설정
}

body,
button,
input,
textarea {
    font-family: $body-font;     // 폰트는 모두 $body-font('Rubik')로 설정
    font-size: $normal-font-size; // 기본 폰트 크기는 $normal-font-size로 설정
}

body {
    background-color: $body-color;  // 배경 색상 설정
    line-height: 1.7;               // 텍스트 줄 간격 설정
    color: $text-color;             // 기본 텍스트 색상 설정
}

h1,
h2,
h3 {
    color: $title-color;            // 제목의 색상 설정
    font-weight: $font-semibold;    // 제목 폰트의 굵기를 설정
    line-height: 1.2;               // 제목 텍스트의 줄 간격 설정
}

ul {
    list-style: none;  // 리스트 항목의 기본 스타일을 없앰 (숫자나 점 대신 깔끔한 리스트)
}

a {
    text-decoration: none;  // 링크의 기본 밑줄을 제거
}

button {
    cursor: pointer;  // 마우스를 올리면 커서가 손가락 모양으로 바뀜
    border: none;     // 기본 버튼 테두리 제거
    outline: none;    // 포커스 시 버튼 테두리가 생기지 않도록 설정
}

img {
    max-width: 100%;  // 이미지가 최대 100% 너비를 차지하도록 설정
    height: auto;     // 이미지의 비율을 유지하며 자동으로 높이 설정
    vertical-align: middle;  // 이미지의 수직 정렬을 중앙으로 맞춤
}

// ======= Reusable SCSS Classes =========
// 재사용 가능한 클래스입니다.

.container {
    max-width: 1080px;         // 컨테이너의 최대 너비 설정
    padding-left: 15px;        // 왼쪽 패딩 설정
    padding-right: 15px;       // 오른쪽 패딩 설정
    margin: 0 auto;            // 가운데 정렬
}

.grid {
    display: grid;             // 그리드 레이아웃 사용
}

.section {
    padding-top: 7rem;         // 위쪽 패딩 7rem
    padding-bottom: 2rem;      // 아래쪽 패딩 2rem
}

.section__title {
    font-size: $h1-font-size;  // 섹션 제목의 폰트 크기 설정
    margin-left: 0.875rem;     // 왼쪽 여백 설정
    font-weight: $font-bold;   // 섹션 제목의 굵기를 굵게 설정
    position: relative;        // 위치를 상대적으로 설정하여 ::before에 효과를 적용
    margin-bottom: 3.75rem;    // 섹션 제목의 아래쪽 여백 설정

    &::before {  // 제목 앞에 도트 무늬 배경 이미지를 추가
        content: '';  // 빈 내용을 추가
        background: url(./assets/dots-bg.svg);  // 이미지 경로
        height: 2.25rem;    // 높이 설정
        width: 2.25rem;     // 너비 설정
        position: absolute; // 절대 위치 설정
        left: -0.875rem;    // 왼쪽 위치 설정
        top: -0.875rem;     // 위쪽 위치 설정
    }
}

.btn {
    padding: 0.75rem 2rem;      // 버튼의 안쪽 여백 설정
    line-height: 1;             // 줄 높이 설정
    border-radius: 1.875rem;    // 버튼의 둥근 모서리 설정
    box-shadow: 0 0 1px rgb(0 0 0 / 0%);  // 그림자 효과 설정
    border: 1px solid transparent;  // 투명한 테두리 설정
    color: #fff;                 // 버튼 텍스트 색상 설정
    display: inline-block;       // 버튼을 인라인 블록으로 표시
    background-color: $first-color;  // 버튼 배경 색상 설정
    font-weight: $font-bold;     // 버튼 텍스트의 굵기를 굵게 설정

    &:hover {  // 버튼에 마우스를 올렸을 때 애니메이션 효과 적용
        animation: button-push 0.3s linear 1;  // 버튼이 눌리는 듯한 애니메이션
    }
}

@keyframes button-push {  // 버튼이 눌리는 듯한 효과를 위한 애니메이션 정의
    50% {
        transform: scale(0.8);  // 50% 지점에서 크기가 80%로 줄어듬
    }
    100% {
        transform: scale(1);    // 다시 원래 크기로 돌아옴
    }
}

// Mixin 적용
@include responsive-font-sizes();  // 반응형 폰트 크기를 적용

```

```
@import '../../styles/variables.scss'; // SCSS 변수 파일을 가져옵니다. 이를 통해 스타일을 쉽게 재사용할 수 있습니다.

.aside {
    position: fixed; // 화면에서 고정 위치(스크롤해도 움직이지 않음).
    left: 0; // 화면의 왼쪽 끝에 배치.
    top: 0; // 화면의 위쪽 끝에 배치.
    background: $body-color; // SCSS 변수에서 가져온 배경색을 사용.
    border-right: 1px solid rgba(0, 0, 0, 0.05); // 오른쪽에 얇은 경계선을 추가.
    padding: 2.5rem; // 컨텐츠와 경계선 사이에 여유 공간(패딩) 추가.
    width: 110px; // 고정된 너비 설정.
    min-height: 100vh; // 높이를 화면 전체 높이로 설정.
    display: flex; // 내부 요소를 플렉스 박스로 정렬.
    flex-direction: column; // 요소들을 세로 방향으로 정렬.
    justify-content: space-between; // 내부 요소들을 위아래 끝으로 배치.
    z-index: 10; // 다른 요소들보다 앞에 보이도록 z축 우선순위 설정.
}

.nav__list {
    display: flex; // 플렉스 박스를 사용.
    flex-direction: column; // 리스트 항목들을 세로로 정렬.
    row-gap: 1rem; // 항목들 사이의 간격을 1rem으로 설정.
}

.nav__link {
    font-size: 1.5rem; // 링크 텍스트 크기를 설정.
    color: $title-color; // 제목 색상 변수 사용.
    font-weight: $font-bold; // 굵은 글꼴을 변수로 설정.
}

.copyright {
    color: hsl(245, 15%, 65%); // 연한 회색 계열 색상 설정.
    font-size: $small-font-size; // 작은 글자 크기 사용.
    transform: rotate(-180deg); // 글자를 180도 회전(거꾸로 배치).
    writing-mode: vertical-rl; // 글자를 세로 쓰기로 설정.
}

```
