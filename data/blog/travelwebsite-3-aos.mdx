## Home 애니메이션 추가

```jsx
import React, { useEffect } from 'react'
import './Home.scss'
import Aos from 'aos' // AOS (Animate On Scroll) 라이브러리 가져오기, 스크롤할 때 애니메이션을 적용해줌
import 'aos/dist/aos.css' // AOS 애니메이션에 필요한 기본 CSS 파일 가져오기

export const Home = () => {
  // 컴포넌트가 처음 렌더링될 때 AOS 애니메이션을 초기화하는 함수
  useEffect(() => {
    // AOS.init() 함수를 사용해서 애니메이션 초기화, duration은 애니메이션 지속 시간을 밀리초로 설정 (2000ms = 2초)
    Aos.init({ duration: 2000 })
  }, []) // [] 빈 배열은 컴포넌트가 처음 마운트될 때만 이 코드를 실행함을 의미

  return (
    <section className="home">
      {' '}
      {/* 전체 섹션을 감싸는 컨테이너 */}
      <div className="secContainer container">
        {' '}
        {/* 섹션 내의 콘텐츠를 감싸는 내부 컨테이너 */}
        <div className="homeText">
          <h1 data-aos="fade-up" className="title">
            {' '}
            {/* 제목에 "fade-up" 애니메이션을 적용. 스크롤할 때 아래에서 위로 나타남 */}
            Plan Your Trip With Travel Dot
          </h1>

          <p data-aos="fade-up" data-aos-duration="2500" className="subTitle">
            {' '}
            {/* 2.5초 동안 나타나는 부제목에 "fade-up" 애니메이션 적용 */}
            Travel to your favorite city with respectful of the environment!
          </p>

          <button data-aos="fade-up" data-aos-duration="3000" className="btn">
            {' '}
            {/* 3초 동안 애니메이션이 실행되는 버튼 */}
            <a href="#">Explore Now</a>
          </button>
        </div>
        <div className="homeCard grid">
          {' '}
          {/* 카드 형태로 구성된 입력 폼 */}
          {/* 입력 폼 각각에 "fade-right" 애니메이션을 적용. 스크롤할 때 오른쪽에서 왼쪽으로 나타남 */}
          <div data-aos="fade-right" data-aos-duration="2000" className="locationDiv">
            <label htmlFor="location">Location</label>
            <input type="text" placeholder="DreamDestination" />
          </div>
          <div data-aos="fade-right" data-aos-duration="2500" className="distDiv">
            <label htmlFor="distance">Distance</label>
            <input type="text" placeholder="11/Meters" />
          </div>
          <div data-aos="fade-right" data-aos-duration="3000" className="priceDiv">
            <label htmlFor="price">Price</label>
            <input type="text" placeholder="$140-$550" />
          </div>
          <button data-aos="fade-left" data-aos-duration="2000" className="btn">
            {' '}
            {/* 2초 동안 왼쪽에서 나타나는 버튼 */}
            Search
          </button>
        </div>
      </div>
    </section>
  )
}
```

```jsx
AOS 초기화:

		useEffect 안에서 Aos.init({ duration: 2000 })으로 초기화되며,

		컴포넌트가 처음 렌더링될 때 2초 동안의 애니메이션을 모든 요소에 적용합니다.

애니메이션 속성 (data-aos):

		각 요소에 data-aos="fade-up" 또는 data-aos="fade-right"와 같은 속성을 주어 애니메이션 효과를 적용합니다.

				fade-up: 요소가 아래에서 위로 천천히 나타남.

				fade-right: 요소가 오른쪽에서 왼쪽으로 나타남.

data-aos-duration:

		애니메이션의 지속 시간을 설정하는 속성입니다.

		예를 들어 data-aos-duration="3000"은 애니메이션이 3초 동안 지속됨을 의미합니다.
```

    return (
        <section className="popular section container">

                    <div data-aos="fade-left" data-aos-duration="2500" className="iconsDiv flex">
                        <BsArrowLeftShort className='icon' />
                        <BsArrowRightShort className='icon' />
                    </div>
                </div>

                {/* 메인 콘텐츠 그리드 */}

                    {
                        // 여행지 데이터를 각각 매핑하여 출력

                            return (
                                // fade-up 애니메이션이 적용된 각각의 여행지 카드
                                <div data-aos="fade-up" className="singleDestination">

                                            <h6>{location}</h6>

                                                    <BsDot className="icon" />
                                                </span>
                                            </span>
                                        </div>
                                    </div>
                                </div>
                            )
                        })
                    }
                </div>
            </div>
        </section>
    )
}
