## 3.Controller 제작

```java
@GetMapping("/download")
    public void excelDownload(HttpServletResponse response,UserVO userVO) throws Exception {     // response 는 기본이고 UserVO 로 데이터를 받아와서 넣어준다 , @RequestMapping("/user/*") 때문에 user 은 생략 (form 의 /user/download)
        List<UserVO> userVOS = userService.getUserList(userVO);
        //        Workbook wb = new HSSFWorkbook();       //xls 로 출력할 때는 이걸 사용해준다
        Workbook wb = new XSSFWorkbook();                 //xlsx 파일을 사용할 때 는 이걸 사용
        Sheet sheet = wb.createSheet("첫번째 시트");
        Row row = null;
        Cell cell = null;
        int rowNum = 0;

        // Header
        row = sheet.createRow(rowNum++);

				cell = row.createCell(0);                         // 이렇게 한 세트로 행을 만들어준다 (value 에 행 이름)
        cell.setCellValue("번호");

				cell = row.createCell(1);
        cell.setCellValue("이름");
        cell = row.createCell(2);
        cell.setCellValue("아이디");
        cell = row.createCell(3);
        cell.setCellValue("이메일");
        cell = row.createCell(4);
        cell.setCellValue("권한");
        cell = row.createCell(5);
        cell.setCellValue("전화번호");

        // Body
        for (int i=0; i<userVOS.size(); i++) {          // 반복문 userVOS.size(); 로 List 크기만큼 반복
            UserVO user = userVOS.get(i);
            row = sheet.createRow(rowNum++);

						cell = row.createCell(0);                   // 이렇게 한 세트로 열의 데이터 출력
            cell.setCellValue(i);

						cell = row.createCell(1);
            cell.setCellValue(user.getName());
            cell = row.createCell(2);
            cell.setCellValue(user.getId());
            cell = row.createCell(3);
            cell.setCellValue(user.getEmail());
            cell = row.createCell(4);
            cell.setCellValue(user.getRoleName());
            cell = row.createCell(5);
            cell.setCellValue(user.getPhoneNumber());
        }

        // 컨텐츠 타입과 파일명 지정
        response.setContentType("ms-vnd/excel");
      //  response.setHeader("Content-Disposition", "attachment;filename=example.xls");
        response.setHeader("Content-Disposition", "attachment;filename=example.xlsx");          // filename=example.xlsx 이 부분의 example.xlsx 이게 이름이다

        // Excel File Output
        wb.write(response.getOutputStream());
        wb.close();
    }
```
