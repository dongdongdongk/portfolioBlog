	}

```java

@GetMapping("add")
	public ModelAndView setBoardAdd() throws Exception {
		ModelAndView mv = new ModelAndView();
		mv.setViewName("board/add");
		return mv;
	}


	@PostMapping("add")
	//메소드명을 명시하는 @PostMapping이 나옴
	public ModelAndView setBoardAdd(QnaDTO qnaDTO, MultipartFile [] files, HttpSession session)throws Exception{
		ModelAndView mv = new ModelAndView();
		int result = qnaService.setBoardAdd(qnaDTO, files, session);   //Service 메서드를 호출하고 그 결과를 result 에 저장

		String message = "등록 실패";                     // 초기값으로 message에 등록실패 대입

		if(result>0) {
			message = "글이 등록 되었습니다";		           // 0 실패 1성공이므로  int result 값이 0보다 클시 message 에 등록완료 대입
		}
		mv.addObject("result", message);                 // result 에 message 값을 대입한다
		mv.addObject("url","list");                      // url 에 ./list 값을 대입한다
		mv.setViewName("common/result");                 // common 에 result 로 이동
		return mv;
	}
```
