# 네이버 쇼핑 크롤링 1

```python
from bs4 import BeautifulSoup
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.common.keys import Keys
from selenium.webdriver.chrome.options import Options
from selenium.webdriver import ActionChains
from selenium.common.exceptions import NoSuchElementException
import time
import requests

options = Options()

user_agent = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36"

options.add_argument("--start-maximized")

options.add_experimental_option("detach", True)

options.add_argument(f"user-agent={user_agent}")

options.add_experimental_option("excludeSwitches", ["enable-automation"]) # 자동중 메세지 없에기

driver = webdriver.Chrome(options=options)

keyword = input("검색할 제품을 입력하세요 : ")

url = f"https://search.shopping.naver.com/search/all?query={keyword}"

driver.get(url)
time.sleep(3)

driver.find_element(By.TAG_NAME, "body").send_keys(Keys.END)
time.sleep(1)

html = driver.page_source

soup = BeautifulSoup(html, "html.parser")

base_divs = soup.select("[class^=product_item]") # 원래 클래스 이름은 .product_item__MDtDF 이지만 뒤에 __ 이후 부터는 알아보기 힘들고 변경되는 경향이 있다 이때문에 product_item 으로 시작하는 클래스를 가져오란것

for num, base_div in enumerate(base_divs,1):
    title = base_div.select_one("[class^=product_title]")
    price = base_div.select_one("[class^=product_price]")
    detail_box_div = base_div.select_one("[class^=product_detail_box]")

    print(num)
    print(title.text)
    print(price.text)
    print(detail_box_div.text)
    print("")

print(len(base_divs))

```

```python
base_divs = soup.select("[class^=product_item]") # 원래 클래스 이름은 .product_item__MDtDF 이지만 뒤에 __ 이후 부터는 알아보기 힘들고 변경되는 경향이 있다 이때문에 product_item 으로 시작하는 클래스를 가져오란것

    title = base_div.select_one("[class^=product_title]")

    print(title.text)
    print(price.text)
    print(detail_box_div.text)

print(len(base_divs))
```

```python
여기서 ^=는 "시작하는 것"을 의미한다.

따라서 [class^=product_detail_box]는 클래스 이름이 "product_detail_box"로 시작하는 모든 요소를 선택한다.
```

이때는 .get_text(" ") 를 사용하면 내부의 텍스트를 전부 가져올 때 보기좋게 가져올 수 있다

## api 에서 사용하기 좋게 제품 정보를 하나로 합쳐주기 (리팩토링)

```python
from bs4 import BeautifulSoup
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.common.keys import Keys
from selenium.webdriver.chrome.options import Options
from selenium.common.exceptions import NoSuchElementException
import time
import requests

# Selenium 옵션 설정
options = Options()
user_agent = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36"
options.add_argument("--start-maximized") # 브라우저를 최대화
options.add_experimental_option("detach", True) # 브라우저를 자동으로 분리
options.add_argument(f"user-agent={user_agent}") # 사용자 에이전트 설정
options.add_experimental_option("excludeSwitches", ["enable-automation"]) # 자동 중앙 메시지를 제거

# 웹 드라이버 초기화
driver = webdriver.Chrome(options=options)

# 사용자로부터 검색할 제품 키워드 입력
keyword = input("검색할 제품을 입력하세요 : ")

# 네이버 쇼핑 검색 페이지 URL 생성
url = f"https://search.shopping.naver.com/search/all?query={keyword}"

# 드라이버를 이용하여 해당 URL 열기
driver.get(url)

# 페이지 로딩 대기를 위한 시간 설정
time.sleep(3)

# 페이지를 스크롤하여 더 많은 제품 로딩
for i in range(3):
    driver.find_element(By.TAG_NAME, "body").send_keys(Keys.END)
    time.sleep(1)

# 현재 페이지 소스를 가져옴
html = driver.page_source

# BeautifulSoup을 사용하여 HTML 파싱
soup = BeautifulSoup(html, "html.parser")

# 제품 요소들을 선택 (class가 'product_item'으로 시작하는 요소들)
base_divs = soup.select("[class^=product_item]")

# 제품 정보를 저장할 리스트 생성
main_text_list = []

# 각 제품 요소에 대해 반복
for num, base_div in enumerate(base_divs,1):
    # 제품 제목 요소 선택
    title = base_div.select_one("[class^=product_title]")
    # 제품 가격 요소 선택
    price = base_div.select_one("[class^=product_price]")
    # 제품 상세 정보 요소 선택
    detail_box_div = base_div.select_one("[class^=product_detail_box]")
    # 제품이 광고인 경우 무시하기 위한 광고 버튼 요소 선택
    ad_button = base_div.select_one("[class^=ad_ad_stk]")

    # 만약 광고인 경우 건너뛰기
    if ad_button:
        continue

    # 제품 제목 텍스트 추출
    title_text = title.text
    # 제품 가격 텍스트 추출
    price_text = price.get_text(" ")

    # 제품 제목과 가격을 합쳐서 문자열로 만듦
    prod_text = f"{title_text} | {price_text}"

    # 만약 제품 상세 정보가 있는 경우, 상세 정보 텍스트를 추출하여 제품 정보에 추가
    if detail_box_div:
        detail_box_text = detail_box_div.get_text(" ")
        prod_text += f"\n{detail_box_text}"

    # 제품 정보 출력
    print(prod_text)

    # 제품 정보를 리스트에 추가
    main_text_list.append(prod_text)
```
