```jsx
import React from 'react'

import Card from '../../shared/components/UIElements/Card'

import './Auth.css'

const Auth = () => {
  // useForm 훅을 사용하여 폼 상태 및 입력 핸들러 설정

    {
      email: {
        value: '',

      },
      password: {
        value: '',
        isValid: false,
      },
    },

  )

  // 폼 제출 핸들러
  const authSubmitHandler = (event) => {
    event.preventDefault()
    console.log(formState.inputs)
  }

  return (
    // 인증 컴포넌트를 감싼 Card 컴포넌트
    <Card className="authentication">
      {/* 제목 */}
      <h1>Login Required</h1>
      {/* 수평선 */}
      <hr />
      {/* 폼 요소들을 감싼 form 태그 */}
      <form onSubmit={authSubmitHandler}>
        {/* 이메일 입력 필드 */}
        <Input
          element="input" // 입력 요소 타입
          id="email" // 식별자
          type="email" // 입력 타입
          label="E-Mail" // 레이블 텍스트

          errorText="Please enter a valid email address." // 오류 메시지
          onInput={inputHandler} // 입력 이벤트 핸들러
        />
        {/* 비밀번호 입력 필드 */}
        <Input
          element="input" // 입력 요소 타입
          id="password" // 식별자
          type="password" // 입력 타입
          label="Password" // 레이블 텍스트
          validators={[VALIDATOR_MINLENGTH(5)]} // 유효성 검사 규칙 배열
          errorText="Please enter a valid password at least 5 Ch." // 오류 메시지
          onInput={inputHandler} // 입력 이벤트 핸들러
        />
        {/* 로그인 버튼 */}
        <Button type="submit" disabled={!formState.isValid}>
          LOGIN
        </Button>
      </form>
    </Card>
  )
}

export default Auth
```

```jsx
5. 로그인 버튼:

	Button 컴포넌트를 사용하여 폼 제출 버튼을 생성하고, 폼이 유효하지 않은 경우 비활성화되도록 설정한다.
```

## 전체적인 흐름 ( input 과 form-hook.js 와 참고할것)

```jsx
Input 컴포넌트에 값 입력:

사용자가 이메일 또는 비밀번호 입력 필드에 값을 입력하면 해당 입력 필드의 onChange 이벤트가 발생한다.

Input 컴포넌트의 changeHandler 함수 호출:

changeHandler 함수는 사용자의 입력에 대응하여 현재 입력 필드의 값을, 유효성 검사 규칙 및 type: 'CHANGE' 액션을 dispatch 한다.

dispatch 함수를 사용하여 inputReducer에 액션을 전달하면서 입력 상태를 업데이트합니다.

inputReducer의 동작:

inputReducer는 type: 'CHANGE' 액션이 전달되면, 현재 입력 필드의 값을 새로운 값으로 설정하고 유효성을 검사하여 isValid를 갱신합니다.

이후, 새로운 입력 상태를 반환합니다.

useEffect 훅의 호출:

useEffect 훅이 호출되어 외부에 등록된 onInput 콜백 함수가 실행됩니다.

useEffect는 입력 값, 유효성, 터치 여부가 변경될 때마다 실행되므로, 여기서는 새로 업데이트된 입력 상태를 onInput으로 전달합니다.

inputHandler 함수 호출:

onInput이 inputHandler 함수와 연결되어 type: 'INPUT_CHANGE' 액션을 dispatch 합니다.

이때, inputHandler 함수는 useForm에서 반환된 dispatch 함수로 폼 상태를 업데이트합니다.

formReducer의 동작:

formReducer는 type: 'INPUT_CHANGE' 액션이 전달되면,

현재 입력 필드에 대한 유효성을 확인하고 모든 입력 필드에 대한 유효성을 종합하여 폼의 전체 유효성을 갱신합니다.

최종적으로, 새로운 폼 상태를 반환합니다.

Button의 활성화/비활성화:

formState.isValid 값을 통해 Button 컴포넌트가 활성화 또는 비활성화 상태로 렌더링됩니다.

이는 모든 입력 필드가 유효한지 여부에 따라 결정됩니다.

폼 제출 (Submit):

사용자가 폼을 제출하면 authSubmitHandler 함수가 호출됩니다.

이 함수에서는 현재 폼 상태의 입력 값을 콘솔에 출력하게 됩니다.

이렇게 각 구성 요소 간의 데이터 흐름과 상태 업데이트가 이루어지며, 폼이 사용자의 입력에 동적으로 반응하여 업데이트되는 과정입니다.
```

```jsx
import React, { useState } from 'react'

import Card from '../../shared/components/UIElements/Card'
import Input from '../../shared/components/FormElements/Input'
import Button from '../../shared/components/FormElements/Button'
import { useForm } from '../../shared/hooks/form-hook'
import {
  VALIDATOR_EMAIL,
  VALIDATOR_MINLENGTH,
  VALIDATOR_REQUIRE,
} from '../../shared/util/validators'

import './Auth.css'

const Auth = () => {
  // 로그인 및 회원가입 모드를 관리하는 상태
  const [isLoginMode, setIsLoginMode] = useState(true)

  // useForm 커스텀 훅을 사용하여 폼 상태 및 관련 함수를 초기화
  const [formState, inputHandler, setFormData] = useForm(
    {
      // 초기 입력 상태: 이메일 및 비밀번호
      email: {
        value: '',
        isValid: false,
      },
      password: {
        value: '',
        isValid: false,
      },
    },
    false
  )

  // 모드 전환 핸들러
  const switchModeHandler = () => {
    // 현재 모드가 LOGIN 모드인 경우
    if (!isLoginMode) {
      // 모든 입력 필드를 복사하고 'name' 필드를 초기화하여 양식 데이터 업데이트
      setFormData(
        {
          ...formState.inputs,
          name: undefined,
        },
        // 이메일 및 비밀번호가 유효한 경우에만 폼을 유효하게 설정
        formState.inputs.email.isValid && formState.inputs.password.isValid
      )
    } else {
      // 현재 모드가 SIGNUP 모드인 경우
      // 모든 입력 필드를 복사하고 'name' 필드를 초기화하여 양식 데이터 업데이트
      setFormData(
        {
          ...formState.inputs,
          name: {
            value: '',
            isValid: false,
          },
        },
        // 폼을 무조건 유효하지 않게 설정
        false
      )
    }

    // 모드를 반전시켜 업데이트
    setIsLoginMode((prevMode) => !prevMode)
  }

  // 폼 제출 핸들러
  const authSubmitHandler = (event) => {
    event.preventDefault()
    console.log(formState.inputs)
  }

  return (
    <Card className="authentication">
      <h1>Login Required</h1>
      <hr />
      <form onSubmit={authSubmitHandler}>
        {/* SIGNUP 모드인 경우에만 이름 입력 필드를 렌더링 */}
        {!isLoginMode && (
          <Input
            element="input"
            id="name"
            type="text"
            label="Name"
            validators={[VALIDATOR_REQUIRE()]}
            errorText="Please enter a name"
            onInput={inputHandler}
          />
        )}

        {/* 이메일 입력 필드 */}
        <Input
          element="input"
          id="email"
          type="email"
          label="E-Mail"
          validators={[VALIDATOR_EMAIL()]}
          errorText="Please enter a valid email address."
          onInput={inputHandler}
        />
        {/* 비밀번호 입력 필드 */}
        <Input
          element="input"
          id="password"
          type="password"
          label="Password"
          validators={[VALIDATOR_MINLENGTH(5)]}
          errorText="Please enter a valid password at least 5 characters."
          onInput={inputHandler}
        />
        {/* 제출 버튼 */}
        <Button type="submit" disabled={!formState.isValid}>
          {isLoginMode ? 'LOGIN' : 'SIGNUP'}
        </Button>
        {/* 모드 전환 버튼 */}
        <Button inverse onClick={switchModeHandler}>
          SWITCH TO {isLoginMode ? 'SIGNUP' : 'LOGIN'}
        </Button>
      </form>
    </Card>
  )
}

export default Auth
```

## form-hook.js

```jsx
import { useCallback, useReducer } from 'react'

// useForm 커스텀 훅 정의
export const useForm = (initialInputs, initialFormValidity) => {
  // useReducer 훅을 사용하여 form 상태와 dispatch 함수 생성
  const [formState, dispatch] = useReducer(formReducer, {
    inputs: initialInputs,
    isValid: initialFormValidity,
  })

  // inputHandler 함수 정의: 입력값 변경을 처리하는 콜백 함수
  const inputHandler = useCallback((id, value, isValid) => {
    dispatch({
      type: 'INPUT_CHANGE', // INPUT_CHANGE 액션
      value: value, // 변경된 입력값
      isValid: isValid, // 변경된 유효성 여부
      inputId: id, // 변경된 입력값의 식별자 (id)
    })
  }, [])

  // setFormData 함수 정의: 양식 데이터와 유효성을 설정하는 콜백 함수
  const setFormData = useCallback((inputData, formValidity) => {
    dispatch({
      type: 'SET_DATA', // SET_DATA 액션
      inputs: inputData, // 변경된 양식 데이터
      formIsValid: formValidity, // 변경된 전체 양식의 유효성 여부
    })
  }, [])

  // [formState, inputHandler, setFormData] 배열 반환
  return [formState, inputHandler, setFormData]
}

// formReducer 함수 정의: useReducer에서 사용되는 상태 업데이트 로직
const formReducer = (state, action) => {
  switch (action.type) {
    case 'INPUT_CHANGE':
      let formIsValid = true

      // 모든 입력 필드에 대해 반복
      for (const inputId in state.inputs) {
        // state.inputs[inputId]가 falsy 값이면 continue (예: name 필드가 없는 경우)
        if (!state.inputs[inputId]) {
          continue
        }

        // 현재 입력 필드가 변경된 필드와 일치하는 경우
        if (inputId === action.inputId) {
          formIsValid = formIsValid && action.isValid // 현재 필드의 유효성을 변경된 값으로 업데이트
        } else {
          formIsValid = formIsValid && state.inputs[inputId].isValid // 다른 필드는 현재 상태를 유지
        }
      }

      // 변경된 입력값과 유효성을 반영한 새로운 상태 반환
      return {
        ...state,
        inputs: {
          ...state.inputs,
          [action.inputId]: { value: action.value, isValid: action.isValid },
        },
        isValid: formIsValid,
      }

    case 'SET_DATA':
      // 변경된 양식 데이터와 전체 양식의 유효성을 반영한 새로운 상태 반환
      return {
        inputs: action.inputs,
        isValid: action.formIsValid,
      }

    default:
      return state // 기본적으로 현재 상태를 그대로 반환
  }
}
```

```jsx
continue는 state.inputs[inputId]가 falsy 값일 때, 즉 해당 입력 필드가 존재하지 않는 경우에

현재 반복을 중지하고 다음 반복으로 넘어간다.

이는 예를 들어, name 필드가 없는 경우를 처리하는데 사용된다.

name 필드가 없으면 state.inputs[inputId]가 falsy하게 되어 continue가 실행되어 현재 반복이 중지된다.

이렇게 함으로써, 입력 필드가 존재하지 않는 경우에 대한 처리를 건너뛰고 다음 입력 필드 반복으로 넘어갈 수 있다
```

# **컨텍스트(Context)**

```jsx
컨텍스트(Context)란?

컨텍스트는 React 애플리케이션에서 전역적으로 데이터를 관리하기 위한 메커니즘이다.

주로 컴포넌트 간의 데이터 공유에 사용되며,

프로퍼티를 통해 데이터를 전달하지 않고도 컴포넌트 트리를 통해 값을 전파할 수 있게 해준다.
```

## **컨텍스트 사용 예제:**

```jsx

```

[App 컴포넌트에서는 AuthContext.Provider로 전역적으로 사용할 사용자 정보를 제공한다.

이 컨텍스트의 value에는 현재 로그인 상태(isLoggedIn)와 사용자 ID(userId), 그리고 로그인 및 로그아웃을 수행하는 함수들이 포함되어 있다.

이렇게 하면 AuthContext를 사용하는 하위 컴포넌트에서는 간단하게 useContext(AuthContext)를 통해 전역적인 사용자 정보에 접근할 수 있다.

이는 로그인 상태를 관리하고 각 페이지에서 해당 정보를 사용할 수 있도록 하는 중요한 부분이다.

**userId 가져오는 원리**

사용자가 로그인에 성공하면 해당 사용자의 ID(userId)를 매개변수로 전달하여 login 함수를 호출한다.

( **Auth.js 의 로그인 부분을 보면 auth.login(responseData.user.\_id); 부분이 있다 로그인 시 매개변수로 user.\_id 를 로그인 함수로 보낸다** )

setIsLoggedIn(true);는 사용자가 로그인되었음을 나타내는 상태(isLoggedIn)를 true로 변경한다.

setUserId(userId);는 userId 상태를 전달받은 값으로 설정한다.

```jsx
// AuthContext.js
import React, { createContext, useState, useContext, useCallback } from 'react'
```

# 더 많은 Auth 컨텍스트 사용 ( 로그아웃, 로그인에 따른 버튼 )

## 버튼 로그인 시 출력하기

```jsx
//PlaceItem.js

import React, { useContext, useState } from 'react'

import Card from '../../shared/components/UIElements/Card'
import Button from '../../shared/components/FormElements/Button'
import Modal from '../../shared/components/UIElements/Modal'
import { AuthContext } from '../../shared/context/auth-context'

import './PlaceItem.css'

const PlaceItem = (props) => {
  const auth = useContext(AuthContext)

  const [showMap, setShowMap] = useState(false)

  const [showConfirmModal, setShowConfirmModal] = useState(false)

  const openMapHandler = () => setShowMap(true)

  const closeMapHandler = () => setShowMap(false)

  const showDeleteWaringHandler = () => {
    setShowConfirmModal(true)
  }

  const cancelDeleteHandler = () => {
    setShowConfirmModal(false)
  }

  const confirmDeleteHandler = () => {
    setShowConfirmModal(false)
    console.log('DELETING....')
  }

  return (
    <>
      <Modal
        show={showMap}
        onCancel={closeMapHandler}
        header={props.address}
        contentClass="place-item__modal-content"
        footerClass="place-item__modal-actions"
        footer={<Button onClick={closeMapHandler}>CLOSE</Button>}
      >
        <div className="map-container">
          <h2>THE MAP</h2>
        </div>
      </Modal>

      <Modal
        show={showConfirmModal}
        onCancel={cancelDeleteHandler}
        header="Are you sure?"
        footerClass="place-item__modal-actions"
        footer={
          <>
            <Button inverse onClick={cancelDeleteHandler}>
              CANCEL
            </Button>
            <Button danger onClick={confirmDeleteHandler}>
              DELETE
            </Button>
          </>
        }
      >
        <p>정말 삭제 하시겠습니까?</p>
      </Modal>

      <li className="place-item">
        <Card className="place-item__content">
          <div className="place-item__image">
            <img src={props.image} alt={props.title} />
          </div>
          <div className="place-item__info">
            <h2>{props.title}</h2>
            <h3>{props.address}</h3>
            <p>{props.description}</p>
          </div>
          <div className="place-item__actions">
            <Button inverse onClick={openMapHandler}>
              VIEW ON MAP
            </Button>

            {auth.isLoggedIn && <Button to={`/places/${props.id}`}>EDIT</Button>}

            {auth.isLoggedIn && (
              <Button danger onClick={showDeleteWaringHandler}>
                DELETE
              </Button>
            )}
          </div>
        </Card>
      </li>
    </>
  )
}

export default PlaceItem
```

## 로그아웃 출력하기

```jsx
import React, { useContext } from 'react'
import { NavLink } from 'react-router-dom'

import { AuthContext } from '../../context/auth-context'

import './NavLinks.css'

const NavLinks = (props) => {
  const auth = useContext(AuthContext)

  return (
    <ul className="nav-links">
      <li>
        <NavLink to="/">ALL USERS</NavLink>
      </li>
      {auth.isLoggedIn && (
        <li>
          <NavLink to="/u1/places">MY PLACES</NavLink>
        </li>
      )}
      {auth.isLoggedIn && (
        <li>
          <NavLink to="/places/new">ADD PLACE</NavLink>
        </li>
      )}
      {!auth.isLoggedIn && (
        <li>
          <NavLink to="/auth">AUTHENTICATE</NavLink>
        </li>
      )}
      {auth.isLoggedIn && <button onClick={auth.logout}>LOGOUT</button>}
    </ul>
  )
}

export default NavLinks
```
