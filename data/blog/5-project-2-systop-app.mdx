### ✅ 2.2 `forEach()`로 이벤트 리스너 추가

```jsx
listItems.forEach((item, index) => {
  item.addEventListener('click', () => {
    hideAllContents()
    hideAllItems()

    // 선택한 탭 활성화
    item.classList.add('active')

    // 선택한 화면 표시
    contents[index].classList.add('show')
  })
})
```

```jsx
listItems.forEach()를 사용해 각 네비게이션 버튼에 클릭 이벤트를 추가함.

hideAllContents()와 hideAllItems()를 실행해 기존 화면을 숨기고 버튼을 비활성화한 후,

새로운 화면을 표시하도록 함.
```

```jsx
listItems.forEach()란?

listItems는 버튼 목록이야. (예: [홈, 소개, 연락처])

forEach()는 목록을 하나씩 돌면서 각각의 버튼에 클릭 이벤트를 추가해.

(item, index) => { ... }

item: 현재 순서의 버튼 요소 (예: 홈 버튼, 소개 버튼 등)

index: 버튼의 순서(인덱스), 0부터 시작 (예: 홈 = 0, 소개 = 1, 연락처 = 2)
```

### ✅ 2.3 화면 숨기기 함수

```jsx
function hideAllContents() {
  contents.forEach((content) => {
    content.classList.remove('show')
  })
}

function hideAllItems() {
  listItems.forEach((item) => {
    item.classList.remove('active')
  })
}
```

```jsx
모든 .content 요소에서 show 클래스를 제거해 안 보이게 함.

네비게이션의 모든 버튼에서 active 클래스를 제거해 버튼 스타일을 초기화함.

결과적으로

		사용자가 네비게이션의 버튼을 클릭하면:

		기존 화면이 숨겨지고 (hideAllContents())

		기존 버튼 스타일이 초기화됨 (hideAllItems())

		클릭한 탭이 활성화되고 (classList.add('active'))

		클릭한 탭의 화면이 표시됨 (classList.add('show'))
```

```jsx
메인 프로세스는 Electron의 핵심적인 역할을 하며,

시스템 리소스와 직접 연결됩니다.
```

### 메인 프로세스 파일: `main.js`

```jsx
const osu = require('node-os-utils') // 시스템 정보 라이브러리
const cpu = osu.cpu // CPU 관련 정보 가져오기
const mem = osu.mem // 메모리 관련 정보 가져오기
const os = osu.os // OS 관련 정보 가져오기
const { app, BrowserWindow, ipcMain } = require('electron')

let mainWindow

function createMainWindow() {
  mainWindow = new BrowserWindow({
    title: 'SysTop',
    width: 700,
    height: 500,
    webPreferences: {
      nodeIntegration: false,
      contextIsolation: true,
      preload: path.join(__dirname, 'preload.js'),
    },
  })

  mainWindow.loadFile('./app/index.html')
}

app.whenReady().then(createMainWindow)

// 시스템 정보 제공 (렌더러 프로세스로 데이터 전송)
ipcMain.handle('getComInfo', async () => {
  const cpuModel = await cpu.model() // CPU 모델 이름 가져오기
  const compName = await os.hostname() // 컴퓨터 이름 가져오기
  const osInfo = `${os.type()} ${os.arch()}` // 운영체제 정보 가져오기
  const memInfo = await mem.info() // 메모리 정보 가져오기

  return {
    model: cpuModel,
    compName: compName,
    osInfo: osInfo,
    memInfo: memInfo,
  }
})
```

```jsx
    backgroundColor: 'white',

      nodeIntegration: false,
```

```jsx
// 시스템 가동 시간을 시, 분, 초로 변환하는 함수
function formatUptime(seconds) {
  const hrs = Math.floor(seconds / 3600)
  const mins = Math.floor((seconds % 3600) / 60)
  const secs = Math.floor(seconds % 60)
  return `${hrs}h ${mins}m ${secs}s`
}

// CPU 및 시스템 가동 시간 업데이트 함수
async function updateStats() {
  const cpuUsage = await window.electron.getCpuUsage()
  const uptime = await window.electron.getUptime()

  document.getElementById('cpu-usage').innerText = `${cpuUsage}%`
  document.getElementById('cpu-free').innerText = `${(100 - cpuUsage).toFixed(2)}%`
  document.getElementById('sys-uptime').innerText = formatUptime(uptime)

  // 프로그레스 바 업데이트
  document.getElementById('cpu-progress').style.width = `${cpuUsage}%`
}

// 2초마다 데이터 업데이트
setInterval(updateStats, 2000)

// 실행 직후 즉시 데이터 표시
updateStats()
```

```jsx
  // CPU 사용량에 따라 프로그레스 바 스타일 변경

  if (cpuUsage > cpuOverload) {
    document.getElementById('cpu-progress').style.backgroundColor = 'red' // CPU 사용량 초과 시 빨간색
  } else {
    document.getElementById('cpu-progress').style.backgroundColor = '#30c88b' // 정상 범위일 경우 초록색
  }
}
```

```jsx
cpu-progress 요소를 선택하여 width와 backgroundColor를 변경
```

# { CPU 사용량이 일정 수준을 넘으면 알람 + 최소 시간 추가 }

```jsx
일정한 주기(기본 1분)에 맞춰 알람을 보내는 기능을 설명했습니다.
```

## CPU 사용량이 일정 수준을 넘으면 알람을 보내는 기능

### (1) `cpuOverload`와 `alertFrequency` 설정

```jsx
let cpuOverload = 15 // CPU 사용량이 15%를 넘으면 알람을 보냄
let alertFrequency = 1 // 알람을 1분마다만 보내도록 설정
let firstRun = true // 첫 실행 여부 확인을 위한 변수
```

```jsx
cpuOverload: CPU 사용량이 이 값을 초과하면 알람을 보냄.

alertFrequency: 알람을 최소 몇 분 간격으로 보낼지 설정.

firstRun: 프로그램이 처음 실행될 때 알람이 즉시 뜨도록 하기 위한 변수.
```

### (2) CPU 사용량을 확인하고 알람을 보내는 `updateStats` 함수

```jsx
async function updateStats() {
  const cpuUsage = await window.electron.getCpuUsage() // CPU 사용량 가져오기
  const uptime = await window.electron.getUptime() // 시스템 가동 시간 가져오기

  document.getElementById('cpu-usage').innerText = `${cpuUsage}%`
  document.getElementById('cpu-free').innerText = `${(100 - cpuUsage).toFixed(2)}%`
  document.getElementById('sys-uptime').innerText = formatUptime(uptime)

  document.getElementById('cpu-progress').style.width = `${cpuUsage}%`
  if (cpuUsage > cpuOverload) {
    document.getElementById('cpu-progress').style.backgroundColor = 'red' // 과부하 시 빨간색으로 표시
  } else {
    document.getElementById('cpu-progress').style.backgroundColor = '#30c88b' // 정상일 때 초록색
  }

  // CPU 사용량이 임계값을 초과하고, 알람을 보낼 시간이 되었다면
  if (cpuUsage >= cpuOverload && runNotify(alertFrequency)) {
    notifyUser({
      title: '알람 테스트',
      body: `CPU 오버${cpuOverload}%`,
      icon: '../../assets/icons/win/icon.png',
    })

    localStorage.setItem('lastNotify', +new Date()) // 현재 시간을 저장 (마지막 알람 시간) +new Date() → 날짜 객체를 **타임스탬프(Number, 밀리초 단위)**로 변환  localStorage에 저장 때 문자로 저장하면 다시 돌려야함 그래서 숫자로 하기 위해
  }
}
```

```jsx
1. CPU 사용량과 시스템 가동 시간 가져오기

		window.electron.getCpuUsage(): 현재 CPU 사용량을 가져옴.

		window.electron.getUptime(): 현재 시스템이 켜져 있는 시간을 가져옴.

2. CPU 사용량을 화면에 업데이트

		CPU 사용량을 % 단위로 표시하고, 프로그레스 바 색깔을 변경.

		cpuOverload(기본 15%)를 초과하면 빨간색으로 표시.

3. 알람을 보내는 조건

		cpuUsage >= cpuOverload: CPU 사용량이 임계값을 초과했는지 확인.

		runNotify(alertFrequency): 마지막 알람 이후 설정한 시간이 지났는지 확인.

		위 두 조건을 만족하면 notifyUser()를 실행하여 알람을 보냄.
```

## 일정 시간이 지나야만 알람이 다시 뜨도록 설정 (`runNotify` 함수)

```jsx
function runNotify(frequency) {
  // ✅ 첫 실행 여부를 확인
  if (firstRun) {
    firstRun = false // 첫 실행 이후에는 false로 변경하여 이후부터 주기 제한 적용
    return true // 첫 실행 시에는 즉시 알람을 실행
  }

  // ✅ 이전 알람 기록이 없는 경우 (즉, localStorage에 'lastNotify' 값이 없음)
  if (localStorage.getItem('lastNotify') === null) {
    localStorage.setItem('lastNotify', +new Date()) // 현재 시간을 타임스탬프로 저장
    return true // 알람을 실행
  }

  // ✅ 이전 알람이 실행된 시간을 가져오기
  const notifyTime = new Date(parseInt(localStorage.getItem('lastNotify'))) // 저장된 값(타임스탬프)을 숫자로 변환 후 Date 객체로 생성
  const now = new Date() // 현재 시간 가져오기

  // ✅ 시간 차이 계산 (밀리초 단위)
  const diffTime = Math.abs(now - notifyTime)
  // Math.abs() : 절대값을 반환하는 함수. 혹시라도 notifyTime이 now보다 클 경우에도 항상 양수값을 유지

  // ✅ 경과된 분 계산
  const minutesPassed = Math.ceil(diffTime / (1000 * 60))
  // diffTime은 밀리초 단위이므로 1000(1초) * 60(1분)으로 나눠 분 단위로 변환
  // Math.ceil() : 소수점이 발생할 경우 올림하여 정수로 변환 (예: 1.2분 → 2분)

  // ✅ 주기(frequency)보다 많이 지났으면 true 반환 → 알람 실행
  return minutesPassed > frequency
}
```

```jsx
1. 첫 실행 시에는 바로 알람을 보냄

		firstRun 변수를 사용하여 처음 실행될 때는 무조건 true를 반환하여 즉시 알람이 뜨도록 함.

		이후부터는 firstRun = false로 바뀌어 주기 제한이 적용됨.

2. 이전 알람 시간이 없으면 현재 시간을 저장하고 알람을 보냄

		localStorage.getItem('lastNotify') === null: 로컬 저장소에 알람 시간이 없으면 현재 시간을 저장하고 true 반환.

3. 이전 알람 시간과 현재 시간 비교

		notifyTime: 마지막 알람 시간을 불러와 new Date() 객체로 변환.

		diffTime: 현재 시간과 마지막 알람 시간의 차이를 밀리초 단위로 계산.

		minutesPassed: 밀리초를 분 단위로 변환 (diffTime / (1000 * 60))

		설정한 주기(frequency)보다 오래되었으면 true, 그렇지 않으면 false 반환.
```
