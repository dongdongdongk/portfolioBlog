# 3. 자바 스크립트 작성

```jsx
<script>
    $(document).ready(function () {   // 페이지 불러오면 바로 시작
        let visibleReviews = 3;
        let totalReviews = $(".review").length;     // 총 리뷰 숫자는 class review 컨테이너의 숫자

        if (visibleReviews >= totalReviews) {  // 현재 보이는 리뷰 버튼이 totalReviews 버튼을 넘어서면 리뷰 더보기 버튼을 숨기는 기능 (처음 페이지에 들어왔을 때 3개 이하면 버튼이 없어지도록
            $("#loadMoreBtn").hide();
        }

        $("#loadMoreBtn").click(function () {
            $(".review:hidden").slice(0, 3).slideDown();  // 현재 숨겨져 있는 "review"의 모든 요소를 선택 (CSS의 display: none)
																													// slice(0, 3): 인덱스 0부터 인덱스 3까지의 요소를 선택 즉 컬렉션에서 처음 세 개의 숨겨진 요소를 선택 .slideDown() 으로 숨겨진 요소 아래로 슬라이드 애니메이션으로 나타나게 하는 기능

            visibleReviews += 3;                          // 버튼 클릭 때 마다 3씩 더하기
            if (visibleReviews >= totalReviews) {         // 최대 리뷰 숫자를 넘어서면 버튼 숨기기
                $("#loadMoreBtn").hide();
            }
        });
    });
</script>
```
