## ProtectedRoute.js

```jsx
// React와 라우터에서 필요한 모듈을 가져옵니다.
import { useSelector } from 'react-redux'
import { Navigate } from 'react-router-dom'

// 사용자의 로그인 상태를 확인하여 보호된 라우트를 처리하는 컴포넌트를 정의합니다.
const ProtectedRoute = ({ children }) => {
  // Redux store에서 사용자 정보 및 로딩 상태를 가져옵니다.
  const { loading, isAuthenticated } = useSelector((state) => state.user)

  // 로딩 상태가 완료된 경우
  if (loading === false) {
    // 사용자가 인증되지 않은 경우 로그인 페이지로 이동합니다.
    if (!isAuthenticated) {
      return <Navigate to="/login" replace />
    }
    // 사용자가 인증된 경우 자식 컴포넌트(보호된 컨텐츠)를 반환합니다.
    return children
  }
}

// 컴포넌트를 내보냅니다.
export default ProtectedRoute
```

## SellerProtectedRoute.js

```jsx
// React와 라우터에서 필요한 모듈을 가져옵니다.
import { useSelector } from 'react-redux'
import { Navigate } from 'react-router-dom'
import Loader from '../components/Layout/Loader'

// 판매자 전용 보호 라우트를 처리하는 컴포넌트를 정의합니다.
const SellerProtectedRoute = ({ children }) => {
  // Redux store에서 판매자 정보와 로딩 상태를 가져옵니다.
  const { loading, isSeller } = useSelector((state) => state.seller)

  // 로딩 중인 경우 로딩 스피너를 표시합니다.
  if (loading) {
    return <Loader />
  } else {
    // 판매자로 인증되지 않은 경우 판매자 로그인 페이지로 이동합니다.
    if (!isSeller) {
      return <Navigate to={`/shop-login`} replace />
    }
    // 판매자로 인증된 경우 자식 컴포넌트(판매자 전용 컨텐츠)를 반환합니다.
    return children
  }
}

// 컴포넌트를 내보냅니다.
export default SellerProtectedRoute
```

```jsx
// React와 Lottie를 사용하여 로딩 스피너를 렌더링하는 Loader 컴포넌트입니다.
import React from 'react'
import Lottie from 'react-lottie'
import Loading from '../../Lottie/LOADING.json'

const Loader = () => {
  // Lottie 애니메이션의 기본 옵션을 설정합니다.
  const defaultOptions = {
    loop: true, // 애니메이션을 반복 재생합니다.
    autoplay: true, // 페이지 로딩 시 자동으로 애니메이션을 시작합니다.
    animationData: Loading, // 사용할 Lottie 애니메이션 데이터입니다.
    rendererSettings: {
      preserveAspectRatio: 'xMidYMid slice', // 애니메이션을 중앙에 유지하고 잘라내기 위한 설정입니다.
    },
  }

  // 화면 전체를 가득 채우고 가운데에 로딩 스피너를 표시합니다.
  return (
    <div className="flex h-screen w-full items-center justify-center">
      {/* Lottie 컴포넌트에 설정된 옵션을 전달하여 애니메이션을 렌더링합니다. */}
      <Lottie options={defaultOptions} width={300} height={300} />
    </div>
  )
}

// Loader 컴포넌트를 내보냅니다.
export default Loader
```

## ShopInfo.jsx 컴포넌트를 만들어준다

```jsx
import axios from 'axios'
import React, { useEffect, useState } from 'react'
import { useParams } from 'react-router-dom'
import styles from '../../styles/styles'
import Loader from '../Layout/Loader'

const ShopInfo = ({ isOwner }) => {
  // 로컬 상태 관리를 위한 useState 훅 사용
  const [data, setData] = useState({})
  const [isLoading, setIsLoading] = useState(false)

  // React Router의 useParams 훅을 사용하여 URL 파라미터 값 추출
  const { id } = useParams()

  // useEffect 훅을 사용하여 컴포넌트가 마운트될 때 한 번만 실행되도록 설정
  useEffect(() => {
    // 데이터 로딩 상태를 true로 설정하여 로딩 중임을 표시
    setIsLoading(true)

    // Axios를 사용하여 서버로부터 상점 정보를 가져옴
    axios
      .get(`http://localhost:5000/api/v2/shop/get-shop-info/${id}`)
      .then((res) => {
        // 서버 응답으로 받은 데이터를 로컬 상태에 저장
        setData(res.data.shop)
        // 데이터 로딩이 완료되었으므로 로딩 상태를 false로 업데이트
        setIsLoading(false)
      })
      .catch((error) => {
        // 에러가 발생하면 콘솔에 에러 로그를 출력하고 로딩 상태를 false로 설정
        console.log(error)
        setIsLoading(false)
      })
  }, []) // 빈 배열을 전달하여 컴포넌트가 처음 렌더링될 때만 useEffect가 실행되도록 설정

  // 로그아웃 핸들러 함수 정의
  const logoutHandler = async () => {
    // Axios를 사용하여 로그아웃 요청을 서버에 전송
    axios.get(`http://localhost:5000/api/v2/shop/logout`, {
      withCredentials: true,
    })

    // 페이지 새로고침을 통해 로그아웃 적용
    window.location.reload()
  }

  // JSX로 컴포넌트 렌더링
  return (
    <>
      {/* 데이터 로딩 중인지 확인하고, 로딩 중이라면 로더 컴포넌트를 표시 */}
      {isLoading ? (
        <Loader />
      ) : (
        <div>
          {/* 상점 정보를 화면에 출력 */}
```

```jsx
import React, { useEffect, useState } from 'react'
```

```jsx
const CreateProduct = () => {
  // Redux에서 판매자 정보 및 상품 생성 성공 여부와 에러를 가져옴
  const { seller } = useSelector((state) => state.seller)
  const { success, error } = useSelector((state) => state.product)

  // React Router DOM의 useNavigate 훅을 사용하여 페이지 이동을 담당하는 함수를 가져옴
  const navigate = useNavigate()
  const dispatch = useDispatch()

  // 상품 등록에 필요한 상태들을 관리하는 useState 훅을 사용
  const [images, setImages] = useState([])

  const [description, setDescription] = useState('')
  const [category, setCategory] = useState('')

  const [discountPrice, setDiscountPrice] = useState()
  const [stock, setStock] = useState()

  // 상품 생성 성공 및 에러 상태를 모니터링하여 메시지를 토스트로 표시하고 페이지를 리로드함
  useEffect(() => {
    if (error) {
      toast.error(error)
    }
    if (success) {
      toast.success('상품등록 완료!')

    newForm.append('description', description)
    newForm.append('category', category)
    newForm.append('tags', tags)
    newForm.append('originalPrice', originalPrice)
    newForm.append('discountPrice', discountPrice)
    newForm.append('stock', stock)
    newForm.append('shopId', seller._id)

    dispatch(createProduct(newForm))
  }

  return (
    <div className="800px:w-[50%] h-[80vh] w-[90%] overflow-y-scroll rounded-[4px] bg-white p-3 shadow">

          </label>
          <input

            value={name}

          <textarea
```

```jsx
const handleImageChange = (e) => {
  // 이벤트 기본 동작을 막음(페이지 새로고침 방지)
  e.preventDefault();
```

```jsx
이 설정은 현재 요청이 multipart/form-data 형식의 데이터를 포함하고 있다는 것을 나타낸다.

multipart/form-data는 주로 파일 업로드와 함께 사용되는 데이터 전송 형식으로,

이 형식을 사용하면 텍스트 데이터와 파일을 동시에 전송할 수 있다. 이 경우, 주로 폼 데이터에 이미지나 파일이 첨부된 상황에서 사용된다.

따라서, config 객체는 Axios로 HTTP 요청 시 사용되며,

해당 요청의 헤더로 "Content-Type"이 "multipart/form-data"로 설정되어 있다.

이를 통해 서버는 전송된 데이터가 파일 등을 포함하는 복합적인 형식임을 이해하고 처리할 수 있게 된다.
```

```jsx
        state.success = false // 성공 상태 초기화
        state.error = null // 에러 초기화
      })
      .addCase(createProduct.fulfilled, (state, action) => {
        state.isLoading = false // 액션 완료 후 로딩 상태 해제
        state.product = action.payload // 액션에서 받아온 제품 정보 저장
        state.success = true // 성공 상태 설정
      })
      .addCase(createProduct.rejected, (state, action) => {
        state.isLoading = false // 액션 수행 실패 시 로딩 상태 해제
        state.error = action.error.message // 액션에서 받아온 에러 메시지 저장
        state.success = false // 실패한 경우 성공 상태 초기화
      })
      // getAllProductsShop 비동기 액션 핸들링
      .addCase(getAllProductsShop.pending, (state) => {
        state.isLoading = true // 액션 수행 중 로딩 상태 설정
        state.error = null // 에러 초기화
      })
      .addCase(getAllProductsShop.fulfilled, (state, action) => {
        state.isLoading = false // 액션 완료 후 로딩 상태 해제
        state.products = action.payload // 액션에서 받아온 제품 목록 저장
      })
      .addCase(getAllProductsShop.rejected, (state, action) => {
        state.isLoading = false // 액션 수행 실패 시 로딩 상태 해제
        state.error = action.error.message // 액션에서 받아온 에러 메시지 저장
      })
      // deleteProduct 비동기 액션 핸들링
      .addCase(deleteProduct.pending, (state) => {
        state.isLoading = true // 액션 수행 중 로딩 상태 설정
      })
      .addCase(deleteProduct.fulfilled, (state, action) => {
        state.isLoading = false // 액션 완료 후 로딩 상태 해제
        state.error = action.payload // 액션에서 받아온 에러 메시지 저장
      })
      .addCase(deleteProduct.rejected, (state, action) => {
        state.isLoading = false // 액션 수행 실패 시 로딩 상태 해제
        state.error = action.payload // 액션에서 받아온 에러 메시지 저장
      })
  },
})
```
