const jwt = require('jsonwebtoken')
const HttpError = require('../models/http-error')

// 토큰 검증 미들웨어
module.exports = (req, res, next) => {
  if (req.method === 'OPTIONS') {
    return next()
  }

  // 1. 요청 헤더에서 토큰 추출
  const token = req.headers.authorization.split(' ')[1]

  try {
    // 2. 토큰이 존재하지 않으면 에러 발생
    if (!token) {
      throw new Error('인증 실패')
    }

    // 3. 토큰 검증
    const decodedToken = jwt.verify(token, process.env.JWT_KEY)

    // 4. 검증이 성공하면 사용자 데이터를 요청 객체에 추가
    req.userData = { userId: decodedToken.userId }

    // 5. 다음 미들웨어로 진행
    next()
  } catch (err) {
    // 6. 토큰 검증 실패 시 403 상태코드와 함께 에러 응답
    const error = new HttpError('인증 실패', 403)
    return next(error)
  }
}

# React에 코드 분할 추가하기

```jsx
import React, { Suspense } from 'react'
import './App.css'
import { Route, Routes, Navigate } from 'react-router-dom'

// 예전 임포트는 주석 처리
// import Users from './user/pages/Users';
// import NewPlace from './places/pages/NewPlace';
// import UpdatePlace from './places/pages/UpdatePlace';
// import UserPlaces from './places/pages/UserPlaces';
// import Auth from './user/pages/Auth';

import MainNavigation from './shared/components/Navigation/MainNavigation'
import { AuthContext } from './shared/context/auth-context'
import useAuth from './shared/hooks/auth-hook'
import LoadingSpinner from './shared/components/UIElements/LoadingSpinner'

const Users = React.lazy(() => import('./user/pages/Users')) // 첫 페이지기 때문에 lazy 로딩이 필요없다
const NewPlace = React.lazy(() => import('./shared/components/Navigation/MainNavigation'))
const UpdatePlace = React.lazy(() => import('./places/pages/UpdatePlace'))
const UserPlaces = React.lazy(() => import('./places/pages/UserPlaces'))
const Auth = React.lazy(() => import('./user/pages/Auth'))

const App = () => {
  const { token, login, logout, userId } = useAuth()

  // 라우팅을 위한 JSX 설정
  let routes
  if (token) {
    // 로그인한 경우의 라우트 설정
    routes = (
      <>
        <Route path="/" element={<Users />} />
        <Route path="/:userId/places" element={<UserPlaces />} />
        <Route path="/places/new" element={<NewPlace />} />
        <Route path="/places/:placeId" element={<UpdatePlace />} />
        <Route path="*" element={<Navigate replace to="/" />} />
      </>
    )
  } else {
    // 로그인하지 않은 경우의 라우트 설정
    routes = (
      <>
        <Route path="/" element={<Users />} />
        <Route path="/:userId/places" element={<UserPlaces />} />
        <Route path="/auth" element={<Auth />} />
        <Route path="*" element={<Navigate replace to="/auth" />} />
      </>
    )
  }

  // AuthContext.Provider를 사용하여 전역적으로 사용되는 사용자 정보 관리
  return (
    <AuthContext.Provider
      value={{
        isLoggedIn: !!token,
        token: token,
        userId: userId,
        login: login,
        logout: logout,
      }}
    >
      <>
        {/* 상단 네비게이션 바 */}
        <MainNavigation />

        {/* 메인 컨텐츠 영역 */}
        <main>
          <Suspense
            fallback={
              <div className="center">
                <LoadingSpinner />
              </div>
            }
          >
            {/* 라우트 설정 */}
            <Routes>{routes}</Routes>
          </Suspense>
        </main>
      </>
    </AuthContext.Provider>
  )
}

export default App
```

## **1. 코드 분할이란 무엇인가요?**

```jsx
코드 분할은 애플리케이션의 모든 코드를 처음에 로드하는 것이 아니라, 필요한 코드 조각만 로드하여 초기 로딩 속도를 향상시키는 기술이다.

React 애플리케이션에서는 React.lazy()와 Suspense를 사용하여 코드를 런타임에서 분할

( 어플리케이션이 실행 중일 때, 특정 코드 조각이 필요한 순간에 해당 코드를 로드한다는 의미 )할 수 있다.
```

## **2. React 코드 분할 설정**

```jsx
2.1. React.lazy() 함수 사용

React.lazy() 함수는 동적으로 import 문을 사용하여 코드를 로드하는 함수를 생성한다. 이 함수는 코드를 처음에 로드하지 않고, 필요한 시점에 로드한다.

2.2. import 문 변경

React.lazy()를 사용하는 컴포넌트의 import 문을 삭제하거나 주석 처리한다. 이로써 해당 컴포넌트는 처음에 로드되지 않는다.

2.3. 다른 컴포넌트에도 적용

필요한 다른 컴포넌트에도 React.lazy()를 적용하여 코드 분할을 확장한다. 각 컴포넌트에 대해 위의 단계를 반복한다.
```

# 배포 대안 이해하기

## **첫 번째 옵션: 개별 앱 두 개를 배포**

```jsx
1. 설명

React 앱과 Node REST API를 각각 개별적으로 호스팅하는 방법.

두 개의 서버가 필요하며, 서로 다른 호스팅 서비스에 배포됨.

2. 장점

두 앱 간 독립적인 배포로 유연성 증가.

React 앱은 정적 호스팅 서비스로, Node REST API는 Node.js 호스팅으로 각자 최적화 가능.

3. 주의사항

CORS 헤더를 설정하여 도메인 간 리소스 공유가 필요.

서로 다른 서버 간의 통신에서 보안 설정 필요.

4. 호스팅 서비스 예시

React 앱: AWS S3, Firebase Hosting 등 정적 호스팅 제공자.

Node REST API: AWS EC2, Elastic Beanstalk, Heroku 등 Node.js 호스팅 제공자.
```

## **두 번째 옵션: 하나의 서버에 결합한 앱을 배포**

```jsx
1. 설명

React 앱과 Node REST API를 하나의 서버에서 실행하고 배포하는 방법.

Node.js 코드에서 React 앱을 반환하도록 설정하여 하나의 호스팅 서비스 사용.

2. 장점

간편한 배포 및 유지보수. Node.js 코드를 실행하는 호스팅 서비스만 필요.

CORS 설정이 필요 없음.

3. 주의사항

React 앱 자산을 정적으로 지원할 수 있는 호스팅 서비스 필요.

4. 호스팅 서비스 예시

AWS EC2, Elastic Beanstalk, Heroku 등 Node.js 호스팅 제공자.
```

## **주요 차이점 및 수정사항**

```jsx
1. 개별 앱 배포

CORS 헤더 설정이 필요.

React 앱은 정적 호스팅을 위한 서버 필요.

Node REST API는 Node.js 호스팅을 위한 서버 필요.

2. 하나의 서버에 결합한 앱 배포

CORS 설정이 필요 없음.

React 앱을 정적 호스팅 제공자로 배포 가능.

3. 코드 수정 사항

개별 앱: 기존 설정 유지, CORS 헤더 필요.

결합 앱: CORS 헤더 제거해도 괜찮음.
```

# 배포 시 주의해야 할 점

```jsx
// create token and saving that in cookies
const sendShopToken = (user, statusCode, res) => {
    const token = user.getJwtToken();

    // Options for cookies
    const options = {
      expires: new Date(Date.now() + 90 * 24 * 60 * 60 * 1000),
      httpOnly: true,
      sameSite: "none",
      secure: true,
    };

    res.status(statusCode).cookie("seller_token", token, options).json({
      success: true,
      user,
      token,
    });
  };

  module.exports = sendShopToken;


  위와 같이 sameSite: "none", 과 secure: true, 을 설정해줘야 쿠키를 사용 가능하다 ( 없으면 발급받지 못하기 때문에 로그인 등등 모든게 불가능해진다 )
```
