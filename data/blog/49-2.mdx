void method1() {

- 바깥클래스(A class) 에서 인스턴스 멤버 class B 는 인스턴스 멤버 필드,메소드 에서만 객체를 생성할 수 있고 정적 멤버 class C 는 모든 필드의 초기값과 메소드에서 사용 가능하다.

### 멤버 클래스에서 사용 제한

![Untitled](<49%20%EC%A4%91%EC%B2%A9%20%ED%81%B4%EB%9E%98%EC%8A%A4%EC%99%80%20%EC%A4%91%EC%B2%A9%20%EC%9D%B8%ED%84%B0%ED%8E%98%EC%9D%B4%EC%8A%A4(2)/Untitled%202.png>)

```java
class A {
	int field1;          //인스턴스 필드,메소드
	void method1() {}

	static int field2;        //정적 필드,메소드
	static void method2() {}

	class B {                 // static 이 없으니 A객체가 있어야 사용가능한 클래스
		void method() {         // B 객체가 있어야 사용가능한 메소드 (결과적으로 A.B가 필요한메소드)
			field1 = 10;          // A가 가지고 있는 인스턴스,정적 메소드와 필드 제한없이 사용가능
			method1();

			field2 = 10;
			method2();
		}
	}
}
```

```java
class A {
	int field1;          //인스턴스 필드,메소드
	void method1() {}

	static int field2;        //정적 필드,메소드
	static void method2() {}

	static class C {          //정적 클래스 C 는 A 객체없이 사용가능
		void method() {
			field1 = 10; (x)        //A객체가 필요한 인스턴스 필드와 메소드는 A가 필요없는 정적
			method1();   (x)        //클래스 C에서는 사용 불가능

			field2 = 10;          //정적 필드 메소드 사용가능
			method2();
		}
	}
}
```
