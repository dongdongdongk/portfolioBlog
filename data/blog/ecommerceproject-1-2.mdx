```jsx
import { Link } from 'react-router-dom'

  const [dropDown, setDropDown] = useState(false)

    if (window.scrollY > 70) {
      setActive(true)

      setActive(false)
    }

  return (

        <div className="800px:h-[50px] 800px:my-[20px] 800px:flex hidden items-center justify-between">

            {/* 로고 이미지에 링크 추가 */}

            <input

              placeholder="Search Product..."
              value={searchTerm}
              onChange={handleSearchChange}
              className="h-[40px] w-full rounded-md border-[2px] border-[#e3bb8b] px-2"
            />
            {/* 검색 아이콘 */}
            <AiOutlineSearch size={30} className="absolute top-1.5 right-2 cursor-pointer" />

            {searchData && searchData.length !== 0 ? (
              <div className="shadow-sm-2 absolute z-[9] min-h-[30vh] bg-slate-50 p-4">
                {/* 검색 결과 목록 매핑 */}
                {searchData.map((i, index) => {
                  const d = i.name

                  // 제품 이름을 URL에 맞게 변환

                    <Link to={`/product/${Product_name}`} key={index}>

          </div>

          <div className={`${styles.button}`}>
            <Link to="/seller">
              <h1 className="flex items-center text-[#fff]">
                Become Seller <IoIosArrowForward className="ml-1" />

      >
        <div className={`${styles.section} relative ${styles.noramlFlex} justify-between`}>
          {/* 카테고리 드롭다운 영역 */}
```

## 검색 기능 제작

```jsx
useState를 사용한 상태 초기화:

검색어 및 검색 결과를 관리할 상태 변수들을 useState를 통해 초기화한다.

const [searchTerm, setSearchTerm] = useState("");
const [searchData, setSearchData] = useState(null);

------------------------------------------------------------------------------------------------------------------

검색어 입력 처리:

handleSearchChange 함수는 검색어 입력 시 호출된다.

입력된 검색어는 setSearchTerm을 통해 상태에 업데이트된다.

includes(term.toLowerCase())는 현재 검색어가 소문자로 변환된 제품 이름에 포함되어 있는지를 체크하여 해당하는 제품들을 필터링

제품 데이터에서 검색어에 해당하는 제품을 찾아 setSearchData를 통해 상태에 업데이트한다.

const handleSearchChange = (e) => {
  const term = e.target.value;
  setSearchTerm(term);

  const filteredProducts = productData.filter((product) =>
    product.name.toLowerCase().includes(term.toLowerCase()) // 대 소문자 구분을 없에기 위해서
  );
  setSearchData(filteredProducts);
};

------------------------------------------------------------------------------------------------------------------

검색 결과 렌더링:

검색어가 입력되면 검색 결과를 화면에 렌더링한다. ( 위에서 setSearchData에 검색 결과를 저장 해놨다 그걸 랜더링 )

검색 결과가 존재하면 검색어에 부합하는 제품 목록을 링크와 함께 화면에 표시한다.

제품의 이미지와 이름이 표시된다.

{searchData && searchData.length !== 0 ? (
  <div className="absolute min-h-[30vh] bg-slate-50 shadow-sm-2 z-[9] p-4">
    {searchData.map((i, index) => {
      const d = i.name;
      const Product_name = d.replace(/\s+/g, "-");

      return (
        <Link to={`/product/${Product_name}`} key={index}>
          <div className="w-full flex items-start-py-3">
            <img
              src={i.image_Url[0].url}
              alt=""
              className="w-[40px] h-[40px] mr-[10px]"
            />
            <h1>{i.name}</h1>
          </div>
        </Link>
      );
    })}
  </div>
) : null}

searchData.map((i, index) => {: searchData 배열을 순회하면서 각 요소(i)와 인덱스(index)에 대한 매핑 작업을 시작한다. ( index 는 솔직히 사용 x)

const d = i.name;:

현재 순회 중인 i 객체의 name 속성 값을 d라는 변수에 할당한다.

const Product_name = d.replace(/\s+/g, "-");:

d 변수에 저장된 제품 이름에서 공백을 모두 "-"로 대체한다. 정규 표현식 / /g를 사용하여 모든 공백을 찾아 대체한다.

이렇게 만들어진 Product_name은 URL에 사용하기 적합한 형태로 변환된 제품 이름이 된다.

이러한 과정을 통해 각 제품의 이름에서 공백이 제거되고, 그 대신에 "-"로 대체된 URL 친화적인 형태의 이름이 Product_name 변수에 저장된다.

이후 이 값은 URL을 구성하는 데 사용되며, 예를 들어 /product/제품이름-변환된-형태와 같은 경로로 이동할 때 활용될 것이다.
```

### 카테고리 DropDown

```jsx
카테고리 버튼 클릭:

카테고리 버튼을 클릭하면 onClick 이벤트가 발생하고, setDropDown(!dropDown) 함수가 호출된다.

setDropDown 함수 호출:

setDropDown 함수가 호출되면서 dropDown 상태가 현재의 반대값으로 업데이트된다. 따라서 dropDown이 true였다면 false로, false였다면 true로 변경된다.

DropDown 컴포넌트 렌더링:

dropDown 상태가 변경되면 해당 상태를 바탕으로 조건부 렌더링이 이루어진다.

dropDown이 true일 때는 DropDown 컴포넌트가 렌더링되고, false일 때는 null이 반환된다.

DropDown 컴포넌트:

DropDown 컴포넌트는 categoriesData와 setDropDown prop을 받아오며, 이를 활용하여 카테고리 목록을 표시한다.

( 이 데이터는 더미 DB 에 들어가 있다 )
```

```jsx
// DropDown.jsx

import React from 'react'
import { useNavigate } from 'react-router-dom'
import styles from '../../styles/styles'

const DropDown = ({ categoriesData, setDropDown }) => {
  const navigate = useNavigate()
  const submitHandle = (i) => {
    navigate(`/products?category=${i.title}`)
    setDropDown(false)
    window.location.reload()
  }
  return (
    <div className="absolute z-30 w-[270px] rounded-b-md bg-[#fff] pb-4 shadow-sm">
      {categoriesData &&
        categoriesData.map((i, index) => (
          <div key={index} className={`${styles.noramlFlex}`} onClick={() => submitHandle(i)}>
            <img
              src={i.image_Url}
              style={{
                width: '25px',
                height: '25px',
                objectFit: 'contain',
                marginLeft: '10px',
                userSelect: 'none',
              }}
              alt=""
            />
            <h3 className="m-3 cursor-pointer select-none">{i.title}</h3>
          </div>
        ))}
    </div>
  )
}

export default DropDown
```

```jsx
카테고리 목록 표시:

카테고리 목록은 map 함수를 사용하여 각 카테고리에 대한 버튼 또는 링크를 생성한다.

DropDown 닫기:

setDropDown(false)를 호출하여 DropDown를 닫는다. 이로써 사용자가 카테고리를 선택한 후에는 드롭다운이 자동으로 닫히게 된다.

페이지 새로고침:

window.location.reload()를 사용하여 현재 페이지를 새로고침한다.

이것은 선택된 카테고리에 해당하는 제품 목록이 표시되도록 강제로 페이지를 다시 불러오는 역할을 한다.
```

```jsx
이벤트 리스너 등록:

window.addEventListener("scroll", () => { ... });

window 객체에 스크롤 이벤트를 감지하는 이벤트 리스너를 등록한다. 즉, 사용자가 페이지를 스크롤할 때마다 내부의 함수가 실행된다.

스크롤 위치 확인:

if (window.scrollY > 70) {
  setActive(true);
} else {
  setActive(false);
}

window.scrollY는 현재 스크롤된 세로 위치를 나타낸다.

조건문을 사용하여 현재 스크롤 위치가 70보다 크면 setActive(true)를 호출하여 active 상태를 true로 설정한다.

그렇지 않으면 setActive(false)를 호출하여 active 상태를 false로 설정한다.
```

```jsx
import React, { useEffect, useState } from "react";
import { productData } from "../../static/data"; // 외부에서 가져온 상품 데이터
import styles from "../../styles/styles"; // 스타일 관련 클래스들이 정의된 모듈
import ProductCard from "./ProductCard"; // ProductCard 컴포넌트를 가져옴

// BestDeals 컴포넌트 정의
const BestDeals = () => {
  const [data, setData] = useState([]); // 상품 데이터를 담을 상태 변수

  useEffect(() => {
    // 컴포넌트가 마운트될 때 한 번만 실행되는 useEffect 훅
    const sortedData =
      productData && productData.sort((a, b) => b.total_sell - a.total_sell); // 상품 데이터를 판매량을 기준으로 내림차순 정렬
    const firstFive = sortedData.slice(0, 5); // 정렬된 데이터 중 상위 다섯 개만 추출
    setData(firstFive); // 추출된 상위 다섯 개의 데이터를 상태 변수에 설정
  }, []); // 두 번째 매개변수가 빈 배열이므로 컴포넌트가 마운트될 때 한 번만 실행

  // JSX로 구성된 화면 반환
  return (
    <div>
      <div className={`${styles.section}`}>
        <div className={`${styles.heading}`}>
          <h1>최다 판매</h1>
        </div>
        {/* 그리드 레이아웃으로 상품 카드를 나열 */}
        <div className="grid grid-cols-1 gap-[20px] md:grid-cols-2 md:gap-[25px] lg:grid-cols-4 lg:gap-[25px] xl:grid-cols-5 xl:gap-[30px] mb-12 border-0">
          {/* 상태 변수에 있는 데이터를 매핑하여 각각의 ProductCard 컴포넌트로 렌더링 */}
          {data && data.map((i, index) => <ProductCard data={i} key={index} />)}
        </div>
      </div>
  );
};

export default BestDeals;
```

```jsx
a와 b는 비교되는 두 요소이다.

(b.total_sell - a.total_sell): 내림차순으로 정렬하기 위해 두 요소의 판매량(total_sell)을 비교한다.

만약 b.total_sell - a.total_sell가 양수면 b가 a보다 크다고 간주되어 내림차순으로 정렬된다.

만약 음수나 0이면 a가 b보다 크거나 같다고 간주되어 정렬되지 않거나 오름차순으로 정렬된다.

예를 들어, 판매량이 각각 10, 5, 8인 상품이 있다면, 내림차순 정렬 결과는 [10, 8, 5]가 된다.

slice(0, 5):

slice 메서드는 첫 번째 매개변수부터 두 번째 매개변수 직전까지의 요소를 포함하는 부분 배열을 반환한다.

즉, 첫 번째 인덱스(0)부터 다섯 번째 인덱스(5 직전)까지의 요소를 추출한다.

( 이는 5개의 상품만 출력하는 부분이다 )

data.map((i, index) => ... ): map 함수를 사용하여 배열의 각 항목을 순회하면서 JSX를 생성한다. i는 배열의 각 항목을 나타낸다
```

## 카운트 다운 컴포넌트

```jsx
import React, { useEffect, useState } from 'react'

const CountDown = () => {
  // useState를 사용하여 남은 시간을 관리하는 상태 변수
  const [timeLeft, setTimeLeft] = useState(calculateTimeLeft())

  useEffect(() => {
    // 1초마다 calculateTimeLeft 함수를 호출하여 남은 시간을 업데이트
    const timer = setTimeout(() => {
      setTimeLeft(calculateTimeLeft())
    }, 1000)

    // 컴포넌트가 언마운트될 때 타이머를 정리(clear)
    return () => clearTimeout(timer)
  })

  // 주어진 날짜까지의 남은 시간을 계산하는 함수
  function calculateTimeLeft() {
    const difference = +new Date('2024-05-10') - +new Date()
    let timeLeft = {}

    if (difference > 0) {
      // 남은 시간을 일, 시간, 분, 초로 변환
      timeLeft = {
        days: Math.floor(difference / (1000 * 60 * 60 * 24)),
        hours: Math.floor((difference / (1000 * 60 * 60)) % 24),
        minutes: Math.floor((difference / 1000 / 60) % 60),
        seconds: Math.floor((difference / 1000) % 60),
      }
    }

    return timeLeft
  }

  // 각 시간 단위마다 시각적으로 표시할 JSX 요소 생성
  const timerComponents = Object.keys(timeLeft).map((interval) => {
    if (!timeLeft[interval]) {
      // 해당 시간 단위의 값이 0이면 표시하지 않음
      return null
    }

    return (
      // 0 이 아니라면
      <span className="text-[25px] text-[#475ad2]">
        {timeLeft[interval]} {interval} // 빈칸 하나 만들어서 보기좋게
      </span>
    )
  })

  // 남은 시간이 있는지 확인하고 시간이 없으면 'Time's Up' 메시지를 표시
  return (
    <div>
      {timerComponents.length ? (
        timerComponents
      ) : (
        <span className="text-[25px] text-[red]">Time's Up</span>
      )}
    </div>
  )
}

export default CountDown
```

```jsx
import React, { useEffect, useState } from 'react'
import { useSearchParams } from 'react-router-dom'
import Footer from '../components/Layout/Footer'
import Header from '../components/Layout/Header'
import ProductCard from '../components/Route/ProductCard'
import { productData } from '../static/data'
import styles from '../styles/styles'

const ProductsPage = () => {
  // URL에서 쿼리 파라미터를 가져오는 Hook 사용
  const [searchParams] = useSearchParams()

  // 카테고리 데이터를 상태로 관리
  const categoryData = searchParams.get('category')

  // 상품 데이터를 관리하는 상태 및 초기화
  const [data, setData] = useState([])

  useEffect(() => {
    // 카테고리 데이터가 없는 경우 모든 상품을 판매량순으로 정렬하여 상태 업데이트
    if (categoryData === null) {
      const sortedData = productData && productData.sort((a, b) => a.total_sell - b.total_sell)
      setData(sortedData)
    } else {
      // 카테고리 데이터가 있는 경우 해당 카테고리에 속하는 상품만 필터링하여 상태 업데이트
      const filteredData =
        productData && productData.filter((item) => item.category === categoryData)
      setData(filteredData)
    }
    // 스크롤을 페이지 상단으로 이동하는 코드 (주석 처리)
    // window.scrollTo(0, 0);
  }, [categoryData]) // 카테고리 데이터가 변경될 때마다 useEffect 실행

  return (
    <div>
      {/* 헤더 컴포넌트 렌더링 및 활성화된 제목 지정 */}
      <Header activeHeading={3} />
      <br />
      <br />
      {/* 스타일된 섹션 렌더링 */}
      <div className={`${styles.section}`}>
        {/* 그리드 레이아웃을 사용하여 상품 카드 표시 */}
        <div className="mb-12 grid grid-cols-1 gap-[20px] md:grid-cols-2 md:gap-[25px] lg:grid-cols-4 lg:gap-[25px] xl:grid-cols-5 xl:gap-[30px]">
          {data && data.map((item, index) => <ProductCard data={item} key={index} />)}
        </div>
        {/* 검색 결과가 없는 경우 안내 메시지 표시 */}
        {data && data.length === 0 ? (

        ) : null}
```

```jsx
import { Link } from 'react-router-dom'
import styles from '../../styles/styles'
import { categoriesData, productData } from '../../static/data'
import { AiOutlineHeart, AiOutlineSearch, AiOutlineShoppingCart } from 'react-icons/ai'
import { CgProfile } from 'react-icons/cg'
import { IoIosArrowDown, IoIosArrowForward } from 'react-icons/io'
import { BiMenuAltLeft } from 'react-icons/bi'
import DropDown from './DropDown'
import Navbar from './Navbar'
import { useSelector } from 'react-redux'

const Header = ({ activeHeading }) => {
  // 사용자의 인증 상태 및 정보를 가져오기 위해 Redux에서 상태 추출
  const { isAuthenticated, user } = useSelector((state) => state.user)
  // 검색어 및 검색 결과 상태
  const [searchTerm, setSearchTerm] = useState('')
  const [searchData, setSearchData] = useState(null)
  // 페이지 스크롤 시 헤더의 활성화 여부 상태
  const [active, setActive] = useState(false)
  // 카테고리 드롭다운 상태
  const [dropDown, setDropDown] = useState(false)

  // 검색어가 변경될 때 호출되는 함수
  const handleSearchChange = (e) => {
    const term = e.target.value
    setSearchTerm(term)

    // 제품 데이터에서 검색어와 일치하는 항목 필터링
    const filteredProducts = productData.filter((product) =>
      product.name.toLowerCase().includes(term.toLowerCase())
    )
    setSearchData(filteredProducts)
  }

  // 페이지 스크롤 이벤트 리스너 등록
  window.addEventListener('scroll', () => {
    if (window.scrollY > 70) {
      setActive(true)
    } else {
      setActive(false)
    }
  })

  return (
    <>
      {/* 데스크탑 화면에서 표시되는 헤더 */}
      <div className={`${styles.section}`}>
        <div className="800px:h-[50px] 800px:my-[20px] 800px:flex hidden items-center justify-between">
          {/* 로고 */}
          <div>
            <Link to="/">
              <img src={`http://localhost:5000/images/QuickDealMarketFix.jpg`} alt="" />
            </Link>
          </div>
          {/* 검색창 */}
          <div className="relative w-[50%]">
            <input
              type="text"
              placeholder="Search Product..."
              value={searchTerm}
              onChange={handleSearchChange}
              className="h-[40px] w-full rounded-md border-[2px] border-[#e3bb8b] px-2"
            />
            {/* 검색 아이콘 */}
            <AiOutlineSearch size={30} className="absolute top-1.5 right-2 cursor-pointer" />
            {/* 검색 결과가 있는 경우 표시 */}
            {searchData && searchData.length !== 0 ? (
              <div className="shadow-sm-2 absolute z-[9] min-h-[30vh] bg-slate-50 p-4">
                {searchData &&
                  searchData.map((i, index) => {
                    const d = i.name

                    const Product_name = d.replace(/\s+/g, '-')
                    return (
                      <Link to={`/product/${Product_name}`}>
                        <div className="items-start-py-3 flex w-full">
                          <img
                            src={i.image_Url[0].url}
                            alt=""
                            className="mr-[10px] h-[40px] w-[40px]"
                          />
                          <h1>{i.name}</h1>
                        </div>
                      </Link>
                    )
                  })}
              </div>
            ) : null}
          </div>
          {/* 판매자 등록 버튼 */}
          <div className={`${styles.button}`}>
            <Link to="/seller">
              <h1 className="flex items-center text-[#fff]">
                판매자 등록 <IoIosArrowForward className="ml-1" />
              </h1>
            </Link>
          </div>
        </div>
      </div>
      {/* 모바일 및 태블릿 화면에서 표시되는 헤더 */}
      <div
        className={`${
          active === true ? 'fixed top-0 left-0 z-10 shadow-sm' : null
        } 800px:flex hidden h-[70px] w-full items-center justify-between bg-[#e3bb8b] transition`}
      >
        <div className={`${styles.section} relative ${styles.noramlFlex} justify-between`}>
          {/* 카테고리 드롭다운 */}
          <div onClick={() => setDropDown(!dropDown)}>
            <div className="1000px:block relative mt-[10px] hidden h-[60px] w-[270px]">
              <BiMenuAltLeft size={30} className="absolute top-3 left-2" />
              <button
                className={`flex h-[100%] w-full items-center justify-between rounded-t-md bg-white pl-10 font-sans text-lg font-[500] select-none`}
              >
                All Categories
              </button>
              <IoIosArrowDown
                size={20}
                className="absolute top-4 right-2 cursor-pointer"
                onClick={() => setDropDown(!dropDown)}
              />
              {/* 드롭다운이 열려있는 경우 카테고리 목록 표시 */}
              {dropDown ? (
                <DropDown categoriesData={categoriesData} setDropDown={setDropDown} />
              ) : null}
            </div>
          </div>
          {/* 네비게이션 바 */}
          <div className={`${styles.noramlFlex}`}>
            <Navbar active={activeHeading} />
          </div>
          {/* 사용자 관련 아이콘 */}
          <div className="flex">
            <div className={`${styles.noramlFlex}`}>
              <div className="relative mr-[15px] cursor-pointer">
                {/* 찜하기 아이콘 */}
                <AiOutlineHeart size={30} color="rgb(255 255 255 / 83%)" />
                <span className="top right absolute top-0 right-0 m-0 h-4 w-4 rounded-full bg-[#3bc177] p-0 text-center font-mono text-[12px] leading-tight text-white">
                  0
                </span>
              </div>
            </div>
            <div className={`${styles.noramlFlex}`}>
              <div className="relative mr-[15px] cursor-pointer">
                {/* 장바구니 아이콘 */}
                <AiOutlineShoppingCart size={30} color="rgb(255 255 255 / 83%)" />
                <span className="top right absolute top-0 right-0 m-0 h-4 w-4 rounded-full bg-[#3bc177] p-0 text-center font-mono text-[12px] leading-tight text-white">
                  1
                </span>
              </div>
            </div>
            <div className={`${styles.noramlFlex}`}>
              <div className="relative mr-[15px] cursor-pointer">
                {/* 사용자 프로필 아이콘 */}
                {isAuthenticated ? (
                  <Link to="/profile">
                    <img
                      src={`http://localhost:5000/${user.avatar}`}
                      className="h-[35px] w-[35px] rounded-full"
                      alt=""
                    />
                  </Link>
                ) : (
                  <Link to="/login">
                    <CgProfile size={30} color="rgb(255 255 255 / 83%)" />
                  </Link>
                )}
              </div>
            </div>
          </div>
        </div>
      </div>
    </>
  )
}

export default Header
```

```jsx
// Redux에서 상태를 추출하여 사용자의 인증 상태 및 정보를 가져옵니다.
const { isAuthenticated, user } = useSelector((state) => state.user);

이 코드는 useSelector 훅을 사용하여 Redux 상태에서 사용자의 인증 상태와 정보를 추출하는 부분이다.

useSelector 훅: 이 훅은 Redux 상태에서 특정 값을 추출하는 데 사용된다. useSelector는 Redux의 store에서 전체 상태를 가져오고, 콜백 함수를 통해 필요한 부분을 선택할 수 있게 해준다.

state.user: Redux의 상태 객체 중 user라는 이름의 하위 객체를 선택한다. 이 객체는 보통 사용자 인증 및 관련 정보와 관련이 있다.

{ isAuthenticated, user }: useSelector에서 선택한 state.user 객체에서 isAuthenticated와 user라는 두 개의 속성을 추출한다.

isAuthenticated: 사용자가 현재 인증되었는지 여부를 나타내는 부울 값이다.

user: 인증된 사용자의 정보를 포함한 객체이다.
```
