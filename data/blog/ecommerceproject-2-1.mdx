# ShopCreate 페이지 생성

```jsx
import React, { useEffect } from 'react'
import { useSelector } from 'react-redux'
import { useNavigate } from 'react-router-dom'
import ShopCreate from '../components/Shop/ShopCreate'

// ShopCreatePage 컴포넌트 정의
const ShopCreatePage = () => {
  // react-router-dom의 useNavigate 훅을 이용하여 페이지 이동을 관리
  const navigate = useNavigate()

  // Redux의 useSelector 훅을 사용하여 상태값 가져오기
  const { isSeller, seller } = useSelector((state) => state.seller)

  // 컴포넌트가 마운트되면 실행되는 useEffect 훅
  useEffect(() => {
    // 만약 isSeller 값이 true라면, 이미 판매자로 등록된 경우
    if (isSeller === true) {
      // 사용자를 해당 판매자의 상점 페이지로 이동시킴
      navigate(`/shop/${seller._id}`)
    }
  }, []) // 빈 배열을 전달하여 컴포넌트 마운트 시 한 번만 실행

  // JSX를 반환하여 컴포넌트 구성
  return (
    <div>
      {/* ShopCreate 컴포넌트 렌더링 */}
      <ShopCreate />
    </div>
  )
}

// ShopCreatePage 컴포넌트 내보내기
export default ShopCreatePage
```

## ShopCreate.jsx 컴포넌트 생성

```jsx
import { React, useState } from 'react'
import { AiOutlineEye, AiOutlineEyeInvisible } from 'react-icons/ai'
import styles from '../../styles/styles'
import { Link, useNavigate } from 'react-router-dom'
import axios from 'axios'
import { toast } from 'react-toastify'
import { RxAvatar } from 'react-icons/rx'

// ShopCreate 컴포넌트 정의
const ShopCreate = () => {
  // react-router-dom의 useNavigate 훅을 이용하여 페이지 이동을 관리
  const navigate = useNavigate()

  // State Hook을 이용하여 각 입력 필드의 상태를 관리
  const [email, setEmail] = useState('')
  const [name, setName] = useState('')
  const [phoneNumber, setPhoneNumber] = useState()
  const [address, setAddress] = useState('')
  const [zipCode, setZipCode] = useState()
  const [avatar, setAvatar] = useState()
  const [password, setPassword] = useState('')
  const [visible, setVisible] = useState(false)

  // 폼 제출 핸들러 함수
  const handleSubmit = async (e) => {
    e.preventDefault()

    // 파일 업로드를 위한 헤더 설정
    const config = { headers: { 'Content-Type': 'multipart/form-data' } }

    // FormData 객체를 생성하여 필드 값들을 추가
    const newForm = new FormData()
    newForm.append('file', avatar)
    newForm.append('name', name)
    newForm.append('email', email)
    newForm.append('password', password)
    newForm.append('zipCode', zipCode)
    newForm.append('address', address)
    newForm.append('phoneNumber', phoneNumber)

    // Axios를 사용하여 서버에 POST 요청을 보냄
    axios
      .post(`http://localhost:5000/api/v2/shop/create-shop`, newForm, config)
      .then((res) => {
        // 성공적인 응답일 경우 토스트 메시지를 띄우고 입력값 초기화
        toast.success(res.data.message)
        setName('')
        setEmail('')
        setPassword('')
        setAvatar()
        setZipCode()
        setAddress('')
        setPhoneNumber()
      })
      .catch((error) => {
        // 에러 응답일 경우 토스트 메시지를 띄움
        toast.error(error.response.data.message)
      })
  }

  // 파일 입력 변경 핸들러 함수
  const handleFileInputChange = (e) => {
    // 파일 선택 시 상태 업데이트
    const file = e.target.files[0]
    setAvatar(file)
  }

  // JSX를 반환하여 컴포넌트 구성
  return (
    <div className="flex min-h-screen flex-col justify-center bg-gray-50 py-12 sm:px-6 lg:px-8">
      <div className="sm:mx-auto sm:w-full sm:max-w-md">
        <h2 className="mt-6 text-center text-3xl font-extrabold text-gray-900">판매자 등록</h2>
      </div>
      <div className="mt-8 sm:mx-auto sm:w-full sm:max-w-[35rem]">
        <div className="bg-white px-4 py-8 shadow sm:rounded-lg sm:px-10">
          {/* 폼 엘리먼트 */}
          <form className="space-y-6" onSubmit={handleSubmit}>
            {/* 각 입력 필드에 대한 라벨과 입력 상자 */}
            {/* 이름 입력 필드 */}
            <div>
              <label htmlFor="email" className="block text-sm font-medium text-gray-700">
                가게명
              </label>
              <div className="mt-1">
                <input
                  type="name"
                  name="name"
                  required
                  value={name}
                  onChange={(e) => setName(e.target.value)}
                  className="block w-full appearance-none rounded-md border border-gray-300 px-3 py-2 placeholder-gray-400 shadow-sm focus:border-yellow-500 focus:ring-yellow-500 focus:outline-none sm:text-sm"
                />
              </div>
            </div>

            {/* 휴대폰 번호 입력 필드 */}
            <div>
              <label htmlFor="email" className="block text-sm font-medium text-gray-700">
                휴대폰 번호
              </label>
              <div className="mt-1">
                <input
                  type="number"
                  name="phone-number"
                  required
                  value={phoneNumber}
                  onChange={(e) => setPhoneNumber(e.target.value)}
                  className="block w-full appearance-none rounded-md border border-gray-300 px-3 py-2 placeholder-gray-400 shadow-sm focus:border-yellow-500 focus:ring-yellow-500 focus:outline-none sm:text-sm"
                />
              </div>
            </div>

            {/* 이메일 입력 필드 */}
            <div>
              <label htmlFor="email" className="block text-sm font-medium text-gray-700">
                이메일
              </label>
              <div className="mt-1">
                <input
                  type="email"
                  name="email"
                  autoComplete="email"
                  required
                  value={email}
                  onChange={(e) => setEmail(e.target.value)}
                  className="block w-full appearance-none rounded-md border border-gray-300 px-3 py-2 placeholder-gray-400 shadow-sm focus:border-yellow-500 focus:ring-yellow-500 focus:outline-none sm:text-sm"
                />
              </div>
            </div>

            {/* 주소 입력 필드 */}
            <div>
              <label htmlFor="email" className="block text-sm font-medium text-gray-700">
                주소
              </label>
              <div className="mt-1">
                <input
                  type="address"
                  name="address"
                  required
                  value={address}
                  onChange={(e) => setAddress(e.target.value)}
                  className="block w-full appearance-none rounded-md border border-gray-300 px-3 py-2 placeholder-gray-400 shadow-sm focus:border-yellow-500 focus:ring-yellow-500 focus:outline-none sm:text-sm"
                />
              </div>
            </div>

            {/* 우편번호 입력 필드 */}
            <div>
              <label htmlFor="email" className="block text-sm font-medium text-gray-700">
                우편번호
              </label>
              <div className="mt-1">
                <input
                  type="number"
                  name="zipcode"
                  required
                  value={zipCode}
                  onChange={(e) => setZipCode(e.target.value)}
                  className="block w-full appearance-none rounded-md border border-gray-300 px-3 py-2 placeholder-gray-400 shadow-sm focus:border-yellow-500 focus:ring-yellow-500 focus:outline-none sm:text-sm"
                />
              </div>
            </div>

            {/* 비밀번호 입력 필드 */}
            <div>
              <label htmlFor="password" className="block text-sm font-medium text-gray-700">
                비밀번호
              </label>
              <div className="relative mt-1">
                <input
                  type={visible ? 'text' : 'password'}
                  name="password"
                  autoComplete="current-password"
                  required
                  value={password}
                  onChange={(e) => setPassword(e.target.value)}
                  className="block w-full appearance-none rounded-md border border-gray-300 px-3 py-2 placeholder-gray-400 shadow-sm focus:border-yellow-500 focus:ring-yellow-500 focus:outline-none sm:text-sm"
                />
                {/* 비밀번호 가리기/보이기 아이콘 */}
                {visible ? (
                  <AiOutlineEye
                    className="absolute top-2 right-2 cursor-pointer"
                    size={25}
                    onClick={() => setVisible(false)}
                  />
                ) : (
                  <AiOutlineEyeInvisible
                    className="absolute top-2 right-2 cursor-pointer"
                    size={25}
                    onClick={() => setVisible(true)}
                  />
                )}
              </div>
            </div>

            {/* 프로필 이미지 업로드 */}
            <div>
              <label htmlFor="avatar" className="block text-sm font-medium text-gray-700"></label>
              <div className="mt-2 flex items-center">
                <span className="inline-block h-8 w-8 overflow-hidden rounded-full">
                  {/* 업로드된 이미지 또는 기본 아바타 아이콘 */}
                  {avatar ? (
                    <img
                      src={URL.createObjectURL(avatar)}
                      alt="avatar"
                      className="h-full w-full rounded-full object-cover"
                    />
                  ) : (
                    <RxAvatar className="h-8 w-8" />
                  )}
                </span>
                {/* 파일 업로드 버튼 */}
                <label
                  htmlFor="file-input"
                  className="ml-5 flex items-center justify-center rounded-md border border-gray-300 bg-white px-4 py-2 text-sm font-medium text-gray-700 shadow-sm hover:bg-gray-50"
                >
                  <span>프로필 등록</span>
                  <input
                    type="file"
                    name="avatar"
                    id="file-input"
                    accept=".jpg,.jpeg,.png"
                    onChange={handleFileInputChange}
                    className="sr-only"
                  />
                </label>
              </div>
            </div>

            {/* 제출 버튼 */}
            <div>
              <button
                type="submit"
                className="group relative flex h-[40px] w-full justify-center rounded-md border border-transparent bg-yellow-500 px-4 py-2 text-sm font-medium text-white hover:bg-yellow-600"
              >
                Submit
              </button>
            </div>

            {/* 로그인 링크 */}
            <div className={`${styles.noramlFlex} w-full`}>
              <h4>이미 회원이십니까?</h4>
              <Link to="/shop-login" className="pl-2 text-yellow-600">
                로그인
              </Link>
            </div>
          </form>
        </div>
      </div>
    </div>
  )
}
```

## shop 모델 만들어주기

```jsx
const mongoose = require('mongoose')
const bcrypt = require('bcryptjs')
const jwt = require('jsonwebtoken')

// MongoDB 스키마 정의
const shopSchema = new mongoose.Schema({
  // 가게명
  name: {
    type: String,
    required: [true, '판매처 이름을 입력해 주세요!'],
  },
  // 이메일
  email: {
    type: String,
    required: [true, '판매처 email을 입력해 주세요!'],
  },
  // 비밀번호
  password: {
    type: String,
    required: [true, '비밀번호는 필수입니다!'],
    minLength: [6, '6자리 이상의 비밀번호를 입력해 주세요'],
    select: false, // 조회 시 비밀번호는 제외
  },
  // 가게 소개
  description: {
    type: String,
  },
  // 주소
  address: {
    type: String,
    required: true,
  },
  // 전화번호
  phoneNumber: {
    type: Number,
    required: true,
  },
  // 역할 (기본값: Seller)
  role: {
    type: String,
    default: 'Seller',
  },
  // 프로필 이미지 경로
  avatar: {
    type: String,
    required: true,
  },
  // 우편번호
  zipCode: {
    type: Number,
    required: true,
  },
  // 생성일
  createdAt: {
    type: Date,
    default: Date.now(),
  },
  // 비밀번호 초기화 토큰
  resetPasswordToken: String,
  // 비밀번호 초기화 토큰 유효 기간
  resetPasswordTime: Date,
})

// 비밀번호 해싱 (저장하기 전에 실행)
shopSchema.pre('save', async function (next) {
  if (!this.isModified('password')) {
    next()
  }
  this.password = await bcrypt.hash(this.password, 10)
})

// JWT 토큰 생성 메서드
shopSchema.methods.getJwtToken = function () {
  return jwt.sign({ id: this._id }, process.env.JWT_SECRET_KEY, {
    expiresIn: process.env.JWT_EXPIRES,
  })
}

// 비밀번호 비교 메서드
shopSchema.methods.comparePassword = async function (enteredPassword) {
  return await bcrypt.compare(enteredPassword, this.password)
}

// Shop 모델을 생성하여 내보내기
module.exports = mongoose.model('Shop', shopSchema)
```

```jsx

  resetPasswordToken: String,

  resetPasswordTime: Date,

resetPasswordToken과 resetPasswordTime은 사용자의 비밀번호 초기화 프로세스를 관리하기 위한 추가적인 정보를 나타내는 필드이다.

이러한 정보는 주로 비밀번호 초기화 요청이 발생했을 때 생성되고 사용된다. 따라서 Shop 모델에는 직접적으로 저장되지 않는다.

비밀번호 초기화 프로세스는 대략적으로 다음과 같은 순서로 이루어진다:

사용자가 비밀번호 초기화를 요청하면 서버는 새로운 랜덤한 토큰을 생성하고,

해당 토큰을 resetPasswordToken 필드에 저장하며, resetPasswordTime 필드에는 현재 시간에 유효 기간을 더한 값을 저장한다.

생성된 토큰과 유효 기간이 이메일 등을 통해 사용자에게 전달된다.
```

```jsx
  // 쿠키 설정 옵션
  const options = {
    expires: new Date(Date.now() + 90 * 24 * 60 * 60 * 1000), // 90일 동안 유효
    httpOnly: true, // 클라이언트에서 쿠키에 접근 불가
    sameSite: 'none', // SameSite 속성 설정
    secure: true, // HTTPS 프로토콜에서만 전송
  }

  // 상태 코드 및 응답 데이터 설정 후 전송 ( 토큰 발급 )
  res.status(statusCode).cookie('seller_token', token, options).json({
    success: true,
    user,
    token,
  })
}

module.exports = sendShopToken
```

```jsx
sendShopToken 함수: 사용자의 JWT 토큰을 생성하고 설정된 쿠키 옵션에 따라 쿠키에 저장하며,

클라이언트에 응답하는 함수.

token: 사용자의 JWT 토큰. ( .getJwtToken() 모델에 이 메서드는 저장되어 있다 ) (유저 정보로 jwt 키를 만듬 )

options: 쿠키 설정 옵션 객체. expires로 토큰의 유효 기간을 설정하고, ( 결국 토큰 만료기간, 쿠키 만료기간 둘다있다

httpOnly로 클라이언트에서 쿠키에 직접 접근 불가능하게 설정. sameSite와 secure로 SameSite 및 HTTPS 설정.

res.status().cookie().json(): 상태 코드와 쿠키 설정, 응답 데이터를 클라이언트에 전송.

module.exports: 모듈로 내보내기.
```

```jsx
// ErrorHandler 및 catchAsyncErrors 모듈 및 필요한 모듈들을 불러옵니다.
const ErrorHandler = require('../util/ErrorHandler')
const catchAsyncErrors = require('./catchAsyncErrors')
const jwt = require('jsonwebtoken')
const User = require('../model/user')
const Shop = require('../model/shop')

// 사용자가 인증되었는지 확인하는 미들웨어
exports.isAuthenticated = catchAsyncErrors(async (req, res, next) => {
  // 요청 쿠키에서 토큰을 가져옵니다.
  const { token } = req.cookies

  // 토큰이 없으면 401 오류를 발생시킵니다.
  if (!token) {
    return next(new ErrorHandler('로그인 후 진행해 주세요', 401))
  }

  // 토큰을 해독하여 사용자 ID를 추출합니다.
  const decoded = jwt.verify(token, process.env.JWT_SECRET_KEY)

  // 요청 객체에 사용자 정보를 추가합니다.
  req.user = await User.findById(decoded.id)

  // 다음 미들웨어로 이동합니다.
  next()
})

// 판매자가 인증되었는지 확인하는 미들웨어
exports.isSeller = catchAsyncErrors(async (req, res, next) => {
  // 요청 쿠키에서 판매자 토큰을 가져옵니다.
  const { seller_token } = req.cookies

  // 판매자 토큰이 없으면 401 오류를 발생시킵니다.
  if (!seller_token) {
    return next(new ErrorHandler('로그인 후 진행해 주세요', 401))
  }

  // 판매자 토큰을 해독하여 판매자 ID를 추출합니다.
  const decoded = jwt.verify(seller_token, process.env.JWT_SECRET_KEY)

  // 요청 객체에 판매자 정보를 추가합니다.
  req.seller = await Shop.findById(decoded.id)

  // 다음 미들웨어로 이동합니다.
  next()
})
```

```jsx
isAuthenticated 미들웨어:

이 미들웨어는 사용자가 로그인되어 있는지 확인한다.

요청 쿠키에서 JWT 토큰을 가져와 해독하여 사용자 ID를 추출하고,

해당 ID로 사용자를 데이터베이스에서 찾아 요청 객체에 추가한다. 토큰이 없거나 유효하지 않으면

해당 ID로 판매자를 데이터베이스에서 찾아 요청 객체에 추가한다.

토큰이 없거나 유효하지 않으면 401 오류를 발생시킨다.
```

# SellerProtectedRoute 구현

```jsx
import { Navigate } from 'react-router-dom'

// SellerProtectedRoute 컴포넌트 정의
const SellerProtectedRoute = ({ isSeller, children }) => {
  // isSeller이 false인 경우, 홈페이지로 Redirect
  if (!isSeller) {
    return <Navigate to={`/`} replace />
  }
  // isSeller이 true인 경우, 자식 컴포넌트(children) 렌더링
  return children
}

// SellerProtectedRoute 컴포넌트를 내보내기
export default SellerProtectedRoute
```

```jsx
Navigate: React Router의 Navigate 컴포넌트를 사용하여 페이지 리디렉션을 처리한다.

SellerProtectedRoute: 판매자 전용 라우트를 나타내는 컴포넌트이다.

isSeller prop을 통해 판매자 여부를 확인하고, 그에 따라 페이지를 Redirect하거나 자식 컴포넌트를 렌더링한다.

if (!isSeller): 판매자가 아닌 경우, 홈페이지로 Redirect한다.

return children: 판매자인 경우, 자식 컴포넌트(children)를 렌더링한다.
```
