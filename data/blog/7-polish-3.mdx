```csharp
public class SwordAction : BaseAction
{
```

## 2. 유효한 공격 위치 검증 로직

### 주변 그리드 위치 찾기

검 공격은 바로 옆 칸에만 가능하므로, 유닛 주변의 그리드 위치를 찾아야 합니다.

```csharp
public class SwordAction : BaseAction
{
    private int maxSwordDistance = 1; // 검의 최대 공격 거리 (1칸)

    public override List<GridPosition> GetValidActionGridPositionList()
    {
        List<GridPosition> validGridPositionList = new List<GridPosition>();

        // 현재 유닛의 그리드 위치
        GridPosition unitGridPosition = unit.GetGridPosition();

        // 유닛 주변 8방향 탐색 (상하좌우 + 대각선)
        for (int x = -maxSwordDistance; x <= maxSwordDistance; x++)
        {
            for (int z = -maxSwordDistance; z <= maxSwordDistance; z++)
            {
                // 상대 위치 계산
                GridPosition offsetGridPosition = new GridPosition(x, z);
                // 절대 위치 계산
                GridPosition testGridPosition = unitGridPosition + offsetGridPosition;

                // 1. 유효한 그리드 위치인지 확인 (맵 밖이 아닌지)
                if (!LevelGrid.Instance.IsValidGridPosition(testGridPosition))
                {
                    continue; // 유효하지 않으면 건너뜀
                }

                validGridPositionList.Add(testGridPosition);
            }
        }

        return validGridPositionList;
    }
}
```

### 적 유닛이 있는 위치만 필터링

검으로 공격하려면 그 위치에 적이 있어야 합니다.

```csharp
public override List<GridPosition> GetValidActionGridPositionList()
{
    List<GridPosition> validGridPositionList = new List<GridPosition>();
    GridPosition unitGridPosition = unit.GetGridPosition();

    for (int x = -maxSwordDistance; x <= maxSwordDistance; x++)
    {
        for (int z = -maxSwordDistance; z <= maxSwordDistance; z++)
        {
            GridPosition offsetGridPosition = new GridPosition(x, z);
            GridPosition testGridPosition = unitGridPosition + offsetGridPosition;

            if (!LevelGrid.Instance.IsValidGridPosition(testGridPosition))
            {
                continue;
            }

            // 2. 해당 위치에 유닛이 없으면 건너뜀
            if (!LevelGrid.Instance.HasAnyUnitOnGridPosition(testGridPosition))
            {
                continue;
            }

            // 3. 해당 위치의 유닛 가져오기
            Unit targetUnit = LevelGrid.Instance.GetUnitAtGridPosition(testGridPosition);

            // 4. 같은 팀이면 건너뜀 (아군은 공격 불가)
            if (targetUnit.IsEnemy() == unit.IsEnemy())
            {
                continue;
            }

            // 모든 조건을 통과한 위치만 추가
            validGridPositionList.Add(testGridPosition);
        }
    }

    return validGridPositionList;
}
```

### 검증 로직 흐름

1. **IsValidGridPosition**: 맵 경계 내에 있는지 확인
2. **HasAnyUnitOnGridPosition**: 해당 위치에 유닛이 있는지 확인
3. **GetUnitAtGridPosition**: 유닛 참조 가져오기
4. **IsEnemy 비교**: 적군인지 아군인지 확인

### 정사각형 범위 표시 함수 추가

검의 공격 범위는 원형이 아닌 정사각형이므로 새로운 시각화 함수가 필요합니다.

**GridSystemVisual.cs에 추가:**

```csharp
// 기존: 원형 범위 (거리 계산 포함)
private void ShowGridPositionRange(GridPosition gridPosition, int range, GridVisualType gridVisualType)
{
    List<GridPosition> gridPositionList = new List<GridPosition>();
    for (int x = -range; x <= range; x++)
    {
        for (int z = -range; z <= range; z++)
        {
            GridPosition testGridPosition = gridPosition + new GridPosition(x, z);

            if (!LevelGrid.Instance.IsValidGridPosition(testGridPosition))
            {
                continue;
            }

            // 맨해튼 거리 계산으로 원형 모양 만듦
            int testDistance = Mathf.Abs(x) + Mathf.Abs(z);
            if (testDistance > range)
            {
                continue; // 범위 밖이면 제외
            }

            gridPositionList.Add(testGridPosition);
        }
    }
    ShowGridPositionList(gridPositionList, gridVisualType);
}

// 새로운 함수: 정사각형 범위 (거리 계산 없음)
private void ShowGridPositionRangeSquare(GridPosition gridPosition, int range, GridVisualType gridVisualType)
{
    List<GridPosition> gridPositionList = new List<GridPosition>();
    for (int x = -range; x <= range; x++)
    {
        for (int z = -range; z <= range; z++)
        {
            GridPosition testGridPosition = gridPosition + new GridPosition(x, z);

            if (!LevelGrid.Instance.IsValidGridPosition(testGridPosition))
            {
                continue;
            }

            // 거리 검증 없이 모든 위치 추가 → 정사각형 모양
            gridPositionList.Add(testGridPosition);
        }
    }
    ShowGridPositionList(gridPositionList, gridVisualType);
}
```

### UpdateGridVisual에서 액션별 시각화 처리

```csharp
public void UpdateGridVisual()
{
    HideAllGridPosition(); // 기존 시각화 모두 숨김

    Unit selectedUnit = UnitActionsSystem.Instance.GetSelectedUnit();
    BaseAction selectedAction = UnitActionsSystem.Instance.GetSelectedAction();

    GridVisualType gridVisualType;

    switch (selectedAction)
    {
        case MoveAction moveAction:
            gridVisualType = GridVisualType.White; // 이동은 흰색
            break;

        case ShootAction shootAction:
            gridVisualType = GridVisualType.Red; // 사격은 빨간색
            // 사격 범위를 연한 빨간색으로 표시
            ShowGridPositionRange(
                selectedUnit.GetGridPosition(),
                shootAction.GetMaxShootDistance(),
                GridVisualType.RedSoft
            );
            break;

        case GrenadeAction grenadeAction:
            gridVisualType = GridVisualType.Yellow; // 수류탄은 노란색
            break;

        case SwordAction swordAction:
            gridVisualType = GridVisualType.Red; // 검 공격은 빨간색

            ShowGridPositionRangeSquare(
                selectedUnit.GetGridPosition(),
                swordAction.GetMaxShootDistance(),
                GridVisualType.RedSoft
            );
            break;

        default:
            gridVisualType = GridVisualType.White;
            break;
    }

    // 실제 유효한 공격 위치는 진한 색으로 표시
    ShowGridPositionList(
        selectedAction.GetValidActionGridPositionList(),
        gridVisualType
    );
}
```

## 4. 상태 머신을 이용한 타이밍 제어

### 상태 머신 개념

검 공격은 시간에 따라 단계적으로 진행되어야 합니다:

1. **휘두르기 전** (회전 + 애니메이션)
2. **타격 순간** (데미지 발생)
3. **휘두르기 후** (애니메이션 마무리)

상태 머신을 사용하면 이런 단계를 명확하게 관리할 수 있습니다.

### 상태 정의

```csharp
private enum State
{
    SwingingSwordBeforeHit,  // 타격 전: 검을 휘두르는 중
    SwingingSwordAfterHit    // 타격 후: 검을 거두는 중
}

private State state;          // 현재 상태
private float stateTimer;     // 상태 전환 타이머
private Unit targetUnit;      // 공격할 대상 유닛
```

### TakeAction에서 초기 상태 설정

```csharp
public override void TakeAction(GridPosition gridPosition, Action onActionComplete)
{
    // 공격 대상 저장
    targetUnit = LevelGrid.Instance.GetUnitAtGridPosition(gridPosition);

    // 초기 상태: 타격 전
    state = State.SwingingSwordBeforeHit;
    float beforeHitStateTime = 0.35f; // 타격까지 0.35초
    stateTimer = beforeHitStateTime;

    ActionStart(onActionComplete);
}

```

### Update에서 상태별 처리

```csharp
private void Update()
{
    if (!isActive)
    {
        return; // 액션이 활성화되지 않으면 실행하지 않음
    }

    // 타이머 감소
    stateTimer -= Time.deltaTime;

    // 현재 상태에 따른 처리
    switch (state)
    {
        case State.SwingingSwordBeforeHit:
            // 타격 전: 적을 향해 회전
            Vector3 aimDir = (targetUnit.GetWorldPosition() - unit.GetWorldPosition()).normalized;
            float rotateSpeed = 10f;
            // 부드럽게 회전 (Lerp 사용)
            transform.forward = Vector3.Lerp(transform.forward, aimDir, Time.deltaTime * rotateSpeed);
            break;

        case State.SwingingSwordAfterHit:
            // 타격 후: 특별한 처리 없음 (애니메이션만 재생)
            break;
    }

    // 타이머가 0 이하가 되면 다음 상태로 전환
    if (stateTimer <= 0f)
    {
        NextState();
    }
}

```

### 상태 전환 로직

```csharp
private void NextState()
{
    switch (state)
    {
        case State.SwingingSwordBeforeHit:
            // 타격 전 → 타격 후로 전환
            state = State.SwingingSwordAfterHit;
            float afterHitStateTime = 0.5f; // 타격 후 0.5초 대기
            stateTimer = afterHitStateTime;

            // ★ 이 시점에서 실제 데미지 발생
            targetUnit.Damage(100); // 100의 막대한 데미지
            break;

        case State.SwingingSwordAfterHit:
            // 타격 후 → 액션 완료
            ActionComplete();
            break;
    }
}

```

### 상태 머신 흐름 정리

```
TakeAction 호출
    ↓
SwingingSwordBeforeHit 상태 시작 (0.35초)
    ↓ (회전 중)
타이머 0 도달
    ↓
NextState 호출
    ↓
★ Damage(100) 발생
    ↓
SwingingSwordAfterHit 상태 시작 (0.5초)
    ↓
타이머 0 도달
    ↓
NextState 호출
    ↓
ActionComplete()

```

1. **Polygon Prototype 패키지**의 **Prefabs/Props** 폴더에서 **Sword** 프리팹 찾기

## 4. InteractAction의 유효 위치 검증 로직

### GetValidActionGridPositionList 구현

```csharp
public override List<GridPosition> GetValidActionGridPositionList()
{
    List<GridPosition> validGridPositionList = new List<GridPosition>();

    GridPosition unitGridPosition = unit.GetGridPosition();  // 현재 유닛의 위치

    // maxInteractDistance 범위 내의 모든 칸을 순회
    for (int x = -maxInteractDistance; x <= maxInteractDistance; x++)
    {
        for (int z = -maxInteractDistance; z <= maxInteractDistance; z++)
        {
            GridPosition offsetGridPosition = new GridPosition(x, z);
            GridPosition testGridPosition = unitGridPosition + offsetGridPosition;

            // 1. 그리드 범위를 벗어나는지 확인
            if (!LevelGrid.Instance.IsValidGridPosition(testGridPosition))
            {
                continue;  // 범위 밖이면 건너뜀
            }

            // 2. 해당 위치에 Door가 있는지 확인
            Door door = LevelGrid.Instance.GetDoorAtGridPosition(testGridPosition);

            if (door == null)
            {
                // Door가 없으면 상호작용 불가
                continue;
            }

            // 3. 모든 조건을 만족하면 유효한 위치로 추가
            validGridPositionList.Add(testGridPosition);
        }
    }

    return validGridPositionList;
}
```

**핵심 개념:**

**1. 이중 for 루프로 주변 칸 탐색:**

- `x`와 `z`가 각각 -1 ~ +1 범위를 순회 (maxInteractDistance = 1일 때)
- 총 9칸을 확인 (자신 포함 3x3 영역)

**2. offsetGridPosition의 역할:**

```csharp
// 예시: 유닛이 (5, 5)에 있을 때
// x=-1, z=0 → offsetGridPosition = (-1, 0)
// testGridPosition = (5, 5) + (-1, 0) = (4, 5)  // 왼쪽 칸
```

**3. 검증 순서:**

1. **그리드 범위 체크**: 맵 밖으로 나가지 않는지 확인
2. **Door 존재 체크**: null이면 상호작용 대상 없음
3. **리스트 추가**: 두 조건을 모두 통과해야 유효한 위치

**4. SwordAction과의 차이점:**

```csharp
// SwordAction에서는 Unit이 있는지 확인했음:
Unit targetUnit = LevelGrid.Instance.GetUnitAtGridPosition(testGridPosition);
if (targetUnit == null) continue;

// InteractAction에서는 Door가 있는지 확인:
Door door = LevelGrid.Instance.GetDoorAtGridPosition(testGridPosition);
if (door == null) continue;
```

```csharp
        // 2. LevelGrid에 이 문을 등록

        // 3. 초기 상태에 따라 문 열거나 닫기
        if (isOpen)
        {
            OpenDoor();
        }
        else
        {
            CloseDoor();
        }
    }
}
```

**핵심 개념:**

**1. SerializeField의 활용:**

```csharp
[SerializeField] private bool isOpen;
```

- Inspector에서 체크박스로 표시됨
- 레벨 디자인 시 어떤 문은 열린 상태로, 어떤 문은 닫힌 상태로 시작하도록 설정 가능
- 코드 재컴파일 없이 게임 플레이 조정 가능

```csharp
// 예: transform.position = (8.0, 0, 4.0) → gridPosition = (8, 4)

// ② LevelGrid의 해당 칸에 이 문을 등록

```

### Interact 메서드 - 공개 인터페이스

```csharp
public void Interact(Action onInteractComplete)
{
    this.onInteractComplete = onInteractComplete;  // 완료 콜백 저장
    isActive = true;                               // 상호작용 시작
    timer = .5f;                                   // 0.5초 후 완료

    // 현재 상태의 반대로 전환
    if (isOpen)
    {
        CloseDoor();  // 열려 있으면 닫기
    }
    else
    {
        OpenDoor();   // 닫혀 있으면 열기
    }
}
```

**핵심 개념:**

**1. Action 델리게이트 패턴:**

```csharp
private Action onInteractComplete;
public void Interact(Action onInteractComplete)
{
    this.onInteractComplete = onInteractComplete;
}
```

- `Action`은 C#의 델리게이트 타입 (매개변수 없고 반환값 없는 함수)
- 나중에 `onInteractComplete()` 호출 시 전달받은 함수가 실행됨
- InteractAction이 "문이 다 열리면 알려줘"라고 요청하는 구조

**2. 토글(Toggle) 패턴:**

```csharp
if (isOpen)
    CloseDoor();   // true → false
else
    OpenDoor();    // false → true

```

- 같은 버튼으로 열고 닫기를 반복할 수 있음
- 현재 상태를 확인하여 반대 동작 수행

**3. 타이머 기반 지연:**

```csharp
isActive = true;
timer = 0.5f;

```

- 즉시 완료되지 않고 0.5초 동안 진행 중 상태 유지
- 턴제 게임에서 액션의 타이밍 제어

---

### Update 메서드 - 타이머 처리

```csharp
private void Update()
{
    // 상호작용 중이 아니면 아무것도 안 함
    if (!isActive)
    {
        return;
    }

    // 매 프레임마다 타이머 감소
    timer -= Time.deltaTime;

    // 타이머가 0 이하로 떨어지면 완료 처리
    if (timer <= 0f)
    {
        isActive = false;              // 상호작용 종료
        onInteractComplete();          // 콜백 함수 호출
    }
}

```

**핵심 개념:**

**1. isActive 가드 절:**

```csharp
if (!isActive) return;
```

- 문이 아무 일도 하지 않을 때는 Update 실행을 건너뜀
- CPU 리소스 절약 (매 프레임 불필요한 연산 방지)

**2. Time.deltaTime:**

```csharp
timer -= Time.deltaTime;
```

- `Time.deltaTime`: 이전 프레임과 현재 프레임 사이의 시간 (초 단위)
- 60 FPS에서는 약 0.0167초
- 프레임레이트와 무관하게 일정한 속도로 타이머 감소

**3. 콜백 실행:**

```csharp
onInteractComplete();
```

- 저장해둔 함수(델리게이트)를 호출
- InteractAction의 `OnInteractComplete` 메서드가 실행됨
- 그러면 InteractAction이 `ActionComplete()`를 호출해서 턴 종료

## 8. 비동기 처리와 액션 완료 콜백

### InteractAction의 TakeAction 구현

```csharp
public override void TakeAction(GridPosition gridPosition, Action onActionComplete)
{
    // 1. 해당 그리드 위치의 Door 가져오기
    Door door = LevelGrid.Instance.GetDoorAtGridPosition(gridPosition);

    // 2. Door에게 상호작용 요청 + 완료 시 알려달라고 콜백 전달
    door.Interact(OnInteractComplete);

    // 3. BaseAction에 액션 시작 알림
    ActionStart(onActionComplete);
}

private void OnInteractComplete()
{
    // Door가 상호작용을 완료했을 때 호출됨
    ActionComplete();  // BaseAction의 완료 처리 실행
}

```

**핵심 개념:**

**1. 비동기 패턴 흐름:**

```csharp
// ① UnitActionSystem이 TakeAction 호출 (onActionComplete 전달)
TakeAction(gridPosition, () => { /* 턴 종료 로직 */ });

// ② Door.Interact에 OnInteractComplete 전달
door.Interact(OnInteractComplete);

// ③ ActionStart는 onActionComplete를 저장만 함 (아직 실행 안 함)
ActionStart(onActionComplete);

// ④ 0.5초 후 Door.Update에서 onInteractComplete() 호출
//    → OnInteractComplete() 실행

// ⑤ OnInteractComplete에서 ActionComplete() 호출
//    → BaseAction이 저장해둔 onActionComplete() 실행
//    → 턴 종료

```

**2. 콜백 체인 (Callback Chain):**

```
UnitActionSystem
    ↓ onActionComplete 전달
InteractAction.TakeAction
    ↓ OnInteractComplete 전달
Door.Interact
    ↓ 0.5초 대기
Door.Update
    ↓ onInteractComplete() 호출
InteractAction.OnInteractComplete
    ↓ ActionComplete() 호출
BaseAction
    ↓ 저장된 onActionComplete() 호출
UnitActionSystem (턴 종료)

```

**3. Update 메서드의 역할 제거:**

```csharp
private void Update()
{
    if (!isActive)
    {
        return;
    }
    // 이전에는 여기서 ActionComplete() 호출했지만
    // 이제는 Door가 완료되면 OnInteractComplete()를 통해 호출됨
}

```
