```java
public class Car {

//필드
int gas;

//메소드
voit set Speed(int speed) {내용}
}

//호출
Car myCar = new Car(); //객체 호출
myCar.gas = 10;        // 호출뒤 사용할수 있는 필드 = 인스턴스 필드
myCar.setSpeed(60);    // 호출뒤 사용할수 있는 메소드 = 인스턴스 메소드
                       // 메소드는 객체 안에 저장 되는것은 아니지만 객체가 있어야 호출가능
                       // 그렇기 때문에 인스턴스 멤버에 포함

Car yourCar = new Car();
yourCar.gas = 20;
yourCar.setSpeed(80);

// 인스턴스 멤버 = 객체를 생성한 후 사용할 수 있는 필드와 메소드
```

```java
// 앞에서 설명했던 this 랑 같은것

Car(String model) {           // 인스턴스 멤버 (필드) this.model 에 접근하기 위해 사용
	this.model = model;         // 필드의 this.model 와 매개변수 model 을 구분하기 위해 사용
}

void setModel(String model) {
	this.model = model;
}
```
