// Comment 컴포넌트는 각각의 댓글을 표시합니다.
const Comment = ({ comment, onReplayClick, onDeleteComment }) => (

      <p className="mt-2.5">
        {new Date(comment.regDate).toLocaleString()}
        {/* 답글 버튼을 클릭하면 onReplayClick 함수를 호출합니다. */}

          onClick={() => onReplayClick(comment.id)}
        >
          Replay
        </button>
        {onDeleteComment && (
          // 삭제 버튼을 클릭하면 onDeleteComment 함수를 호출합니다.
          <button

            className="ml-4 text-red-500"
            onClick={(e) => {
              e.preventDefault(); // 폼 제출을 방지합니다.
              console.log("Delete button clicked");
              onDeleteComment(comment.id);
            }}
          >
            Delete
          </button>
        )}
      </p>
      {/* 댓글의 내용을 표시합니다. */}
      <p className="mt-5">{comment.content}</p>
    </div>
  </div>

// CommentSection 컴포넌트는 댓글과 답글을 관리하고 표시합니다.
const CommentSection = ({
  comments,
  onReplayClick,
  selectedCommentId,
  showReplyInput,
  replyContent,
  setReplyContent,
  handleCancelReply,
  handleReplySubmit,
  onDeleteComment,
}) => (
  <div className="comments">
    {/* 댓글 섹션의 제목을 표시합니다. */}
    <h3 className="h5 inline-block border-b-[3px] border-primary font-primary font-medium leading-8">
      댓글
    </h3>
    {/* 각 댓글을 순회하면서 Comment 컴포넌트를 렌더링합니다. */}
    {comments.map((comment) => (
      <div key={comment.id}>
        <Comment
          comment={comment}
          onReplayClick={onReplayClick}
          onDeleteComment={onDeleteComment}
        />
        {/* 선택된 댓글이 있고, 답글 입력 상태인 경우에 답글을 작성할 수 있는 입력창을 표시합니다. */}
        {selectedCommentId === comment.id && showReplyInput && (
          <div>
            {/* 답글을 입력할 수 있는 텍스트 에어리어를 표시합니다. */}
            <textarea
              cols="30"
              rows="3"
              value={replyContent}
              onChange={(e) => setReplyContent(e.target.value)}
              placeholder="답글을 입력하세요..."
            />
            {/* 답글 작성 취소 버튼 및 답글 작성 버튼을 표시합니다. */}
            <button onClick={handleCancelReply}>취소</button>
            <button onClick={() => handleReplySubmit(comment.id)}>
              답글 작성
            </button>
          </div>
        )}
        {/* 해당 댓글의 답글을 재귀적으로 렌더링합니다. */}
        {renderReplies(
          comment.children,
          onReplayClick,
          selectedCommentId,
          showReplyInput,
          replyContent,
          setReplyContent,
          handleCancelReply,
          handleReplySubmit,
          onReplayClick
        )}
      </div>
    ))}
  </div>
);

const NoticeDetailComponent = () => {
  // URL 파라미터에서 noticeId를 추출합니다.
  const { noticeId } = useParams();
  // 공지사항 정보를 저장할 상태 변수를 선언합니다.
  const [notice, setNotice] = useState({});
  // 댓글 정보를 저장할 상태 변수를 선언합니다.
  const [comments, setComments] = useState([]);
  // 새로운 댓글의 내용을 저장할 상태 변수를 선언합니다.
  const [newCommentContent, setNewCommentContent] = useState("");
  // 새로운 댓글의 부모 댓글 ID를 저장할 상태 변수를 선언합니다.
  const [newCommentParentId, setNewCommentParentId] = useState("");
  // 답글 입력 상태를 저장할 상태 변수를 선언합니다.
  const [showReplyInput, setShowReplyInput] = useState(false);
  // 답글 내용을 저장할 상태 변수를 선언합니다.
  const [replyContent, setReplyContent] = useState("");
  // 선택된 댓글 ID를 저장할 상태 변수를 선언합니다.
  const [selectedCommentId, setSelectedCommentId] = useState(null);
  // React Router의 useNavigate 훅을 사용하여 페이지 이동 함수를 받아옵니다.
  const navigate = useNavigate();
  // 사용자 정보를 Redux의 상태에서 가져옵니다.
  const { user } = useSelector((state) => state.user);

  // 페이지 로드 시, 공지사항 데이터와 댓글 데이터를 가져오기 위한 useEffect 훅을 사용합니다.
  useEffect(() => {
    // 공지사항 데이터를 가져오는 비동기 함수를 정의합니다.
    const fetchNoticeData = async () => {
      try {
        // 서버에서 공지사항 데이터를 가져옵니다.
        const response = await axios.get(
          process.env.REACT_APP_SERVER + `/v1/notices/${noticeId}`
        );
        // 가져온 데이터를 상태에 저장합니다.
        setNotice(response.data);
        console.log(response.data);
      } catch (error) {
        console.error("데이터를 불러오는 중 에러 발생:", error);
      }
    };

    // 댓글 데이터를 가져오는 비동기 함수를 정의합니다.
    const fetchCommentData = async () => {
      try {
        // 서버에서 댓글 데이터를 가져옵니다.
        const response = await axios.get(
          process.env.REACT_APP_SERVER + `/v1/comment/${noticeId}`
        );
        // 가져온 데이터를 상태에 저장합니다.
        setComments(response.data);
      } catch (error) {
        console.error("댓글을 불러오는 중 에러 발생:", error);
      }
    };

    // 정의한 비동기 함수들을 실행합니다.
    fetchNoticeData();
    fetchCommentData();
  }, [noticeId]);

  // 공지사항 삭제를 처리하는 함수입니다.
  const handleDeleteNotice = async () => {
    try {
      // 로컬 스토리지에서 액세스 토큰을 가져옵니다.
      const accessToken = localStorage.getItem("accessToken");

      // 사용자가 공지사항 작성자인 경우에만 삭제를 허용합니다.
      if (user && user.id === notice.writer) {
        // 서버에 공지사항 삭제를 요청합니다.
        const response = await axios.delete(
          process.env.REACT_APP_SERVER + `/v1/notices/${noticeId}`,
          {
            headers: {
              Authorization: accessToken,
            },
          }
        );

        console.log(response.data);
        // 성공 메시지를 토스트로 표시하고, 목록 페이지로 이동합니다.
        toast.success(response.data.returnMessage);
        navigate("/notice");
      } else {
        // 작성자가 아닌 경우 삭제 권한이 없음을 알리는 에러 메시지를 토스트로 표시합니다.
        toast.error("글을 삭제할 권한이 없습니다.");
      }
    } catch (error) {
      console.error("글 삭제 중 에러 발생:", error);
      // 에러 메시지를 토스트로 표시합니다.
      toast.error(error.response.data.message);
    }
  };

            <form onSubmit={handleCommentSubmit}>
              <div className="mb-6">

              <div className="mb-4">
                <textarea

                  value={newCommentContent}

                  placeholder="댓글을 작성하세요..."
                  className="w-full p-4 border border-border rounded-md"
                  required
                />
              </div>
              {/* 댓글 작성 버튼을 표시합니다. */}
              <button

                댓글 작성

          </div>
        )}
      </div>
    </section>
  );
};
