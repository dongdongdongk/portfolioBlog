  const { user, error, successMessage } = useSelector((state) => state.user)

const AllOrders = () => {

      field: 'status',
      headerName: '상태',
      minWidth: 130,
      flex: 0.7,
      cellClassName: (params) => {
        return params.getValue(params.id, 'status') === '배송 완료' ? 'greenColor' : 'redColor'
      },
    },
    { field: 'itemsQty', headerName: '상품 수량', type: 'number', minWidth: 130, flex: 0.7 },
    { field: 'total', headerName: '총액', type: 'number', minWidth: 130, flex: 0.8 },

      flex: 1,

              <Button>
                <AiOutlineArrowRight size={20} />
              </Button>
            </Link>
          </>

// 비밀번호 변경 컴포넌트
const ChangePassword = () => {
  const [oldPassword, setOldPassword] = useState("");
  const [newPassword, setNewPassword] = useState("");
  const [confirmPassword, setConfirmPassword] = useState("");

  // 비밀번호 변경 요청 핸들러
  const passwordChangeHandler = async (e) => {
    e.preventDefault();

    await axios
      .put(
        `http://localhost:5000/api/v2/user/update-user-password`,
        { oldPassword, newPassword, confirmPassword },
        { withCredentials: true }
      )
      .then((res) => {
        toast.success(res.data.message);
        setOldPassword("");
        setNewPassword("");
        setConfirmPassword("");
      })
      .catch((error) => {
        toast.error(error.response.data.message);
      });
  };
  return (
    <div className="w-full px-5">
      <h1 className="block text-[25px] text-center font-[600] text-[#000000ba] pb-2">
        비밀번호 변경
      </h1>
      <div className="w-full">
        <form
          aria-required
          onSubmit={passwordChangeHandler}
          className="flex flex-col items-center"
        >
          <div className=" w-[100%] 800px:w-[50%] mt-5">
            <label className="block pb-2">기존 비밀번호 입력</label>
            <input
              type="password"
              className={`${styles.input} !w-[95%] mb-4 800px:mb-0`}
              required
              value={oldPassword}
              onChange={(e) => setOldPassword(e.target.value)}
            />
          </div>
          <div className=" w-[100%] 800px:w-[50%] mt-2">
            <label className="block pb-2">새 비밀번호 입력</label>
            <input
              type="password"
              className={`${styles.input} !w-[95%] mb-4 800px:mb-0`}
              required
              value={newPassword}
              onChange={(e) => setNewPassword(e.target.value)}
            />
          </div>
          <div className=" w-[100%] 800px:w-[50%] mt-2">
            <label className="block pb-2">새 비밀번호 확인</label>
            <input
              type="password"
              className={`${styles.input} !w-[95%] mb-4 800px:mb-0`}
              required
              value={confirmPassword}
              onChange={(e) => setConfirmPassword(e.target.value)}
            />
            <input
              className={`w-[95%] h-[40px] border border-[#3a24db] text-center text-[#3a24db] rounded-[3px] mt-8 cursor-pointer`}
              required
              value="수정"
              type="submit"
            />
          </div>
        </form>
      </div>
    </div>
  );
};

// 사용자 주소 관리 컴포넌트
const Address = () => {
  const [open, setOpen] = useState(false);
  const [country, setCountry] = useState("");
  const [city, setCity] = useState("");
  const [zipCode, setZipCode] = useState();
  const [address1, setAddress1] = useState("");
  const [address2, setAddress2] = useState("");
  const [addressType, setAddressType] = useState("");
  const { user } = useSelector((state) => state.user);
  const dispatch = useDispatch();

  // 주소 유형 데이터
  const addressTypeData = [
    { name: "기본" },
    { name: "집" },
    { name: "회사" },
  ];

  // 주소 추가/수정 제출 핸들러
  const handleSubmit = async (e) => {
    e.preventDefault();

    if (addressType === "" || country === "" || city === "") {
      toast.error("모든 필드를 입력하세요!");
    } else {
      console.log(country, city, address1, address2, zipCode, addressType);
      dispatch(
        updateUserAddress({
          country,
          city,
          address1,
          address2,
          zipCode,
          addressType,
        })
      );
      setOpen(false);
      setCountry("");
      setCity("");
      setAddress1("");
      setAddress2("");
      setZipCode(null);
      setAddressType("");
    }
  };

  // 주소 삭제 핸들러
  const handleDelete = (item) => {
    const id = item._id;
    dispatch(deleteUserAddress(id));
  };

  // Redux의 useSelector 훅을 사용하여 상점 정보를 가져옴
  const { seller } = useSelector((state) => state.seller)

  // 상태 변수들을 초기화
  const [avatar, setAvatar] = useState()
  const [name, setName] = useState(seller && seller.name)
  const [description, setDescription] = useState(
    seller && seller.description ? seller.description : ''
  )
  const [address, setAddress] = useState(seller && seller.address)
  const [phoneNumber, setPhoneNumber] = useState(seller && seller.phoneNumber)
  const [zipCode, setZipcode] = useState(seller && seller.zipCode)

  const dispatch = useDispatch()

  // 이미지 업로드 핸들러
  const handleImage = async (e) => {
    e.preventDefault()
    const file = e.target.files[0]
    setAvatar(file)

    // 이미지를 포함한 FormData 객체 생성
    const formData = new FormData()
    formData.append('image', e.target.files[0])

    try {
      // axios를 사용하여 이미지를 업로드
      await axios.put(`http://localhost:5000/api/v2/shop/update-shop-avatar`, formData, {
        headers: {
          'Content-Type': 'multipart/form-data',
        },
        withCredentials: true,
      })

      // Redux 상태 업데이트를 위해 loadSeller 액션 호출
      dispatch(loadSeller())
      // 성공 메시지 표시
      toast.success('아바타가 성공적으로 업데이트되었습니다!')

      // 에러 메시지 표시
      toast.error(error.response.data.message)

  const updateHandler = async (e) => {
    e.preventDefault()

    try {
      // axios를 사용하여 상점 정보를 업데이트
      await axios.put(
        `http://localhost:5000/api/v2/shop/update-seller-info`,
        {
          name,
          address,
          zipCode,
          phoneNumber,
          description,
        },
        { withCredentials: true }
      )

      // Redux 상태 업데이트를 위해 loadSeller 액션 호출
      dispatch(loadSeller())
      // 성공 메시지 표시
      toast.success('상점 정보가 성공적으로 업데이트되었습니다!')
    } catch (error) {
      // 에러 메시지 표시
      toast.error(error.response.data.message)
    }
  }

  return (
    <div className="flex min-h-screen w-full flex-col items-center">

        <div className="flex w-full items-center justify-center">
          <div className="relative">

## WithdrawMoney.jsx

```jsx
import React, { useEffect, useState } from 'react'
import { useDispatch, useSelector } from 'react-redux'
import { getAllOrdersOfShop } from '../../redux/order/orderAction' // 주문 액션을 가져옵니다.
import styles from '../../styles/styles' // 스타일을 가져옵니다.

// 출금 화면 컴포넌트
const WithdrawMoney = () => {
  const dispatch = useDispatch() // useDispatch 훅을 사용하여 디스패치 함수를 가져옵니다.
  const { orders } = useSelector((state) => state.orders) // Redux 스토어에서 주문 정보를 가져옵니다.
  const { seller } = useSelector((state) => state.seller) // Redux 스토어에서 판매자 정보를 가져옵니다.
  const [deliveredOrder, setDeliveredOrder] = useState(null) // 배송 완료된 주문을 저장할 상태를 선언합니다.

  useEffect(() => {
    dispatch(getAllOrdersOfShop(seller._id)) // 해당 판매자의 모든 주문을 가져오는 액션을 디스패치합니다.

    // 주문 중 배송 완료된 주문만 필터링하여 상태에 저장합니다.
    const orderData = orders && orders.filter((item) => item.status === '배송 완료')
    setDeliveredOrder(orderData)
  }, [dispatch]) // 디스패치 함수가 업데이트될 때마다 useEffect가 실행됩니다.

  // 세금 제외 수입 계산
  const totalEarningWithoutTax =
    deliveredOrder && deliveredOrder.reduce((acc, item) => acc + item.totalPrice, 0)

  // 서비스 수수료 계산
  const serviceCharge = totalEarningWithoutTax * 0.1

  // 사용 가능 잔액 계산
  const availableBalance = (totalEarningWithoutTax - serviceCharge).toFixed(2)

  return (
    <div className="h-[90vh] w-full p-8">
      <div className="flex h-full w-full flex-col items-center justify-center rounded bg-white">
        {/* 사용 가능한 잔액을 화면에 표시합니다. */}
        <h5 className="pb-4 text-[20px]">사용 가능한 잔액: ${availableBalance}</h5>
        {/* 출금하기 버튼을 화면에 표시합니다. */}
        <div className={`${styles.button} !h-[42px] !rounded text-white`}>출금하기</div>
      </div>
    </div>
  )
}

export default WithdrawMoney // 출금 화면 컴포넌트를 내보냅니다.
```

```jsx
reduce() 메서드는 배열의 각 요소에 대해 주어진 콜백 함수를 실행하여 하나의 결과값을 반환

reduce() 메서드는 초기값 0을 가지는 누산기 acc와 배열의 각 요소 item에 대해 실행된다.

각 주문 항목의 totalPrice를 acc에 더해나간다.

최종적으로 모든 항목이 처리된 후 누산기 acc에는 모든 주문의 총 가격이 합산되어 있다.
```
