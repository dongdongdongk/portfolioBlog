const checkAll = document.getElementById('checkAll') // chckAll변수에 "chckAll" 이라는 아이디로 가져오자
const checks = document.getElementsByClassName('checks')
const agree = document.getElementById('agree')

agree.addEventListener('click', function () {
  // agree id의 버튼
  if (checkAll.checked) {
    // chekcAll.checked 속성이 true 라면
    //alert('모두 동의함');
    location.href = './memberJoin' // location.href 기능으로 memberJoin 으로 이동해준다
  } else {
    alert('모든 약관에 동의 해야 합니다') // 거짓이라면 alert 기능을 활용하여 경고창을 띄운다
  }
})

checkAll.addEventListener('click', function () {
  for (let i = 0; i < checks.length; i++) {
    //checks 는 여러개니 배열로 들어온다. checks.length 로 길이 구하고 반복
    checks[i].checked = checkAll.checked //checks[i].checked 에 각각   checkAll.checked (checked 는 체크박스의 속성) 의 속성 값을 넣어주어 checkAll 을 체크하면 모두 체크가 되게한다.
  }
})

for (let i = 0; i < checks.length; i++) {
  // 모든 checks 체크 확인을 위해 위쪽에 반복문
  checks[i].addEventListener('click', function () {
    //checks[i] 가 모두 true 라면 checkAll에 true(check 넣어주기)
    let result = true // 변수 result (boolean) = true 넣기
    for (let j = 0; j < checks.length; j++) {
      // 만약 checks[j] 가 체크가 아니라면
      if (!checks[j].checked) {
        //result = checks[j].checked;
        //result=false;
        result = !result // result 는 !result = !ture 는 false
        break // 반복문 빠져나가기
      }
    }

    checkAll.checked = result // 결과 chckAll.checked 는  result 의 값 true 또는 false 가 들어간다
  })
}

result.addEventListener('click', function (e) {
  // 부모 result(div) 에 클릭 속성을 부여하고 이를 클릭하면 current target 과 그냥 target이 있다 지금 필요한건 target (DELETE 버튼클릭) 추가로 이벤트 객체를 받을 수 있는 매개변수 e를 설정해준다.(click 이 e로 들어간다)

    let id = e.target.getAttribute('data-delete-id') // click id 에 타겟의 data-delete-id 속성을 부여한다 (각각 다른 id를 부여해야하기 때문에)

	let todayFormat = year + '-' + month  + '-' + day;

	let minutes = ('0' + today.getMinutes()).slice(-2);
	let seconds = ('0' + today.getSeconds()).slice(-2);

	let timeFormatResult = year + '-' + month  + '-' + day + hours + ':' + minutes  + ':' + seconds;

const formEl = document.querySelector('.form')

// list에는 이전에 저장한 할 일 목록이 JavaScript 객체 배열의 형태로 저장됨

function toDoList(work) {
  let newWork = inputEl.value
  if (work) {
    newWork = work.name
  }
  const liEl = document.createElement('li')
  if (work && work.checked) {
    liEl.classList.add('checked')
  }
  liEl.innerText = newWork
  ulEl.appendChild(liEl)
  inputEl.value = ''

  const checkBtnEl = document.createElement('div')
  checkBtnEl.innerHTML = `<i class="fa-regular fa-calendar-check">`
  liEl.appendChild(checkBtnEl)

  const trashBrnEl = document.createElement('div')
  trashBrnEl.innerHTML = `<i class="fa-solid fa-trash">`
  liEl.appendChild(trashBrnEl)

  checkBtnEl.addEventListener('click', () => {
    liEl.classList.toggle('checked')
    // classList.toggle("checked") liEl 요소에 checked 가 있으면 추가하고 없으면 삭제 toggle
    updateLocalStorage()
  })

  trashBrnEl.addEventListener('click', () => {
    liEl.remove()
    updateLocalStorage()
  })
  updateLocalStorage()
}

//로컬 스토리지에 저장 하는기능
function updateLocalStorage() {
  const liEls = document.querySelectorAll('li') // 모든 li 요소를 선택 이를 liEls 변수에 할당
  let list = [] // 변수를 빈 배열로 초기화 이 배열은 로컬 스토리지에 저장할 데이터를 담는 데 사용
  liEls.forEach((liEl) => {
    list.push({
      name: liEl.innerText, // name 에 현제 li 의 텍스트를 넣음
      checked: liEl.classList.contains('checked'), // class 중에 checked 가 있는지 확인 있으면 true checked 에 저장
    })
  })
  localStorage.setItem('list', JSON.stringify(list)) // 로컬 스토리지에 "list"라는 키로 데이터를 저장
  // 웹 페이지를 다시 열거나 새로 고침할 때에도 "list" 키로 저장된 JSON 문자열을 로컬 스토리지에서 검색하여 이전 데이터를 복원할 수 있음
  // JSON.stringify() 함수를 사용하여 데이터를 문자열로 변환하는 이유는 로컬 스토리지에는 문자열 형태로만 데이터를 저장할 수 있기 때문
}
