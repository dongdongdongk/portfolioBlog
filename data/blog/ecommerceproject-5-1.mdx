## AllOrders.jsx 컴포넌트 작성

```jsx
import { Button } from '@material-ui/core'
import { DataGrid } from '@material-ui/data-grid'
import React, { useEffect } from 'react'
import { useDispatch, useSelector } from 'react-redux'
import { Link } from 'react-router-dom'
import Loader from '../Layout/Loader'
import { getAllOrdersOfShop } from '../../redux/order/orderAction'
import { AiOutlineArrowRight } from 'react-icons/ai'

const AllOrders = () => {
  const { orders, isLoading } = useSelector((state) => state.orders)
  const { seller } = useSelector((state) => state.seller)

  const dispatch = useDispatch()

  useEffect(() => {
    dispatch(getAllOrdersOfShop(seller._id))
  }, [dispatch])

  const columns = [
    { field: 'id', headerName: '주문 ID', minWidth: 150, flex: 0.7 },

    {
      field: 'status',
      headerName: '상태',
      minWidth: 130,
      flex: 0.7,
      cellClassName: (params) => {
        return params.getValue(params.id, 'status') === '배송 완료' ? 'greenColor' : 'redColor'
      },
    },
    {
      field: 'itemsQty',
      headerName: '상품 수량',
      type: 'number',
      minWidth: 130,
      flex: 0.7,
    },

    {
      field: 'total',
      headerName: '총액',
      type: 'number',
      minWidth: 130,
      flex: 0.8,
    },

    {
      field: ' ',
      flex: 1,
      minWidth: 150,
      headerName: '',
      type: 'number',
      sortable: false,
      renderCell: (params) => {
        return (
          <>
            <Link to={`/order/${params.id}`}>
              <Button>
                <AiOutlineArrowRight size={20} />
              </Button>
            </Link>
          </>
        )
      },
    },
  ]

  const row = []

  orders &&
    orders.forEach((item) => {
      row.push({
        id: item._id,
        itemsQty: item.cart.length,
        total: item.totalPrice + '원',
        status: item.status,
      })
    })

  return (
    <>
      {isLoading ? (
        <Loader />
      ) : (
        <div className="mx-8 mt-10 w-full bg-white pt-1">
          <DataGrid rows={row} columns={columns} pageSize={10} disableSelectionOnClick autoHeight />
        </div>
      )}
    </>
  )
}

export default AllOrders
```

## OrderDetails.jsx 작성

```jsx
import React, { useEffect, useState } from 'react'
import styles from '../../styles/styles'
import { BsFillBagFill } from 'react-icons/bs'
import { Link, useNavigate, useParams } from 'react-router-dom'
import { useDispatch, useSelector } from 'react-redux'
import { getAllOrdersOfShop } from '../../redux/order/orderAction'
import axios from 'axios'
import { toast } from 'react-toastify'

const OrderDetails = () => {
  // Redux에서 필요한 상태 및 액션을 가져오기
  const { orders, isLoading } = useSelector((state) => state.orders)

  const [status, setStatus] = useState('') // 주문 상태 업데이트를 위한 상태
  const navigate = useNavigate()

  // React Router의 useParams를 사용하여 URL에서 주문 ID 가져오기
  const { id } = useParams()

  // 페이지 로드 시 판매자의 모든 주문을 가져오는 액션 디스패치
  useEffect(() => {
    dispatch(getAllOrdersOfShop(seller._id))
  }, [dispatch])

  // 선택된 주문의 상세 정보 가져오기
  const data = orders && orders.find((item) => item._id === id)

  // 주문 상태를 업데이트하는 핸들러
  const orderUpdateHandler = async () => {
    try {
      // 주문 상태 업데이트를 서버에 요청
      await axios.put(
        `http://localhost:5000/api/v2/order/update-order-status/${id}`,
        { status },
        { withCredentials: true }
      )
      toast.success('주문이 업데이트되었습니다!')
      navigate('/dashboard-orders')
    } catch (error) {
      toast.error(error.response.data.message)
    }
  }

  // 환불 주문 상태를 업데이트하는 핸들러
  const refundOrderUpdateHandler = async () => {
    try {
      // 환불 주문 상태 업데이트를 서버에 요청
      await axios.put(
        `http://localhost:5000/api/v2/order/order-refund-success/${id}`,
        { status },
        { withCredentials: true }
      )
      toast.success('주문이 업데이트되었습니다!')
      dispatch(getAllOrdersOfShop(seller._id))
    } catch (error) {
      toast.error(error.response.data.message)
    }
  }

  return (
    <div className={`min-h-screen py-4 ${styles.section}`}>
      {/* 페이지 상단의 주문 정보 및 목록 이동 링크 */}
      <div className="flex w-full items-center justify-between">
        <div className="flex items-center">
          <BsFillBagFill size={30} color="crimson" />
          <h1 className="pl-2 text-[25px]">주문 상세 정보</h1>
        </div>
        <Link to="/dashboard-orders">
          <div
            className={`${styles.button} !h-[45px] !rounded-[4px] !bg-[#fce1e6] text-[18px] font-[600] text-[#e94560]`}
          >
            주문 목록
          </div>
        </Link>
      </div>

      {/* 주문 ID 및 주문일 표시 */}
      <div className="flex w-full items-center justify-between pt-6">
        <h5 className="text-[#00000084]">
          주문 ID: <span>#{data?._id?.slice(0, 8)}</span>
        </h5>
        <h5 className="text-[#00000084]">
          주문일: <span>{data?.createdAt?.slice(0, 10)}</span>
        </h5>
      </div>

      {/* 주문 상품 목록 표시 */}
      <br />
      <br />
      {data &&
        data?.cart.map((item, index) => (
          <div className="mb-5 flex w-full items-start" key={index}>
            <img
              src={`http://localhost:5000/${item.images[0]}`}
              alt=""
              className="h-[80px] w-[80x]"
            />
            <div className="w-full">
              <h5 className="pl-3 text-[20px]">{item.name}</h5>
              <h5 className="pl-3 text-[20px] text-[#00000091]">
                {item.discountPrice} x {item.qty} 원
              </h5>
            </div>
          </div>
        ))}

      {/* 총 금액 표시 */}
      <div className="w-full border-t text-right">
        <h5 className="pt-3 text-[18px]">
          총 금액: <strong>{data?.totalPrice} 원</strong>
        </h5>
      </div>
      <br />
      <br />

      {/* 배송 주소 및 결제 정보 표시 */}
      <div className="800px:flex w-full items-center">
        <div className="800px:w-[60%] w-full">
          <h4 className="pt-3 text-[20px] font-[600]">배송 주소:</h4>
          <h4 className="pt-3 text-[20px]">
            {data?.shippingAddress.address1 + ' ' + data?.shippingAddress.address2}
          </h4>
          <h4 className="text-[20px]">{data?.shippingAddress.country}</h4>
          <h4 className="text-[20px]">{data?.shippingAddress.city}</h4>
          <h4 className="text-[20px]">{data?.user?.phoneNumber}</h4>
        </div>
        <div className="800px:w-[40%] w-full">
          <h4 className="pt-3 text-[20px]">결제 정보:</h4>
          <h4>Status: {data?.paymentInfo?.status ? data?.paymentInfo?.status : '미결제'}</h4>
        </div>
      </div>
      <br />
      <br />

      {/* 주문 상태 업데이트를 위한 셀렉트 박스 */}
      <h4 className="pt-3 text-[20px] font-[600]">주문 상태:</h4>
      {data?.status !== '환불 처리 중' && data?.status !== '환불 완료' && (
        <select
          value={status}
          onChange={(e) => setStatus(e.target.value)}
          className="mt-2 h-[35px] w-[200px] rounded-[5px] border"
        >
          {['처리 중', '배송 파트너로 이관됨', '발송 중', '수령함', '배송 중', '배송 완료']
            .slice(
              [
                '처리 중',
                '배송 파트너로 이관됨',
                '발송 중',
                '수령함',
                '배송 중',
                '배송 완료',
              ].indexOf(data?.status)
            )
            .map((option, index) => (
              <option value={option} key={index}>
                {option}
              </option>
            ))}
        </select>
      )}

      {/* 환불 상태 업데이트를 위한 셀렉트 박스 */}
      {data?.status === '환불 처리 중' || data?.status === '환불 완료' ? (
        <select
          value={status}
          onChange={(e) => setStatus(e.target.value)}
          className="mt-2 h-[35px] w-[200px] rounded-[5px] border"
        >
          {['환불 처리 중', '환불 완료']
            .slice(['환불 처리 중', '환불 완료'].indexOf(data?.status))
            .map((option, index) => (
              <option value={option} key={index}>
                {option}
              </option>
            ))}
        </select>
      ) : null}

      {/* 주문 상태 업데이트 버튼 */}
      <div
        className={`${styles.button} mt-5 !h-[45px] !rounded-[4px] !bg-[#FCE1E6] text-[18px] font-[600] text-[#E94560]`}
        onClick={data?.status !== '환불 처리 중' ? orderUpdateHandler : refundOrderUpdateHandler}
      >
        상태 업데이트
      </div>
    </div>
  )
}

export default OrderDetails
```

## UserOrderDetails.jsx 작성

```jsx
import React, { useEffect, useState } from 'react'
import { BsFillBagFill } from 'react-icons/bs'
import { Link, useParams } from 'react-router-dom'
import { useDispatch, useSelector } from 'react-redux'
import styles from '../styles/styles'
import { getAllOrdersOfUser } from '../redux/order/orderAction'
import { RxCross1 } from 'react-icons/rx'
import { AiFillStar, AiOutlineStar } from 'react-icons/ai'
import axios from 'axios'
import { toast } from 'react-toastify'

const UserOrderDetails = () => {
  // Redux에서 필요한 상태 및 액션을 가져옴
  const { orders } = useSelector((state) => state.orders)
  const { user } = useSelector((state) => state.user)
  const dispatch = useDispatch()

  // 상태 초기화
  const [open, setOpen] = useState(false)
  const [comment, setComment] = useState('')
  const [selectedItem, setSelectedItem] = useState(null)
  const [rating, setRating] = useState(1)

  // URL에서 주문 ID 파라미터를 가져옴
  const { id } = useParams()

  // 컴포넌트가 마운트되면 해당 유저의 주문 목록을 불러옴
  useEffect(() => {
    dispatch(getAllOrdersOfUser(user._id))
  }, [dispatch])

  // 현재 주문에 대한 데이터를 가져옴
  const data = orders && orders.find((item) => item._id === id)

  // 리뷰 작성 핸들러
  const reviewHandler = async (e) => {
    try {
      // 서버에 리뷰 작성 요청을 보냄
      const res = await axios.put(
        `http://localhost:5000/api/v2/product/create-new-review`,
        {
          user,
          rating,
          comment,
          productId: selectedItem?._id,
          orderId: id,
        },
        { withCredentials: true }
      )

      // 성공적으로 작성되면 토스트 메시지를 띄우고 주문 목록을 갱신
      toast.success(res.data.message)
      dispatch(getAllOrdersOfUser(user._id))

      // 입력 필드 초기화 및 팝업 닫기
      setComment('')
      setRating(1)
      setOpen(false)
    } catch (error) {
      // 에러 발생 시 토스트 메시지를 띄움
      toast.error(error.message)
    }
  }

  // 환불 신청 핸들러
  const refundHandler = async () => {
    try {
      // 서버에 환불 신청 요청을 보냄
      const res = await axios.put(`http://localhost:5000/api/v2/order/order-refund/${id}`, {
        status: '환불 처리 중',
      })

      // 성공적으로 처리되면 토스트 메시지를 띄우고 주문 목록을 갱신
      toast.success(res.data.message)
```

```jsx
await product.save({ validateBeforeSave: false });:

상품 객체(product)를 데이터베이스에 저장한다.

validateBeforeSave: false 옵션은 데이터베이스에 저장하기 전에 데이터 유효성 검사를 비활성화한다.

이는 일부 필수 필드의 누락 등과 같은 유효성 검사 오류를 허용하고, 저장 프로세스를 더 유연하게 만든다.

await Order.findByIdAndUpdate(...):

주문(Order) 모델을 업데이트한다.

findByIdAndUpdate 함수는 MongoDB에서 주어진 조건에 맞는 문서를 찾고 업데이트한다.

첫 번째 매개변수로는 업데이트할 주문의 ID(orderId)를 받는다.

두 번째 매개변수로는 업데이트할 내용을 정의한 객체를 전달한다.

$set 연산자를 사용하여 업데이트할 필드를 정의한다.

여기서는 주문 내의 cart 배열에서 특정 상품("elem._id": productId)의 isReviewed 필드를 true로 설정한다.

( 주문(Order) 객체의 cart 배열에서 특정 상품(elem._id가 productId와 일치하는)을 찾아서 해당 상품의 isReviewed 값을 true로 변경하는

것이라고 할 수 있다. 이렇게 하면 해당 상품에 대한 리뷰가 작성되었다는 표시를 나타낼 수 있다. )

( $[elem]의 값은 업데이트 작업의 arrayFilters 옵션에서 제공된 배열 필터에 의해 결정 )

arrayFilters를 사용하여 cart 배열 내에서 어떤 요소를 업데이트할지를 조건으로 지정한다.

이것은 특정 상품 ID와 일치하는 배열 요소를 찾아 업데이트한다.

new: true 옵션은 업데이트 이후에 업데이트된 문서를 반환하도록 한다.

이 과정을 통해 상품 리뷰 작성 또는 업데이트 후에 해당 주문의 cart 배열에서 해당 상품의 isReviewed 필드가 true로 업데이트되어,

해당 상품이 리뷰가 작성되었다는 표시를 갖게 된다.

* $[elem] 구문은 MongoDB의 위치 지정 연산자의 일부로, 특정 조건을 기반으로 배열 내의 요소를 업데이트할 수 있게 해준다
```

```jsx
        <div className={`${styles.button} text-white`}>메시지 보내기</div>
      </Link>

      <br />
    </div>
  )
}
```

```jsx
// 환불 요청 ---- 사용자
router.put(
  "/order-refund/:id",
  catchAsyncErrors(async (req, res, next) => {
    try {
      const order = await Order.findById(req.params.id);

      if (!order) {
        return next(new ErrorHandler("주문을 찾을 수 없습니다.", 400));
      }

      order.status = req.body.status;

      await order.save({ validateBeforeSave: false });

      res.status(200).json({
        success: true,
        order,
        message: "주문 환불 요청이 성공적으로 완료되었습니다!",
      });
    } catch (error) {
      return next(new ErrorHandler(error.message, 500));
    }
  })
);
```
