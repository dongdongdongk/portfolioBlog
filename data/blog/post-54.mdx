  System.setProperty("webdriver.chrome.driver", "/path/to/chromedriver");      // 뒷쪽에 내 크롬 드라이버 경로를 적어준다

	WebDriver driver = new ChromeDriver();             // 드라이버 객체 생성 (창을 켜지 않는다 같은 옵션을 추가할 때 는 옵션 객체를 만들어서 매개변수로 넣어준다)

  driver.get("http://www.google.com/");              // 이동할 URL 입력

  Thread.sleep(5000);  // 창이 켜지고 5초간 대기

  WebElement searchBox = driver.findElement(By.name("q")); //q 라는 이름을 가진 요소 검색

  searchBox.sendKeys("ChromeDriver");

  searchBox.submit();

  Thread.sleep(5000);  // Let the user actually see something!

  driver.quit();

 }

}
