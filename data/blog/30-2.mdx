```java
//매개 변수를 다르게 하는 생성자 여러개 선언
public class Car {
Car() {초기화}
Car(String model) {}
Car(String model, String color) {}
Car(String model, String color, int maxSpeed){}
}

// 다양한 객체 생성자 호출 코드
Car car1 = new Car();
Car car2 = new Car("그랜저");
Car car3 = new Car("그랜저", "흰색");
Car car4 = new Car("그랜저", "흰색", 300);

이처럼 매개 변수를 다르게 가지는 생성자를 여러개 선언할수 있다

Car(String model, String color) {}
Car(String color, String model) {} // 이렇게 순서만 바꾸는건 생성자 오버로딩이 아니다.
*컴파일 오류
```

```java
Car(Srting model) {
	this.model = model
	this.color = "은색" // 매개변수가 없어 color 값을 받을수 없으니 생성자에서 적어줌 없으면 초기값
	this.maxSpeed = 250;
}

Car(String model, String color) {
	this.model = model;
	this.color = color;
	this.maxSpeed = 250;
}

//중복되는 코드를 this() 로 축소
Car(string model) {
this(model, "은색:, 250);
}

Car(String model, String color) {
this(model, color, 250);
}

//이때 공통 생성자,실행 코드가 있어야 한다 다른 실행자가 이걸 호출한다.(this() 줄여도 x)
Car(String model, String color, int masSpeed) {
this.model = model;
this.color = color;
this.maxSpeed = maxSpeed;
}
```
