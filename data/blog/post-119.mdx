## todoList 추가 , 조회 작성

```jsx
const express = require('express')
const app = express()
const bodyParser = require('body-parser')

app.use(bodyParser.json())
app.use(bodyParser.urlencoded({ extended: true }))

let id = 2
const todoList = [
  {
    id: 1,
    text: '할일 1',
    done: false,
  },
]

app.get('/api/todo', (req, res) => {
  res.json(todoList)
})

app.post('/api/todo', (req, res) => {
  const { text, done } = req.body
  todoList.push({
    id: id++,
    text,
    done,
  })
  return res.send('success')
})

app.listen(3000, () => {
  console.log('Server Start')
})
```

### CORS 오류가 생긴이유?

```jsx
현재 상황은

Client : localhost:3000 ( react 개발서버가 )

Server : localhost:4000

이렇게 되면 두 서버는 오리진이 다르다 ( 오리진이라는 거는 host 와 port 를 포함한 데이터의 출저를 의미하는 거다 )

오리진이 다르다는건

구글과 네이버 같은거다 서로가 서로의 데이터를 막 꺼내간다면 큰 문제다 그래서 일단 막는다가 기본이다

그래서 해결을 하려면 CORS 정책을 풀어줘야 하는데 은행을 예로들면 창구 직원(서버)이 돈을(데이터를) 줄지말지 정하는거지

우리(클라이언트)가 정하는게 아니다

그렇기 때문에 서버에서 CORS 를 풀어야한다

```

import { useEffect, useState } from 'react'

function App() {
  const [todoList, setTodoList] = useState([])

  useEffect(() => {
    fetch('http://localhost:4000/api/todo')
      .then((response) => response.json())
      .then((data) => setTodoList(data))

## 데이터 추가하기

![Untitled](https://res.cloudinary.com/du1acdhxq/image/upload/v1788381435/notion-blog/_/The_Web_Developer_2022/REACT/_/Untitled_13.png)

```jsx
import { useEffect, useState } from 'react'

function App() {
  const [todoList, setTodoList] = useState([])

  useEffect(() => {
    fetch('http://localhost:4000/api/todo')
      .then((response) => response.json())
      .then((data) => setTodoList(data))
  }, [])

  const onSubmitHandler = (e) => {
    e.preventDefault()
    const text = e.target.text.value
    const done = e.target.done.checked
    fetch('http://localhost:4000/api/todo', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        text,
        done,
      }),
    })
  }

  return (
    <div className="App">
      <h1>todoList</h1>
      {/* 데이터 입력 */}
      <form onSubmit={onSubmitHandler}>
        <input name="text" />
        <input name="done" type="checkbox" />
        <input type="submit" value="추가" />
      </form>

      {todoList.map((todo) => (
        <div key={todo.id} style={{ display: 'flex' }}>
          <div>{todo.id}</div>
          <div>{todo.text}</div>
          <div>{todo.done ? 'Y' : 'N'}</div>
        </div>
      ))}
    </div>
  )
}

export default App
```

```jsx
데이터를 추가하는 코드를 작성한다

이벤트 핸들러와 <form> 을 사용해서 데이터를 전송한다

( submit 버튼이 input 으로 사용될 수 있다는게 신기하다 )

지금 이 코드의 문제점은 데이터를 추가하면 바로 화면에 나타나지 않는다는거다
```

### 오류 개선 + 리팩토링

```jsx
import { useEffect, useState } from 'react'

function App() {
  const [todoList, setTodoList] = useState([])

  // 재사용을 위해서 함수를 따로 빼줬다
  const fetchData = () => {
    fetch('http://localhost:4000/api/todo')
      .then((response) => response.json())
      .then((data) => setTodoList(data))
  }

  // 이제 함수를 받아서 실행
  useEffect(() => {
    fetchData()
  }, [])

  const onSubmitHandler = (e) => {
    e.preventDefault()
    const text = e.target.text.value
    const done = e.target.done.checked
    fetch('http://localhost:4000/api/todo', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        text,
        done,
      }),
    }).then(() => fetchData()) // 요청이 끝나면 다시 랜더링
  }

  return (
    <div className="App">
      <h1>todoList</h1>
      {/* 데이터 입력 */}
      <form onSubmit={onSubmitHandler}>
        <input name="text" />
        <input name="done" type="checkbox" />
        <input type="submit" value="추가" />
      </form>

      {todoList.map((todo) => (
        <div key={todo.id} style={{ display: 'flex' }}>
          <div>{todo.id}</div>
          <div>{todo.text}</div>
          <div>{todo.done ? 'Y' : 'N'}</div>
        </div>
      ))}
    </div>
  )
}

export default App
```

```jsx
화면이 바로 나타나게 해줬고 이 과정에서 같은 함수가 중복되게 되어

리팩토링을 진행하였다
```
