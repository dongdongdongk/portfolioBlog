```jsx
// Redux에서 사용할 스토어를 생성합니다.
import { createStore } from 'redux'

// 초기 상태와 액션을 처리하는 리듀서 함수를 정의합니다.
function reducer(currentState, action) {
  // 초기 상태가 없는 경우, 기본적으로 { number: 1 }로 설정합니다.
  if (currentState === undefined) {
    return {
      number: 1,
    }
  }

  // 현재 상태를 복제하여 새로운 상태 객체를 생성합니다.
  const newState = { ...currentState }

  // 'PLUS' 액션이 발생하면 상태의 number 값을 1 증가시킵니다.
  if (action.type === 'PLUS') {
    newState.number++
  }

  // 새로운 상태를 반환합니다.
  return newState
}

// Redux 스토어를 생성하고 리듀서를 전달합니다.
const store = createStore(reducer)

// React에서 Redux를 사용할 수 있게 도와주는 Provider 컴포넌트를 불러옵니다.
import { Provider, useSelector, useDispatch } from 'react-redux'

// 앱의 루트 컴포넌트인 App을 정의합니다.
export default function App() {
  return (
    <div id="container">
      <h1>Root</h1>
      <div id="grid">
        {/* Redux 스토어를 제공하는 Provider 컴포넌트로 감싸 하위 컴포넌트에서 스토어에 접근할 수 있도록 합니다. */}
        <Provider store={store}>
          <Left1></Left1>
          <Right1></Right1>
        </Provider>
      </div>
    </div>
  )
}

// 왼쪽 영역의 세 번째 컴포넌트 Left3를 정의합니다.
function Left3(props) {
  // useSelector 훅을 사용하여 Redux 스토어에서 상태를 가져와서 number 값을 읽어옵니다.
  const number = useSelector((state) => state.number)
  return (
    <div>
      <h1>Left3 : {number}</h1>
    </div>
  )
}

// 오른쪽 영역의 세 번째 컴포넌트 Right3를 정의합니다.
function Right3(props) {
  // useDispatch 훅을 사용하여 액션을 디스패치할 수 있는 dispatch 함수를 가져옵니다.
  const dispatch = useDispatch()
  return (
    <div>
      <h1>Right3</h1>
      {/* '+' 버튼을 클릭하면 dispatch를 사용하여 'PLUS' 액션을 디스패치합니다. */}
      <input
        type="button"
        value="+"
        onClick={() => {
          dispatch({ type: 'PLUS' })
        }}
      ></input>
    </div>
  )
}
```

```jsx
// action-types.js
export const INCREMENT = 'INCREMENT';
export const DECREMENT = 'DECREMENT';

액션 타입 정의

Redux에서는 액션의 타입을 문자열 상수로 정의합니다. 이는 오타나 중복을 방지하고 유지보수를 쉽게 만듭니다.
```

```jsx
// actions.js
import { INCREMENT, DECREMENT } from './action-types';

export const increment = () => ({ type: INCREMENT });
export const decrement = () => ({ type: DECREMENT });

액션 생성자 함수 정의

액션 생성자 함수는 액션을 반환하는 함수입니다.

이 함수를 호출하면 액션 객체가 생성되어 Redux 스토어로 전달됩니다.

type은 필수 속성이며, 액션에 필요한 데이터는 payload 속성에 포함됩니다.

* payload 속성에 포함된다"는 말은 Redux에서 액션 객체를 생성할 때, 액션에 필요한 실제 데이터를 payload라는 속성에 담아 전달한다는 의미

export const addTodo = (text) => ({
  type: 'ADD_TODO',
  payload: {
    text: text,
  },
});

이렇게 함으로써 액션 객체가 가지는 정보가 단순한 type뿐만 아니라 추가적인 데이터도 가질 수 있게 되며,

리듀서에서는 이 데이터를 활용하여 상태를 업데이트할 수 있습니다.
```

```jsx
// reducer.js
import { INCREMENT, DECREMENT } from './action-types';

const initialState = {
  count: 0,
};

const counterReducer = (state = initialState, action) => {
  switch (action.type) {
    case INCREMENT:
      return { ...state, count: state.count + 1 };
    case DECREMENT:
      return { ...state, count: state.count - 1 };
    default:
      return state;
  }
};

export default counterReducer;

리듀서 함수 정의

리듀서 함수는 현재 상태와 액션을 받아 새로운 상태를 반환하는 순수 함수입니다.

여기서 switch 문을 사용하여 액션의 타입에 따라 다른 동작을 수행합니다.
```

```jsx
// index.js
import React from 'react';
import ReactDOM from 'react-dom';
import { createStore } from 'redux';
import { Provider } from 'react-redux';
import counterReducer from './reducer';
import App from './App';

const store = createStore(counterReducer);

ReactDOM.render(
  <Provider store={store}>
    <App />
  </Provider>,
  document.getElementById('root')

스토어 생성 및 Provider로 감싸기

Redux 스토어를 생성하고, Provider 컴포넌트로 앱을 감싸 스토어를 제공합니다.

이렇게 하면 하위 컴포넌트에서 Redux 스토어에 쉽게 접근할 수 있습니다.
```

```jsx
// CounterComponent.js
import React from 'react';
import { useSelector, useDispatch } from 'react-redux';
import { increment, decrement } from './actions';

const CounterComponent = () => {
  const count = useSelector((state) => state.count);
  const dispatch = useDispatch();

  return (
    <div>
      <h1>Count: {count}</h1>
      <button onClick={() => dispatch(increment())}>Increment</button>
      <button onClick={() => dispatch(decrement())}>Decrement</button>
    </div>
  );
};

export default CounterComponent;

컴포넌트에서 useSelector와 useDispatch 사용

useSelector 훅을 사용하여 스토어의 상태를 읽어오고, useDispatch 훅을 사용하여 액션을 디스패치합니다.

버튼 클릭 시 dispatch(increment())를 호출하여 INCREMENT 액션이 발생하고, 리듀서가 이에 반응하여 상태를 업데이트합니다.
```

# **Redux toolkit**

```jsx
Redux : 상태를 관리자 프로그램이 사용하는 상태를 체계적으로 관리

React-Redux : 리액트에서 사용하기 어려운 Redux 를 쓰기 편하게 만들어준 것

Redux toolkit : 그럼에도 사용하기 어려웠다 설정, 미들웨어 설치, 반복되는 코드, 불변성 유지 어려움
```

## **Redux toolkit 의 장점**

```jsx
// 리액트와 리덕스 관련 모듈 불러오기
import React from 'react'
import './style.css'
import { createStore } from 'redux'
import { Provider, useSelector, useDispatch } from 'react-redux'

// 리듀서 함수 정의
function reducer(state, action) {
  // 'up' 액션이 발생하면 현재 값에 action.step만큼 더한 새로운 상태 반환
  if (action.type === 'up') {
    return { ...state, value: state.value + action.step }
  }

  // 액션 타입이 'up'이 아니면 현재 상태 그대로 반환
  return state
}

// 초기 상태 정의
const initialState = { value: 0 }
// createStore를 사용하여 Redux 스토어 생성, 초기 상태와 리듀서 전달
const store = createStore(reducer, initialState)

// 카운터 컴포넌트 정의
function Counter() {
  // useDispatch 훅을 사용하여 디스패치 함수를 가져옴
  const dispatch = useDispatch()
  // useSelector 훅을 사용하여 스토어의 상태에서 필요한 값을 추출
  const count = useSelector((state) => state.value)

  return (
    <div>
      {/* '+' 버튼 클릭 시 'up' 액션을 디스패치하고, action.step에 2를 전달 */}
      <button
        onClick={() => {
          dispatch({ type: 'up', step: 2 })
        }}
      >
        +
      </button>
      {/* 현재 카운터 값 화면에 출력 */}
      {count}
    </div>
  )
}

// 앱 컴포넌트 정의
export default function App() {
  return (
    // Provider 컴포넌트로 애플리케이션을 감싸고 Redux 스토어를 제공
    <Provider store={store}>
      <div>
        {/* Counter 컴포넌트 렌더링 */}
        <Counter></Counter>
      </div>
    </Provider>
  )
}
```

```jsx
'+' 버튼을 누를 때마다 dispatch({ type: 'up', step: 2 }); 코드가 실행되어 'up' 액션이 스토어로 전달된다.

이 변경된 상태는 다시 useSelector를 통해 읽어와서 화면에 렌더링되어, 카운터의 값이 업데이트된다.
```

```jsx
import React from 'react'
import './style.css'
import { createStore } from 'redux'
import { createSlice, configureStore } from '@reduxjs/toolkit'
import { Provider, useSelector, useDispatch } from 'react-redux'

const counterSlice = createSlice({
  // slice 생성 ( 이런 작은 슬라이스를 여러개 만들게 된다 )
  name: 'counter',
  initialState: { value: 0 }, // 초기값 설정

  reducers: {
    // 리듀서를 공급 ( s 가 붙는거 주의 )
    up: (state, action) => {
      // 예전처럼 action.type === 로 받는게 아니라 함수를 받는다
      state.value = state.value + action.payload // 더이상 불변성은 필요없다, payload 로 값을 받게된다 액션 크리에이터 덕분에
    },
  },
})

const store = configureStore({
  // 작은 slice 들을 모아서 store 로 만들때는 cofigureStore 를 사용한다
  reducer: {
    // configureStore 는 reducer 이 필수적으로 필요하며
    counter: counterSlice.reducer, // 이는 위의 up 같은 함수들이 모여서 하나의 reducer을 자동으로 만들어 주는데 그렇게 만들어 진게 counter : counterSlice.reducer 이거다 ( counter 에 대한 reducer )
  },
})

//   }

function Counter() {
  const dispatch = useDispatch()
  const count = useSelector((state) => state.counter.value) // 여기는 .counter 을 추가해 줬는데 counter 은 위의 configureStore 에 선언한 counter 을 말하는 것이다
  return (
    <div>
      <button
        onClick={() => {
          // dispatch({type : 'counterSlice/up', step : 2}); // type 은 counterSlice 에 속해있는 redusers 에 up 을 선택 한다 라는 뜻
          dispatch(counterSlice.actions.up(2)) // 이렇게 더 쉽게 사용도 가능하다 이는 툴킷의 액션 크리에이터를 사용한것인데 액션 크리에이터는 자동으로 reducers 를 참고해서 액션을 만들어 낸다 up 에서 보낸값은 payload 라는 값으로 전달된다
        }}
      >
        +
      </button>{' '}
      {count}
    </div>
  )
}

// 같은 방식으로 Provider 로 전달하며 store={store} 를 사용해서 값을 전달해준다
export default function App() {
  return (
    <Provider store={store}>
      <div>

      </div>
    </Provider>
  )
```

```jsx
**Redux Toolkit의 createSlice:**

createSlice 함수는 Redux Toolkit에서 제공하는 편리한 함수로, 리듀서 및 액션을 한 번에 생성할 수 있다.

name:

슬라이스의 이름을 지정한다. 여기서는 "counter"로 정의되었다.

initialState:

슬라이스의 초기 상태를 설정합니다. value 속성이 0인 객체로 초기화되었다.

reducers:

액션을 처리하는 리듀서 함수를 정의하는 객체입니다. up라는 액션을 처리하는 리듀서를 정의하였다.

**Redux Toolkit의 configureStore:**

configureStore 함수는 Redux Toolkit에서 제공하는 함수로, 스토어를 구성하는 데 사용되었다.

reducer:

스토어에 등록할 리듀서를 지정합니다. 여기서는 counterSlice.reducer로 단일 리듀서를 사용했다.

**React-Redux Hooks:**

useSelector:

React 컴포넌트에서 Redux 스토어의 상태를 읽어올 때 사용하는 Hook이다.

여기서는 state.counter.value 대신에 state.counter.value를 간단하게 사용할 수 있다.

useDispatch:

React 컴포넌트에서 Redux 스토어로 액션을 디스패치할 때 사용하는 Hook이다.

dispatch 함수를 반환하며, 여기서는 up 액션을 디스패치하기 위해 사용되었다.

**액션 크리에이터 (counterSlice.actions):**

Redux Toolkit의 createSlice에서 생성된 슬라이스에는 자동으로 액션 크리에이터가 포함된다.

이를 통해 액션을 쉽게 생성할 수 있다.

counterSlice.actions.up(2):

up 액션을 생성하고, 액션의 payload 속성에 2를 전달하여 액션을 디스패치한다.

액션 크리에이터를 사용하면 불필요한 오타 등을 방지할 수 있다.

**Redux의 핵심 개념:**

액션(Action):

스토어에서 상태 변화를 일으키는 객체이다. 여기서는 up라는 액션이 정의되었다.

리듀서(Reducer):

현재 상태와 액션을 받아 새로운 상태를 반환하는 함수입니다. 여기서는 counterSlice.reducer에 up 액션을 처리하는 리듀서가 정의되었다.

스토어(Store):

애플리케이션의 상태를 저장하고 관리하는 객체이다. configureStore로 스토어를 설정하였다.

Hooks (useSelector 및 useDispatch):

React 컴포넌트에서 Redux 기능을 사용하기 위한 편리한 함수이다.
```

```jsx
* 액션(Action):

액션은 어떤 상태 변화가 필요한지를 나타내는 객체이다.

액션은 반드시 type이라는 속성을 가지고 있어야 하며, 필요에 따라 추가적인 데이터를 포함할 수 있다.
```

```jsx
사용자 관련 액션과 리듀서를 정의합니다. redux/userSlice.js 파일을 만들어서 다음과 같이 작성한다
```

```jsx
// redux/userSlice.js

// Redux Toolkit에서 createSlice와 createAsyncThunk를 가져옵니다.
import { createSlice, createAsyncThunk } from '@reduxjs/toolkit'

// 가상의 로그인 API 함수를 정의합니다.
const fakeLoginApi = async (username, password) => {
  // 여기에서 실제 로그인 로직을 구현합니다.
  // 예를 들어, 서버에 로그인 요청을 보내고 응답을 받아오는 등의 작업을 수행합니다.
  return { username, token: 'fakeToken' }
}

// createAsyncThunk를 사용하여 비동기 작업을 처리하는 액션 생성자를 정의합니다.
export const loginUser = createAsyncThunk('user/login', async (credentials) => {
  // 비동기 작업이 진행되는 부분입니다.
  // fakeLoginApi 함수를 호출하여 가상의 로그인 요청을 보내고 응답을 받아옵니다.
  const response = await fakeLoginApi(credentials.username, credentials.password)
  return response // 받아온 응답은 액션 페이로드로 반환됩니다.
})

// 사용자 정보를 담은 초기 상태를 정의합니다.
const initialState = {
  user: null, // 로그인된 사용자의 정보를 담을 변수
  status: 'idle', // 비동기 작업 상태를 나타내는 변수 ('idle', 'loading', 'succeeded', 'failed' 중 하나)
  error: null, // 비동기 작업 중 에러 발생 시 에러 정보를 담을 변수
}

// createSlice를 사용하여 리듀서와 액션을 생성합니다.
const userSlice = createSlice({
  name: 'user', // Slice의 이름을 지정합니다.
  initialState, // 초기 상태를 전달합니다.
  reducers: {}, // 추가적인 동기 액션 생성자는 여기에 정의할 수 있습니다.
  extraReducers: (builder) => {
    // createAsyncThunk에서 생성된 액션에 대한 리듀서를 추가합니다.
    builder
      .addCase(loginUser.pending, (state) => {
        // 로그인 요청이 보내지고 있는 동안 상태를 'loading'으로 업데이트합니다.
        state.status = 'loading'
      })
      .addCase(loginUser.fulfilled, (state, action) => {
        // 로그인 요청이 성공하면 상태를 'succeeded'로 업데이트하고 사용자 정보를 저장합니다.
        state.status = 'succeeded'
        state.user = action.payload
      })
      .addCase(loginUser.rejected, (state, action) => {
        // 로그인 요청이 실패하면 상태를 'failed'로 업데이트하고 에러 정보를 저장합니다.
        state.status = 'failed'
        state.error = action.error.message
      })
  },
})

export default userSlice.reducer // 생성된 Slice의 리듀서를 내보냅니다.
```

```jsx
초기 상태 정의하기:

initialState라는 변수에 상태의 초기값을 정의한다. 이 값은 Redux 스토어의 초기 상태로 사용된다.

const initialState = {
  user: null,
  status: 'idle',
  error: null,
};

액션 생성자는 액션을 생성하는 함수로, 여기서는 createAsyncThunk를 사용하여 loginUser 액션을 생성한다.

  // 비동기 작업 수행 후 응답을 액션 페이로드로 반환
  const response = await fakeLoginApi(credentials.username, credentials.password);
  return response;
});

------------------------------------------------------------------------------------------------

리듀서 생성:

createSlice 함수를 사용하여 리듀서와 관련된 액션 생성자와 액션을 한 번에 정의할 수 있다.

액션 핸들링:

extraReducers 내에서 각 액션에 대한 상태 변경 로직을 작성한다.

pending, fulfilled, rejected 등은 createAsyncThunk로 생성된 액션의 서로 다른 상태를 나타낸다.

  name: 'user',         // Slice의 이름
  initialState,         // 초기 상태
  reducers: {},         // 추가적인 동기 액션 생성자 정의 가능

        state.status = 'loading';

        state.status = 'failed';
        state.error = action.error.message;

});
```

```jsx
// components/Login.js

// React와 useState 훅을 가져옵니다.
import React, { useState } from 'react'

// react-redux에서 useDispatch와 useSelector 훅을 가져와서 사용합니다.
import { useDispatch, useSelector } from 'react-redux'

// Redux 스토어에서 정의한 loginUser 액션 생성자를 가져옵니다.
import { loginUser } from '../redux/userSlice'

// Login 컴포넌트를 정의합니다.
const Login = () => {
  // useDispatch 훅을 사용하여 액션 디스패치 함수를 가져옵니다.
  const dispatch = useDispatch()

  // useSelector 훅을 사용하여 Redux 스토어의 상태를 조회합니다.
  // state.user.user는 사용자 정보, state.user.status는 비동기 작업 상태, state.user.error는 에러 정보를 나타냅니다.
  const user = useSelector((state) => state.user.user)
  const status = useSelector((state) => state.user.status)
  const error = useSelector((state) => state.user.error)

  // 로그인 폼의 입력값을 관리하기 위한 상태 변수들을 정의합니다.
  const [username, setUsername] = useState('')
  const [password, setPassword] = useState('')

  // handleLogin 함수는 로그인 버튼이 클릭되었을 때 호출되는 함수입니다.
  const handleLogin = () => {
    // loginUser 액션을 디스패치하여 로그인 요청을 시작합니다.
    dispatch(loginUser({ username, password }))
  }

  // JSX를 통해 로그인 폼을 렌더링합니다.
  return (
    <div>
      <h2>Login</h2>

      {/* 로딩 중일 때는 'Loading...' 메시지를 표시합니다. */}
      {status === 'loading' && <p>Loading...</p>}

      {/* 로그인이 실패했을 때는 에러 메시지를 표시합니다. */}
      {status === 'failed' && <p>Error: {error}</p>}

      {/* 로그인이 성공했을 때는 환영 메시지를 표시합니다. */}
      {status === 'succeeded' && <p>Welcome, {user.username}!</p>}

      {/* 사용자명과 비밀번호를 입력할 수 있는 인풋 요소들과 로그인 버튼을 표시합니다. */}
      <input
        type="text"
        placeholder="Username"
        value={username}
        onChange={(e) => setUsername(e.target.value)}
      />
      <input
        type="password"
        placeholder="Password"
        value={password}
        onChange={(e) => setPassword(e.target.value)}
      />
      <button onClick={handleLogin}>Login</button>
    </div>
  )
}

// Login 컴포넌트를 내보냅니다.
export default Login
```

```jsx
  // useSelector 훅을 사용하여 Redux 스토어의 상태 조회
  const count = useSelector((state) => state.counter.value)

  return (
    <div>
      <h2>Counter: {count}</h2>

      {/* 액션 디스패치하여 카운터 증가 */}
      <button onClick={() => dispatch(increment())}>Increment</button>

      {/* 액션 디스패치하여 카운터 감소 */}
      <button onClick={() => dispatch(decrement())}>Decrement</button>
    </div>
  )
}

export default Counter
```

```jsx
useDispatch 훅을 사용하여 액션 디스패치 함수를 가져온다.

useSelector 훅을 사용하여 Redux 스토어의 상태를 조회한다.

버튼 클릭 시 각각의 액션을 디스패치하여 상태를 변경한다.

상태의 변경으로 인해 컴포넌트가 리렌더링되고, 변경된 상태가 화면에 반영된다.
```

## 동기 예제 2

```jsx
Redux Toolkit의 createAsyncThunk와 createSlice를 활용하여 간단한 비동기 작업을 수행하는 리덕스 코드
```

```jsx
import { createSlice, createAsyncThunk } from '@reduxjs/toolkit'
import axios from 'axios'
import { server } from '../../server'

// 비동기 액션 생성자 정의
export const fetchData = createAsyncThunk('example/fetchData', async (_, { dispatch }) => {
  try {
    dispatch(dataRequest())

    // 여기에서 비동기 작업 수행 (예: API 호출)
    const { data } = await axios.get(`${server}/api/data`)

    dispatch(dataSuccess(data))
  } catch (error) {
    dispatch(dataFail(error.response.data.message))
  }
})

// 초기 상태 정의
const initialState = {
  isLoading: false,
  data: null,
  error: null,
}

// 슬라이스 생성
const exampleSlice = createSlice({
  name: 'example',
  initialState,
  reducers: {
    dataRequest: (state) => {
      state.isLoading = true
    },
    dataSuccess: (state, action) => {
      state.isLoading = false
      state.data = action.payload
      state.error = null
    },
    dataFail: (state, action) => {
      state.isLoading = false
      state.error = action.payload
    },
  },
})

// 액션 생성자 추출
export const { dataRequest, dataSuccess, dataFail } = exampleSlice.actions

// 리듀서 내보내기
export default exampleSlice.reducer
```

```jsx
import { useDispatch, useSelector } from 'react-redux'
import { fetchData } from './exampleSlice'

// 컴포넌트에서 사용하는 부분
const ExampleComponent = () => {
  const dispatch = useDispatch()
  const { isLoading, data, error } = useSelector((state) => state.example)

  const fetchDataHandler = () => {
    dispatch(fetchData())
  }

  return (
    <div>
      <button onClick={fetchDataHandler}>데이터 불러오기</button>
      {isLoading && <p>Loading...</p>}
      {data && <p>{data}</p>}
      {error && <p>Error: {error}</p>}
    </div>
  )
}
```

```jsx
비동기 액션 생성자 (createAsyncThunk):

createAsyncThunk를 사용하여 fetchData라는 비동기 액션 생성자를 만든다.

이 액션 생성자는 데이터를 가져오는 비동기 작업을 수행하고, 상태를 업데이트하는 역할을 한다.

async (_, { dispatch }) => { 부분은 비동기 작업을 시작하는 부분이다. 여기서는 _를 사용하여 첫 번째 파라미터를 무시하고,

dispatch 함수를 두 번째 파라미터로 추출한다.

비동기 작업이 성공하면 dispatch(dataSuccess(data));를 호출하여 상태를 업데이트하고,

실패하면 dispatch(dataFail(error.response.data.message));를 호출하여 에러를 처리한다.

리듀서 (createSlice):

createSlice를 사용하여 슬라이스를 정의합니다. 슬라이스는 상태와 리듀서를 함께 정의하는 방식이다.

dataRequest, dataSuccess, dataFail이라는 리듀서 함수를 정의합니다. 각각은 비동기 액션에서 사용되며, 상태를 업데이트하는 역할을 합니다.

초기 상태 (initialState):

초기 상태는 isLoading이 false, data가 null, error가 null인 객체입니다.

useSelector를 사용한 상태 선택:

컴포넌트에서 useSelector를 사용하여 슬라이스의 상태를 선택합니다.
const { isLoading, data, error } = useSelector((state) => state.example); 부분은 example 슬라이스의 상태를 선택하여 사용합니다.

컴포넌트에서 액션 디스패치:

컴포넌트에서 useDispatch를 사용하여 fetchData 액션을 디스패치합니다.

버튼 클릭 시 dispatch(fetchData());를 호출하여 비동기 작업을 시작합니다.

* 첫 번째 파라미터 (_): 언더스코어(_)는 일반적으로 사용하지 않는 변수를 나타낸다.

여기에서는 이 함수에서 사용하지 않는 것을 나타내기 위해 언더스코어를 사용 (실제로 이 부분에는 어떤 값도 전달되지 않는다.)
```

### 주요문법

```jsx
Redux Toolkit의 createAsyncThunk:

createAsyncThunk는 비동기 액션 생성자를 생성하는 유틸리티입니다.

액션 생성자의 첫 번째 인자는 비동기 작업의 인자를 받고, 두 번째 인자는 Redux Toolkit에서 제공하는 thunkAPI로써,

여기서는 { dispatch }를 비구조화 할당하여 사용했습니다.

Redux Toolkit의 createSlice:

createSlice는 슬라이스를 생성하는 함수로, 상태와 리듀서를 포함합니다.

reducers 객체를 사용하여 여러 개의 액션을 다룹니다.

비동기 작업 처리:

비동기 작업은 async/await 문법을 사용하여 처리하며, 비동기 액션 생성자에서 dispatch 함수를 사용하여 상태를 업데이트합니다.

useSelector와 useDispatch 훅 사용:

useSelector는 리덕스 스토어의 상태를 선택하는데 사용됩니다.

useDispatch는 액션을 디스패치하는데 사용됩니다.
```
