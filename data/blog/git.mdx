# 실제 예시와 명령어

```jsx
# 1. Git 저장소 초기화 (처음 한 번)

		git init

# 2. 파일 생성 또는 수정

		echo "console.log('Hello, Git!');" > index.js

# 3. 변경된 파일 확인

		git status

# 4. 변경사항을 staging area에 추가

		git add index.js

# 또는 모든 변경사항 추가

		git add .

# 5. 커밋 생성 (의미 있는 메시지를 작성)

		git commit -m "feat: create index.js with hello message"

# 6. 원격 저장소 연결 (처음 한 번만 실행)

		git remote add origin https://github.com/username/repo.git

# 7. 커밋 푸시 (main 브랜치로 푸시)

		git push origin main

# 브랜치 이름이 master일 경우

		# git push origin master

```

```jsx
# 프로젝트 디렉토리로 이동

# 변경 사항 스테이징

git add .

# 변경 사항 커밋

git commit -m "Add Base64Converter component and update page layout"

# GitHub에 푸시 (기본 브랜치: main)

git push origin main
```

## 브랜치 전략 스테이징 제외 예시

```jsx
#!/bin/bash

# 프로젝트 디렉토리로 이동

cd /home/user/base64-web-app

# .gitignore에 node_modules, .next, .env.local 추가 (이미 있으면 무시됨)

echo "node_modules/" >> .gitignore

echo ".next/" >> .gitignore

echo ".env.local" >> .gitignore

# Git 사용자 정보 설정

git config --global user.name "Jane Smith"

git config --global user.email "jane.smith@example.com"

# 원격 저장소 설정 (이미 설정되어 있으면 무시됨)

git remote add origin https://github.com/janesmith/base64-web-app.git

# develop 브랜치로 전환 (없으면 생성)

git checkout develop || git checkout -b develop

git push origin develop

# feature/add-geolocation 브랜치 생성 및 전환

git checkout -b feature/add-geolocation

# 변경 사항 확인

echo "변경 사항 확인:"

git status

# 특정 파일만 스테이징 (src/app/page.tsx만 추가, Base64Converter.tsx는 제외)

git add src/app/page.tsx

# 만약 실수로 다른 파일이 스테이징되었다면 제외

git restore --staged src/components/Base64Converter.tsx

# 스테이징 상태 확인

echo "스테이징된 파일:"

git status

# 변경 사항 커밋

git commit -m "Update page.tsx for geolocation feature"

# feature 브랜치 푸시

git push origin feature/add-geolocation

# develop 브랜치로 전환

git checkout develop

# feature 브랜치를 develop에 병합

git merge feature/add-geolocation --no-ff -m "Merge feature/add-geolocation into develop"

# develop 브랜치 푸시

git push origin develop

# (옵션) feature 브랜치 삭제 (로컬 및 원격)

git branch -d feature/add-geolocation

git push origin --delete feature/add-geolocation
```

## 로컬 vs 원격 브랜치 차이 확인

```jsx
# 1. 원격 최신 상태 가져오기
git fetch

# 2. 커밋 차이 확인
git log HEAD..origin/main --oneline      # 내가 pull 받을 커밋
git log origin/main..HEAD --oneline      # 내가 push 안한 커밋

# 3. 변경된 코드 직접 비교
git diff origin/main

# 4. 요약 상태 확인
git status
```
