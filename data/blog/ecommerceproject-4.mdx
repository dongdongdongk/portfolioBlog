# Cart.jsx ( 장바구니 컴포넌트 작성 )

![Untitled](https://res.cloudinary.com/du1acdhxq/image/upload/v1788381462/notion-blog/_/_/EcommerceProject/EcommerceProject_-4/Untitled.png)

```jsx
import React, { useState } from 'react'
import { RxCross1 } from 'react-icons/rx'
import { IoBagHandleOutline } from 'react-icons/io5'
import { HiOutlineMinus, HiPlus } from 'react-icons/hi'
import styles from '../../styles/styles'
import { Link } from 'react-router-dom'
import { useDispatch, useSelector } from 'react-redux'
import { addToCart, removeFromCart } from '../../redux/cart/cartAction'
import { toast } from 'react-toastify'

// 카트 팝업 컴포넌트
const Cart = ({ setOpenCart }) => {
  // Redux에서 카트 정보 가져오기
  const { cart } = useSelector((state) => state.cart)
  const dispatch = useDispatch()

  // 카트에서 상품 삭제 핸들러
  const removeFromCartHandler = (data) => {
    dispatch(removeFromCart(data))
  }

  // 카트에 담긴 상품의 총 가격 계산
  const totalPrice = cart.reduce((acc, item) => acc + item.qty * item.discountPrice, 0)

  // 상품 수량 변경 핸들러
  const quantityChangeHandler = (data) => {
    dispatch(addToCart(data))
  }

  return (
    <div className="fixed top-0 left-0 z-10 h-screen w-full bg-[#0000004b]">
      <div className="fixed top-0 right-0 flex h-full w-[25%] flex-col justify-between overflow-y-scroll bg-white shadow-sm">
        {cart && cart.length === 0 ? (
          <div className="flex h-screen w-full items-center justify-center">
            <div className="fixed top-3 right-3 flex w-full justify-end pt-5 pr-5">
              <RxCross1 size={25} className="cursor-pointer" onClick={() => setOpenCart(false)} />
            </div>
            <h5>장바구니에 상품이 없습니다!</h5>
          </div>
        ) : (
          <>
            <div>
              <div className="flex w-full justify-end pt-5 pr-5">
                <RxCross1 size={25} className="cursor-pointer" onClick={() => setOpenCart(false)} />
              </div>
              {/* 장바구니 아이템 개수 표시 */}
              <div className={`${styles.noramlFlex} p-4`}>
                <IoBagHandleOutline size={25} />
                <h5 className="pl-2 text-[20px] font-[500]">{cart && cart.length} items</h5>
              </div>

              {/* 카트에 담긴 상품 목록 */}
              <br />
              <div className="w-full border-t">
                {cart &&
                  cart.map((i, index) => (
                    <CartSingle
                      key={index}
                      data={i}
                      quantityChangeHandler={quantityChangeHandler}
                      removeFromCartHandler={removeFromCartHandler}
                    />
                  ))}
              </div>
            </div>

            <div className="mb-3 px-5">
              {/* 체크아웃 버튼 */}
              <Link to="/checkout">
                <div
                  className={`flex h-[45px] w-[100%] items-center justify-center rounded-[5px] bg-[#e44343]`}
                >
                  <h1 className="text-[18px] font-[600] text-[#fff]">
                    Checkout Now ({totalPrice}원)
                  </h1>
                </div>
              </Link>
            </div>
          </>
        )}
      </div>
    </div>
  )
}

// 카트에 담긴 개별 상품 표시 컴포넌트
const CartSingle = ({ data, quantityChangeHandler, removeFromCartHandler }) => {
  // 로컬 상태 사용 (수량 변경 시 반영)
  const [value, setValue] = useState(data.qty)
  const totalPrice = data.discountPrice * value

  // 상품 수량 증가 함수
  const increment = (data) => {
    if (data.stock < value) {
      toast.error('상품 제고를 초과하였습니다!')
    } else {
      setValue(value + 1)
      const updateCartData = { ...data, qty: value + 1 }
      quantityChangeHandler(updateCartData)
    }
  }

  // 상품 수량 감소 함수
  const decrement = (data) => {
    setValue(value === 1 ? 1 : value - 1)
    const updateCartData = { ...data, qty: value === 1 ? 1 : value - 1 }
    quantityChangeHandler(updateCartData)
  }

  return (
    <div className="border-b p-4">
      <div className="flex w-full items-center">
        <div>
          {/* 수량 증가 버튼 */}
          <div
            className={`h-[25px] w-[25px] rounded-full border border-[#e4434373] bg-[#e44343] ${styles.noramlFlex} cursor-pointer justify-center`}
            onClick={() => increment(data)}
          >
            <HiPlus size={18} color="#fff" />
          </div>
          {/* 수량 표시 */}
          <span className="pl-[10px]">{data.qty}</span>
          {/* 수량 감소 버튼 */}
          <div
            className="flex h-[25px] w-[25px] cursor-pointer items-center justify-center rounded-full bg-[#a7abb14f]"
            onClick={() => decrement(data)}
          >
            <HiOutlineMinus size={16} color="#7d879c" />
          </div>
        </div>
        {/* 상품 이미지 */}
        <img
          src={`http://localhost:5000/${data?.images[0]}`}
          alt=""
          className="mr-2 ml-2 h-min w-[130px] rounded-[5px]"
        />
        <div className="pl-[5px]">
          {/* 상품 정보 및 가격 */}
          <h1>{data.name}</h1>
          <h4 className="text-[15px] font-[400] text-[#00000082]">
            ${data.discountPrice} * {value}
          </h4>
          {/* 총 가격 */}
          <h4 className="font-Roboto pt-[3px] text-[17px] font-[600] text-[#d02222]">
            {totalPrice} 원
          </h4>
        </div>
        {/* 카트에서 상품 삭제 버튼 */}
        <RxCross1 className="cursor-pointer" onClick={() => removeFromCartHandler(data)} />
      </div>
    </div>
  )
}

export default Cart
```

```jsx
장바구니 팝업 (Cart 컴포넌트):

Redux를 사용하여 전역 상태인 장바구니 정보를 가져와 화면에 표시한다.

장바구니가 비어있으면 비어있다는 안내 메시지가 나타난다.

장바구니에 상품이 있는 경우 각 상품의 정보를 나타내고, 총 가격을 계산하여 표시한다.

상품 수량을 증가 또는 감소시킬 수 있으며, 해당 변경은 Redux를 통해 전역 상태에 반영된다.

Checkout Now 버튼을 클릭하면 결제 페이지로 이동한다.

장바구니 개별 상품 표시 (CartSingle 컴포넌트):
```

# Ratings.jsx 컴포넌트 추가 ( 별 표시 )

```jsx
import React from 'react'
import { AiFillStar, AiOutlineStar } from 'react-icons/ai'
import { BsStarHalf } from 'react-icons/bs'

// Ratings 컴포넌트
const Ratings = ({ rating }) => {
  // 별 아이콘을 담을 배열 초기화
  const stars = []

  // 별 아이콘을 반복문을 통해 생성
  for (let i = 1; i <= 5; i++) {
    if (i <= rating) {
      // 현재 인덱스가 평점 이하인 경우, 채워진 별 아이콘 추가
      stars.push(<AiFillStar key={i} size={20} color="#f6b100" className="mr-2 cursor-pointer" />)
    } else if (i === Math.ceil(rating) && !Number.isInteger(rating)) {
      // 소수점이 존재하는 경우, 반 채워진 별 아이콘 추가
      stars.push(<BsStarHalf key={i} size={17} color="#f6ba00" className="mr-2 cursor-pointer" />)
    } else {
      // 나머지 경우 빈 별 아이콘 추가
      stars.push(
        <AiOutlineStar key={i} size={20} color="#f6ba00" className="mr-2 cursor-pointer" />
      )
    }
  }

  // 생성된 별 아이콘을 화면에 표시
  return <div className="flex"> {stars}</div>
}

export default Ratings
```
