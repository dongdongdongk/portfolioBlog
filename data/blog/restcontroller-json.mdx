```java
package org.zerock.jpaboard.service;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.zerock.jpaboard.dto.ReplyDTO;
import org.zerock.jpaboard.entity.Board;
import org.zerock.jpaboard.entity.Reply;
import org.zerock.jpaboard.repository.ReplyRepository;

import java.util.List;
import java.util.stream.Collectors;

@Service
@RequiredArgsConstructor
public class ReplyServiceImpl implements ReplyService{

    private final ReplyRepository replyRepository;

    // 댓글 등록
    @Override
    public Long register(ReplyDTO replyDTO) {

        Reply reply = dtoToEntity(replyDTO);

        replyRepository.save(reply);

        return reply.getRno();
    }

    // 댓글 리스트 불러오기
    @Override
    public List<ReplyDTO> getList(Long bno) {

        List<Reply> result = replyRepository.getRepliesByBoardOrderByRno(Board.builder().bno(bno).build());

        return result.stream().map(reply -> entityToDTO(reply)).collect(Collectors.toList());
    }

    // 댓글 수정
    @Override
    public void modify(ReplyDTO replyDTO) {

        Reply reply = dtoToEntity(replyDTO);

        replyRepository.save(reply);

    }

    // 댓글 삭제
    @Override
    public void remove(Long rno) {

        replyRepository.deleteById(rno);

    }

}
```

## 댓글 조회 컨트롤러

```jsx
@RestController
@RequestMapping("/replies")
@Slf4j
@RequiredArgsConstructor
public class ReplyController {

    private final ReplyService replyService; // 자동주입을 위해 final

    @GetMapping(value =  "/board/{bno}",produces = MediaType.APPLICATION_JSON_VALUE)
    public ResponseEntity<List<ReplyDTO>> getListByBoard(@PathVariable("bno") Long bno) {

        log.info("bno: " + bno);

        return new ResponseEntity<>(replyService.getList(bno), HttpStatus.OK);

    }

}
```

```jsx
* @RestContriller 의 경우 모든 메서드이 리턴 타입은 기본으로 JSON을 사용한다

* ResponseEntity 라는 객체를 이용하면 HTTP 의 상테 코드등을 같이 전달 가능하다

* URL 의 변수에 {bno] 를 이용하는데 @PathVariable 이라는 걸 활용하면 /replies/board/100 과 같이 특정 게시물 번호로 조회할 때

	100 이라는 데이터를 변수로 처리하는것이 가능해진다
```

## 댓글 상세 출력 추가

![Untitled](<@RestController%EC%99%80%20JSON%20%EC%B2%98%EB%A6%AC(%EB%8C%93%EA%B8%80)/Untitled%2015.png>)

```jsx
댓글을 클릭하면 댓글의 상세 내용을 모달창에 넣어서 보여준다 여기서는 삭제와 수정이 가능하다

closest 을 사용하여 내부 요소를 클릭해도 상위에서 card-body 를 찾도록 하고

댓글이 가지고 있는 속성중 하나인 data-rno 를 사용해서 값을 가져온다

( 모달창에 숨겨둔 rno는 후에 삭제등에 사용한다 )
```

```jsx
// 댓글 클릭 상세출력
listGroup.addEventListener('click', function (event) {
  // 넓은 클릭을 위해 부모 요소들을 거슬러 올라가면서 .card-body 클래스를 가진 요소를 찾도록 closest
  if (event.target.closest('.card-body')) {
    //dataset.rno 는 data-rno 속성의 값을 가져오는것
    let rno = event.target.dataset.rno

    let cardTitleElement = event.target.querySelector('.card-title')
    let cardSubTitleElement = event.target.querySelector('.card-subtitle')
    if (cardTitleElement) {
      // replyText의 값을 클릭된 "card-body"의 .card-title HTML 내용으로 설정
      replyText.value = cardTitleElement.innerHTML
    }
    if (cardSubTitleElement) {
      replyer.value = cardSubTitleElement.innerHTML
    }

    let inputRno = document.querySelector("input[name ='rno']")
    inputRno.value = rno

    // modal-footer 클래스 내의 모든 버튼을 숨김
    // document.querySelectorAll(".modal-footer .btn").forEach(function(button) {
    //     button.style.display = "none";
    // });
    // .replySave와 .replyRemove 버튼을 보이도록 변경
    // document.querySelector(".replySave").style.display = "block";
    // document.querySelector(".replyRemove").style.display = "block";

    modal.style.display = 'block'
  }
})
```

## 댓글 삭제 기능

![Untitled](<@RestController%EC%99%80%20JSON%20%EC%B2%98%EB%A6%AC(%EB%8C%93%EA%B8%80)/Untitled%2016.png>)

```jsx
댓글을 클릭하고 모달창이 뜨면 Remove 로 댓글을 삭제한다

아까 숨겨운 rno 값을 가져와서 대상을 지정하며

( 현재 서버의 응답 객체가 json 형태가 아닌 success 여서 조금더 복잡해졌다 )
```

```jsx
// 댓글 삭제기능
document.querySelector('.replyRemove').addEventListener('click', function () {
  let rno = document.querySelector("input[name='rno']").value

  fetch('/replies/' + rno, {
    method: 'DELETE',
  })
    .then((response) => {
      if (!response.ok) {
        throw new Error('Network response was not ok')
      }
      // 여기서 JSON 형식이 아니면 문자열로 처리(에러가 발생하면 .catch() success 를 반환)
      return response.json().catch(() => 'success')
    })
    .then((data) => {
      if (data === 'success') {
        // 성공적으로 삭제되었을 때의 처리
        alert(rno + '번 삭제')
        modal.style.display = 'none'
        listGroup.innerHTML = ''
        loadJSONDate()
      } else {
        // JSON 형식이 아닌 경우의 처리
        console.error('Invalid JSON response:', data)
      }
    })
    .catch((error) => {
      console.error('Delete error:', error)
    })
})
```

## 댓글 수정 처리

![Untitled](<@RestController%EC%99%80%20JSON%20%EC%B2%98%EB%A6%AC(%EB%8C%93%EA%B8%80)/Untitled%2017.png>)

```jsx
댓글 수정 기능을 구현한다 ( 아까전에 위에서 버튼 숨긴거 다 보이게 해주자 )

댓글 객체를 생성해서 fetch 로 보내주며

엄밀히 말하면 Delete 랑 거의 똑같다
```

```jsx
// 댓글 수정 기능
document.querySelector('.replyModify').addEventListener('click', function () {
  let rno = document.querySelector("input[name = 'rno']").value
  let text = document.querySelector("input[name = 'replyText']").value
  let replyer = document.querySelector("input[name = 'replyer']").value

  // 댓글 객체 생성
  let reply = {
    rno: rno,
    bno: bno,
    text: text,
    replyer: replyer,
  }

  fetch('/replies/' + rno, {
    method: 'PUT',
    // 서버에게 클라이언트가 보낸 데이터가 JSON 형식이라는 것을 알려주는 것
    headers: {
      'Content-Type': 'application/json',
    },
    //  reply 객체를 JSON 형식의 문자열로 변환 후 body 에 담아서 보낸다
    body: JSON.stringify(reply),
  })
    .then((response) => {
      if (!response.ok) {
        throw new Error('연결실패')
      }
      //  response에서 JSON 추출을 시도하고, 만약 에러가 발생하면 "success"를 반환
      return response.json().catch(() => 'success')
    })
    .then((data) => {
      if (data === 'success') {
        //등록 완료시 처리
        alert(rno + '번 수정 완료')
        modal.style.display = 'none'
        listGroup.innerHTML = ''
        loadJSONDate()
      } else {
        // JSON 형식이 아닌 경우의 처리
        console.error('Invalid JSON response:', data)
      }
    })
    .catch((error) => {
      console.error('Modify error:', error)
    })
})
```
